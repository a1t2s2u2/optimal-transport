\chapter{最適輸送の拡張}
\label{c-extensions}

本章では，最適輸送の Kantorovich 定式化と関連する（そして同じ構造を共有する）いくつかの変分問題を詳述する．目標は，最適輸送をより一般的な設定へ拡張することである：複数の入力ヒストグラムや測度，正規化されていないもの，より一般的な測度のクラス，そして総輸送コストではなく局所的な正則性（源測度において近い点は，目標測度においても近い点へ写されるべきである）に着目した測度間の最適輸送であり，これら二つの測度が異なる距離空間に存在する場合も含む．

\section{多周辺問題}
\label{sec-multimarginal}

Kantorovich 定式化~\eqref{eq-mk-discr} を用いて二つの入力ヒストグラムをカップリングする代わりに，$\a_s \in \simplex_{n_s}$ である $S$ 個のヒストグラム $(\a_s)_{s=1}^S$ を，次の多周辺問題を解くことでカップリングすることができる：
\eql{\label{eq-multimarginal-discr}
	\umin{\P \in \CouplingsD(\a_s)_s}
		\dotp{\C}{\P} \eqdef \sum_s \sum_{i_s=1}^{n_s} \C_{i_1,\ldots,i_S} \P_{i_1,\ldots,i_S},
}
ここで，妥当なカップリングの集合は
\eq{
	\CouplingsD(\a_s)_s = \enscond{\P \in \RR^{n_1 \times \ldots \times n_S}}{
		\foralls s, \foralls i_s, \sum_{\ell \neq s} \sum_{i_\ell=1}^{n_\ell}  \P_{i_1,\ldots,i_S} = \a_{s,i_s}
	}.
}
エントロピー正則化スキーム~\eqref{eq-regularized-discr} は，この設定へ自然に拡張される
\eq{
	\umin{\P \in \CouplingsD(\a_s)_s}
		\dotp{\P}{\C} - \varepsilon \HD(\P),
}
そして，多重添字ベクトル $i=(i_1,\ldots,i_S)$ で添字付けられた各成分について，スケーリング形式での最適な $\P$ を計算するために Sinkhorn のアルゴリズムを適用することができる
\eq{
		\P_i = \K_i \prod_{s=1}^S \uD_{s,i_s}
		\qwhereq
		\K \eqdef e^{-\frac{\C}{\varepsilon}},
}
ここで $\uD_s \in \RR_+^{n_s}$ は（未知の）スケーリングベクトルであり，$s=1,\ldots,S$ を繰り返し巡回することで反復的に更新される，
\eql{\label{eq-sinkh-multimarg}
	\uD_{s,i_s} \leftarrow
	\frac{ \a_{s,i_s} }{
		\sum_{\ell \neq s} \sum_{i_\ell=1}^{n_\ell}  \K_i \prod_{r \neq s} \uD_{\ell,i_r}
	}
}.

\begin{rem2}{一般の測度}
離散の多周辺問題~\eqref{eq-multimarginal-discr} は，空間 $(\X_1,\ldots,\X_S)$ 上の測度 $(\al_s)_s$ に対して，カップリング測度
\eql{\label{eq-multi-marginal-generic}
	\umin{\pi \in \Couplings(\al_s)_s} \int_{\X_1 \times \ldots \times \X_S} c(x_1,\ldots,x_S) \d\pi(x_1,\ldots,x_S),
}
を計算することによって一般化される．ここで，カップリングの集合は
\eq{
	\Couplings(\al_s)_s \eqdef \enscond{ \pi \in \Mm_+^1(\X_1 \times \ldots \times \X_S) }{
		\foralls s=1,\ldots,S, P_{s,\sharp} \pi = \al_s,
	}
}
であり，$P_s : \X_1 \times \ldots \times \X_S \rightarrow \X_s$ は第 $s$ 成分への射影 $P_s(x_1,\ldots,x_S)=x_s$ である．たとえば~\citep{GangboSciech} を参照されたい．
%
多周辺 OT 問題の主要な性質のレビューについては~\citep{PassMultiReview,PassMultiMarginalStructure} を参照されたい．
%
多周辺 OT の典型的な応用は，量子化学問題の解の近似を計算すること，特に密度汎関数理論~\citep{CotarDFT,GorSeiVig,BuDePGor} における計算である．この問題は，特異な Coulomb 相互作用コスト
\eq{
	c(x_1,\ldots,x_S) = \sum_{i \neq j} \frac{1}{\norm{x_i-x_j}}.
}
を考えるときに得られる．
\end{rem2}

\begin{rem2}{バリセンターの多周辺定式化}\label{eq-multimarg-bary}
	線形計画最適化~\eqref{eq-barycenter-generic} を，$\X^{S+1}$ 上の単一のカップリングに対する最適化として再構成することが可能である．ここで最後の周辺はバリセンターであり，その他は入力測度 $(\al_s)_{s=1}^S$ である
	\eql{\label{eq-bary-multi-full}
		\umin{ \bar \pi \in \Mm_+^1(X^{S+1}) }
			\int_{\X^{S+1}} \sum_{s=1}^S \la_s c(x,x_s) \d\bar\pi(x_1,\ldots,x_s,x)
	}
	\eq{
		\text{subject to} \quad
				\foralls s=1,\ldots,S, \quad P_{s,\sharp}\bar\pi=\al_s.
	}
	これは「貼り合わせ補題（gluing lemma）」に由来する．この補題は，$\pi_s \in \Couplings(\al_s,\al)$ であるカップリング $(\pi_s)_{s=1}^S$ が与えられたとき，周辺が $\pi_s$ である高次元のカップリング $\bar \pi \in \Mm_+^1(X^{S+1})$，すなわち $Q_{s\sharp}\bar\pi=\pi_s$（ここで $Q_s(x_1,\ldots,x_S,x) \eqdef (x_s,x) \in \X^2$）となるものを構成できることを述べている．
	\todoK{also relate this to celebrated result in graphical model, see book of Wainright/Jordan}
	%
	\eqref{eq-bary-multi-full} を最後の周辺（$x \in \X$ に対応する）に関して明示的に最小化することにより，バリセンター問題~\eqref{eq-barycenter-generic} の解 $\al$ が $\al = A_{\la,\sharp} \pi$ として計算できることが得られる．ここで $A_\la$ は次で定義される「バリセンター写像」である
	\eq{
		A_\la : (x_1,\ldots,x_S) \in \X^S \mapsto
		\uargmin{x \in \X} \sum_s \la_s c(x,x_s)
	}
	（この写像が一価であると仮定する）．ここで $\pi$ はコスト
	\eql{\label{eq-cost-bary-multi}
		c(x_1,\ldots,x_S) = \sum_\ell \la_\ell c(x_\ell, A_\la(x_1,\ldots,x_S)).
	}
	をもつ多周辺問題~\eqref{eq-multi-marginal-generic} の任意の解である．たとえば $c(x,y)=\norm{x-y}^2$ に対しては，定数の二乗項を除去すると，
	\eq{
		c(x_1,\ldots,x_S) = -\sum_{r \leq s} \la_r \la_s \dotp{x_r}{x_s},
	}
	となり，これは~\citet{GangboSciech} で研究されている問題である．
	%
	より詳しくは~\citet{Carlier_wasserstein_barycenter} を参照されたい．
	%
	この公式は，すべての入力測度が離散 $\be_s = \sum_{i_s=1}^{n_s} \a_{s,i_s} \de_{x_{s,i_s}}$ であれば，バリセンター $\al$ もまた離散であり，次の公式を用いて得られることを示している
	\eq{
		\al = \sum_{(i_1,\ldots,i_S)} \P_{(i_1,\ldots,i_S)}
			\de_{ A_\la(x_{i_1},\ldots,x_{i_S}) },
	}
	ここで $\P$ は，\eqref{eq-cost-bary-multi} で定義されたコスト行列 $\C_{i_1,\ldots,i_S} = c(x_{i_1},\ldots,x_{i_S})$ をもつ~\eqref{eq-multimarginal-discr} の最適解である．$\P$ は $\prod_s n_s$ 次元の非負テンソルであり，$\sum_s n_s-S+1$ 個の等式制約をもつ線形計画の解として得られるので，高々 $\sum_s n_s-S+1$ 個の非零値をもつ最適解 $\P$ が得られる．したがって，高々 $\sum_s n_s-S+1$ 個の点の台をもつバリセンター $\al$ が得られる．この結果および離散の場合のその他の考察は~\citet{anderes2016discrete} に見出される．
\end{rem2}

\begin{rem2}{Euler 方程式の緩和}
非圧縮性流体力学の Euler 方程式の凸緩和が~\citeauthor{BrenierGeneralized}（\citeyear{BrenierEulerAMS}, \citeyear{BrenierEulerARMA}, \citeyear{BrenierEulerCPAM}, \citeyear{BrenierGeneralized}）および~\citep{AmbrosioFigalliEuler} によって提案されている．
%
\S\ref{sec-entropic-dynamic} で示した設定と同様に，これは流体中の粒子の運動を記述する，すべての経路 $\ga : [0,1] \rightarrow \Xx$ の集合 $\bar\Xx$ 上の確率分布 $\bar\pi \in \Mm_+^1(\bar\Xx)$ を求める問題に対応する．
%
これは初期の偏微分方程式モデルの緩和版である．なぜなら，OT の Kantorovich 定式化と同様に，質量が分割されうるからである．時間に伴う発展は，基礎となる空間 $\X$ の微分同相写像を必ずしも定義しない．
%
流体のダイナミクスは，\eqref{eq-ot-pathsspace} と同様に各経路のエネルギー $\int_0^1 \norm{\ga'(t)}^2 \d t$ を最小化することで得られる．
%
経路の空間上の OT との違いは，流体の非圧縮性という追加の条件である．この非圧縮性は，粒子の密度が任意の時刻 $t \in [0,1]$ で一様であるべきだと課すことによって扱われる（古典的な OT のように初期時刻と最終時刻 $t \in \{0,1\}$ でのみ課されるのではない）．$\X$ がコンパクトであると仮定し，$\Xx$ 上の一様分布を $\rho_{\Xx}$ と表すと，これは $\bar P_{t,\sharp} \bar\pi = \rho_{\Xx}$ と書ける．ここで $\bar P_t : \ga \in \bar\X \rightarrow \ga(t) \in \X$ である．
%
この問題は，連続な経路 $(\ga(t))_{t \in [0,1]}$ をグリッド $(x_k)_{k=1}^n \subset \X$ 上の $S$ 個の点の列 $(x_{i_1}, x_{i_2},\ldots,x_{i_S})$ に置き換えることで離散化できる．そして $\bar\Pi$ は $S$-重カップリング $\P \in \RR^{n^S} \in \Couplings(\a_s)_s$ で表され，周辺は一様 $\a_s=n^{-1} \ones_n$ である．
%
対応する多周辺問題のコストは，このとき
\eql{\label{eq-separable-euler}
	\C_{i_1,\ldots,i_S} = \sum_{s=1}^{S-1} \norm{x_{i_s}-x_{i_{s+1}}}^2 + R \norm{x_{\sigma(i_1)}-x_{i_{S}}}^2.
}
である．ここで $R$ は十分大きいペナルティ定数であり，初期時刻と最終時刻の間の粒子の運動を強制するためのものである．この運動は置換 $\sigma : \range{n} \rightarrow \range{n}$ によって指定される．
%
この結果として得られる多周辺問題は，~\citep{2015-benamou-cisc} で詳述されているように，コストの特別な構造を用いることで Sinkhorn 反復~\eqref{eq-sinkh-multimarg} と組み合わせて効率的に実装される．
%
実際，\eqref{eq-sinkh-multimarg} に現れる分母の計算に必要な $O(n^S)$ のコストの代わりに，それを $S$ 回の行列ベクトル積の連続として分解でき，したがって $Sn^2$ という低コストで済む．
%
なお，たとえば~\S\ref{s-semidiscrete} で示した半離散の枠組みを用いた他のソルバーも提案されている．~\citep{deGoes2015,gallouet2017lagrangian} を参照されたい．
\end{rem2}

\section{非平衡最適輸送}
\label{sec-unbalanced}

通常の形における最適輸送の大きなボトルネックは，二つの入力測度 $(\al,\be)$ が同じ総質量をもつことを要求する点にある．多くの回避策が提案されてきた（入力測度を再正規化することや，\S~\ref{sec-dual-norms} で詳述したような双対ノルムを用いることを含む）が，満足のいく統一的な理論が発展したのは最近のことである．ここでは単純だが重要な一つの特別な場合だけを概説する．

\citet{LieroMielkeSavareLong} に従い，任意の正のヒストグラム $(\a,\b) \in \RR_+^n \times \RR_+^m$ を扱うために，初期の Kantorovich 定式化~\eqref{eq-mk-discr} は，\eqref{eq-discr-diverg} で定義されるあるダイバージェンス $\DivergmD_\phi$ を用いて周辺の偏差のみをペナルティ化することによって「緩和」される．これは同等に，近似測度間の OT 距離を最小化することに対応する
\begin{align}\label{eq-unbalanced-pbm}
	\MKD_{\C}^\tau(\a,\b) &= \umin{\tilde\a,\tilde\b} \MKD_{\C}(\a,\b) + \tau_1 \DivergmD_\phi(\a,\tilde \a) + \tau_2 \DivergmD_\phi(\b,\tilde\b) \\
		&= \umin{\P \in \RR_+^{n \times m}} \dotp{\C}{\P} + \tau_1\DivergmD_\phi(\P\ones_m|\a) + \tau_2\DivergmD_\phi(\P^\top\ones_m|\b),
\end{align}
ここで $(\tau_1,\tau_2)$ は，質量の輸送に対して質量の変動をどれだけペナルティ化するかを制御する．極限 $\tau_1=\tau_2 \rightarrow +\infty$ において，$\sum_i \a_i=\sum_j \b_j$（「平衡」の場合）を仮定すると，硬い周辺制約をもつ元の最適輸送定式化~\eqref{eq-mk-discr} が回復される．

この形式論は多くの異なる先行研究を回復する．たとえば $\DivergmD_\phi$ に $\ell^2$ ノルム~\citep{benamou2003numerical} を導入したものや，部分輸送~\citep{FigalliPartial,CaffarelliMcCannPartial} におけるように $\ell^1$ ノルムを導入したものである．
%
特に重要な場合は，$\DivergmD_\phi=\KLD$ として Kullback--Leibler ダイバージェンスを用いるときであり，Remark~\ref{rem-wfr-static} で詳述される．
%
このコストに対して，極限 $\tau=\tau_1=\tau_2 \rightarrow 0$ において，いわゆる二乗 Hellinger 距離が得られる（Example~\ref{exmp-hellinger} も参照）
\eq{
	\MKD_{\C}^\tau(\a,\b) \overset{\tau \rightarrow 0}{\longrightarrow}
	\Hellinger^2(\a,\b) = \sum_i ( \sqrt{\a_i}-\sqrt{\b_i} )^2.
}

Sinkhorn 反復~\eqref{eq-sinkhorn} は，\S\ref{sec-generalized} で詳述される一般化アルゴリズムを用いることで，この問題に適応できる．解は~\eqref{eq-scaling-form} の形をもち，スケーリングは次のように更新される
\eql{\label{eq-iterate-gen-sinkh}
	\uD \leftarrow \pa{ \frac{\a}{\K \vD} }^{\frac{\tau_1}{\tau_1+\varepsilon}}
	\qandq
	\vD \leftarrow \pa{ \frac{\b}{\transp{\K}\uD} }^{\frac{\tau_2}{\tau_2+\varepsilon}}.
}

\begin{rem2}{一般の測度}
任意の二つの測度 $(\al,\be)$ に対して，\eqref{eq-mk-generic} の非平衡版（「log-entropic」とも呼ばれる）は次のように書ける
\begin{align*}
	\MK_\c^\tau(\al,\be) \eqdef
	&\umin{\pi \in \Mm_+(\X \times \Y)}
		\int_{\X \times \Y} \c(x,y) \d\pi(x,y) \\
		&+ \tau \Divergm_\phi(P_{1,\sharp}\pi|\al)
		+ \tau \Divergm_\phi(P_{2,\sharp}\pi|\be),
\end{align*}
ここで測度間のダイバージェンス $\Divergm_\phi$ は~\eqref{eq-phi-div} で定義される．
%
特別な場合 $c(x,y) = \norm{x-y}^2$，$\Divergm_\phi=\KL$ において，$\MK_\c^\tau(\al,\be)^{1/2}$ は Gaussian--Hellinger 距離~\citep{LieroMielkeSavareLong} であり，これは $\Mm_+^1(\RR^d)$ 上の距離であることが示されている．
\end{rem2}

バリセンター問題~\eqref{eq-barycenter-generic} は，$\MK_{\c}$ を $\MK_\c^\tau$ に置き換えることで，非平衡の設定を扱うように一般化できる．
%
Figure~\ref{fig-interpolation-static} は，結果として得られる補間を示しており，緩和パラメータ $\tau$ の有用性をよく示している．
%
入力測度は，質量の異なる二つの Gaussian の混合である．古典的な OT は最も左の山を二つに分割することを要求し，正則でない補間を与える．これとは対照的に，非平衡 OT は補間の途中で質量が変動することを許すので，山は分割されず，分布の局所的なモードが滑らかにマッチングされる．
%
有限の値の $\tau$ を用いること（OT は $\tau=\infty$ と同等であることを思い出されたい）は，したがって，「硬い」質量保存制約のために生じる質量の分割によって発生する不規則な補間を防ぐために重要である．
%
結果として得られる最適化問題は，エントロピー正則化および~\S\ref{sec-generalized} で詳述される一般化 Sinkhorn アルゴリズムを用いて数値的に取り組むことができる．

実際には，入力データがノイズを含むか，あるいは完全には既知でないような応用（画像処理や機械学習など）において，非平衡 OT の手法は古典的な OT を上回るように見える．また，総質量によって測られる測度の信号強度が考慮されなければならない場合や，正規化が意味をなさない場合にも，それらは決定的に重要である．これは~\citet{FrognerNIPS} の本来の動機であり，その目標は画像を記述するために用いられる単語ラベルの集合を比較することであった．非平衡 OT および対応する Sinkhorn 反復は，~\citep{schiebinger2017reconstruction} における細胞のダイナミクスへの応用にも用いられている．

\section{カップリングに追加制約をもつ問題}

文献では，他にも多くの OT 様の問題が提案されてきた．それらは典型的に，元の OT 問題~\eqref{eq-mk-generic} に現れる実行可能なカップリングの集合に，追加の制約 $\Cc$ を加えることに対応する
\eql{\label{ot-extra-constr}
	\umin{\pi \in \Couplings(\al,\be)}
		\enscond{ \int_{\X \times \Y} \c(x,y) \d\pi(x,y) }{ \pi \in \Cc }.
}
代表的な二つの例を挙げる．
%
容量制約をもつ最適輸送~\citep{km1} は，密度 $\density{\pi}$（たとえば Lebesgue 測度に関する）が上から有界であることを課すことに対応する
\eql{\label{eq-capacity-constr}
	\Cc = \enscond{ \pi }{ \density{\pi} \leq \kappa }
}
（ある $\kappa>0$ に対して）．この制約は，Monge 写像上に局所化された特異なカップリングを排除する．
%
マルチンゲール輸送問題（たとえば~\citet{GalichonMartingale,dolinsky2014martingale,TanTouzi,beiglbock2013model} を参照）は，金融において多くの応用をもち，$\X=\Y=\RR^d$ のときカップリングの条件付き平均にいわゆるマルチンゲール制約を課す：
\eql{\label{eq-martingale-constr}
	\Cc = \enscond{ \pi }{ \foralls x \in \RR^d,  \int_{\RR^d} y \frac{\d \pi(x,y)}{\d\al(x)\d\be(y)} \d\be(y) = x }.
}
この制約は，任意の許容されるカップリングのバリセンター射影写像~\eqref{eq-bary-proj} が恒等写像に等しくなければならないことを課す．
%
任意の $(\al,\be)$ に対して，この集合 $\Cc$ は典型的に空であるが，$\Cc \neq \emptyset$ を保証する必要十分条件（$\al$ と $\be$ が「凸順序」になければならない）が存在し，それによって $(\al,\be)$ がマルチンゲール制約を満たす．この制約は，既存の問題を離散化するとき，数値的に強制するのが難しいことがある．
%
また，それは解が単一の Monge 写像に集中することを禁じ，複数のグラフの和集合に集中するカップリング（「多価」Monge 写像）や，さらに複雑な台集合に至ることがある．
%
\eqref{eq-entropic-generic} のようにエントロピーペナルティを用いることで，~\citet{2015-benamou-cisc} で説明されているように Dykstra アルゴリズムを用いて~\eqref{ot-extra-constr} を近似的に解くことができる．これは~\S\ref{sec-sinkhorn} で示した Sinkhorn のアルゴリズムの一般化である．これは $\KL$ ダイバージェンスに対する $\Cc$ への射影の計算を要求する．それは~\eqref{eq-capacity-constr} に対しては容易だが，\eqref{eq-martingale-constr} に対しては閉じた形で行うことができず，したがって部分反復を必要とする．より詳しくは~\citep{guo2017computational} を参照されたい．

\section{スライス Wasserstein 距離とバリセンター}\label{sec-sliced}

$\RR^\dim$ 上で定義された二つの測度 $(\al,\be)$ の間の距離を，球面のすべての方向への射影間の 1 次元 Wasserstein 距離を集約することによって定義できる．これは次を定義する
\eql{\label{eq-sw-def}
	\SW(\al,\be)^2 \eqdef \int_{\SS^d} \Wass_2( P_{\th,\sharp} \al, P_{\th,\sharp} \be )^2 \d\th,
}
ここで $\SS^\dim = \ensconds{\th \in \RR^\dim}{\norm{\th}=1}$ は $\dim$ 次元の球面であり，$P_\th : x \in \RR^\dim \rightarrow \RR$ は射影である．このアプローチは，Marc Bernot のアイデアに従って~\citep{2013-Bonneel-barycenter} で詳述されている．これは測度空間上の Radon 反転の問題~\citep{AbrahamRadon} に関連している．

%%%
\paragraph{Lagrange 的離散化と確率的勾配降下法．}

この汎関数の利点は，~\S\ref{sec:specialcases} で詳述されているように 1 次元 Wasserstein 距離が計算しやすいことである．$m=n$ かつ
\eql{\label{eq-lagrangian-sliced}
	\al=\frac{1}{n}\sum_{i=1}^n \de_{x_i}
	\qandq
	\be=\frac{1}{n}\sum_{i=1}^m \de_{y_i},
}
である特定の場合，これは単に点をソートすることで達成される
\eq{
	\SW(\al,\be)^2 = \int_{\SS^d} \pa{
		 \sum_{i=1}^n | \dotp{ x_{\si_\th(i)} - y_{\kappa_\th(i)} }{\th} |^2
		}
		\d\th,
}
ここで $\si_\th,\kappa_\th \in \Perm(n)$ は，それぞれ $(\dotp{x_i}{\th})_i$ と $(\dotp{y_i}{\th})_i$ を増加順に並べる順序付け置換である．

ベクトル $y$ を固定すると，関数 $\Ee_\be(x) \eqdef \SW(\al,\be)^2$ は滑らかであり，この関数を用いて勾配降下法によって写像を定義できる
\eql{\label{eq-grad-sliced-lagr}
	x \leftarrow x - \tau \nabla\Ee_\be(x)
	\qwhereq
}
\eq{
	\nabla\Ee_\be(x)_i =
		2 \int_{\SS^d} \pa{
		 	\dotp{ x_{i} - y_{\kappa_\th \circ \si_\th^{-1}(i)} }{\th}  \th
		}
		\d\th
}
ここで十分小さいステップサイズ $\tau>0$ を用いる．
%
この方法を扱いやすくするために，この積分をランダムに引いた方向 $\th \in \SS^\dim$ に対する離散和で置き換える確率的勾配降下法（SGD）を用いることができる（SGD についての詳細は~\S\ref{sec-sgd} を参照）．
%
フロー~\eqref{eq-grad-sliced-lagr} は，関数 $\al \mapsto \SW(\al,\be)^2$ の Wasserstein 勾配フロー（\S\ref{sec-grad-flows} の意味での）の（Lagrange 的な実装）として理解できる．数値的には，このフローには局所最小点がなく，したがって $\al=\be$ に収束することがわかる．Lagrange 的ソルバーの有用性は，収束時に，二つの分布の間のマッチング（Monge 写像に類似した）を定義することである．この方法は~\citep{rabin-ssvm-11} においてカラー転送とテクスチャ合成に成功裏に用いられており，~\citep{pitie2007automated} で詳述される交互最小化アプローチに関連している．

この Lagrange 的スキームを，Wasserstein バリセンターの Frechet による定義~\eqref{eq-barycenter-generic} を模倣して
\eql{\label{eq-barycenter-sliced}
		\umin{\al \in \Mm_+^1(\X)} \sum_{s=1}^S \la_s \SW(\al,\be_s)^2,
}
を最小化することで，測度の近似的な「スライス」バリセンターを計算するように拡張するのは単純である．ここで固定された入力測度の集合 $(\be_s)_{s=1}^S$ が与えられる．$\al$ と $(\be_s)_s$ の両方に対して~\eqref{eq-lagrangian-sliced} の形の Lagrange 的離散化を用いると，位置 $x=(x_i)_i$ に対する非凸最小化
\eql{\label{eq-bary-lagrangian-sliced}
	\umin{x} \Ee(x) \eqdef \sum_{s} \la_s \Ee_{\be_s}(x),
	\qandq
	\nabla \Ee(x) = \sum_{s} \la_s \nabla \Ee_{\be_s}(x),
}
を，公式~\eqref{eq-grad-sliced-lagr} を用いて $\nabla \Ee_{\be_s}(x)$ を計算しながら（方向 $\th$ のランダムサンプリングと組み合わせて）勾配降下法によって実行できる．




%%%
\paragraph{Euler 的離散化と Radon 変換．}

反復的な最小化スキームに頼ることなく近似的なスライスバリセンターを計算する関連した方法は，\eqref{eq-sw-def} が Radon 変換 $\Rr(\al)$ と $\Rr(\be)$ の間の距離を計算するという事実を用いることである．ここで
\eq{
	\Rr(\al) \eqdef ( P_{\th,\sharp} \al )_{\th \in \SS^d}.
}
である．決定的な点は，Radon 変換が可逆であり，その逆がフィルター付き逆投影公式を用いて計算できることである．
%
測度の集まり $\rho = (\rho_\th)_{\th \in \SS^d}$ が与えられたとき，フィルター付き逆投影作用素を次のように定義する
\eql{\label{eq-filter-backproj}
	\Rr^+(\rho) = C_\dim \De^{\frac{d-1}{2}} \Bb(\rho),
}
ここで $\xi = \Bb(\rho) \in \Mm(\RR^d)$ は次の関係を通じて定義される測度である
\eql{\label{eq-backproj}
	\foralls g \in \Cc(\RR^d), \quad
	\int_{\RR^d} g(x) \d\xi(x) =
	\int_{\SS^d}  \int_{\RR^{\dim-1}}   \int_{\RR}
		g( r \th + U_\th z )
	\d \rho_\th(r) \d z \d \th,
}
ここで $U_\th$ は $\th^\bot$ の任意の正規直交基底であり，$C_\dim \in \RR$ は次元に依存する正規化定数である．
%
ここで $\De^{\frac{d-1}{2}}$ は分数階 Laplacian であり，Fourier 領域上で $\hat \De^{\frac{d-1}{2}}(\om) = \norm{\om}^{d-1}$ として定義されるハイパスフィルターである．
%
逆投影~\eqref{eq-backproj} の定義は，すべての測度 $(\rho_\th)_\th$ の寄与を，各々を $\th$ に直交する方向に定数として拡張することによって足し合わせる．
%
このとき左逆の関係 $\Rr^+ \circ \Rr = \Identity_{\Mm(\RR^\dim)}$ が成り立ち，したがって $\Rr^+$ は妥当な再構成公式である．

\eql{\label{eq-barycenter-radon-sliced}
		\rho_\th^\star \in \uargmin{ (\rho_\th \in \Mm_+^1(\RR)) } \sum_{s=1}^S \la_s \Wass_2( \rho_\th, P_{\th,\sharp} \be_s)^2.

を考えるのは理にかなっている．各 1 次元バリセンター問題は，Remark~\ref{rem-bary-1d} で詳述されるように単調再配置を用いて容易に計算される．
%
\eqref{eq-barycenter-generic} を解くスライスバリセンターの Radon 近似 $\al_{R} \eqdef \Rr^+( \rho^\star )$ は，このとき逆 Radon 変換 $\Rr^+$ によって得られる．一般に $\al_R$ は~\eqref{eq-barycenter-generic} の解ではないことに注意されたい．なぜなら Radon 変換は全射ではないので，Radon 変換 $\Rr(\be_s)$ のバリセンターとして得られる $\rho^\star$ が必ずしも $\Rr$ の値域に属するとは限らないからである．しかし数値的には，実際にはほとんどその通りになるように見える~\citep{2013-Bonneel-barycenter}．

\paragraph{スライス Wasserstein カーネル．}

スライス Wasserstein 距離は，その計算の単純さに加えて，もう一つの利点をもつ．それは，Remark~\ref{rem-1d-ot-generic}，特に公式~\eqref{eq-wass-cumul} で詳述されるように，Euclid 距離と等長であること（したがって「Hilbert 的」計量であること）である．
%
\S\ref{sec-non-embeddability} で強調されているように，これは $\RR^\dim$ 上の Wasserstein 距離 $\Wass_2$ と対照をなすべきである．後者は次元 $\dim \geq 2$ において Hilbert 的でない．
%
したがって，このスライス距離を用いて，分布の空間 $\Mm_1^+(\RR^\dim)$ に再生核 Hilbert 空間構造を備えることが可能である（\S\ref{sec-non-embeddability} で詳述される）．
%
たとえば，指数カーネルとエネルギー距離カーネル
\eq{
	\Krkhs(\al,\be) = e^{-\frac{\SW(\al,\be)^p}{2\si^p}}
	\qandq
	\Krkhs(\al,\be) = -\SW(\al,\be)^p
}
を用いることができる．指数カーネルに対しては $1 \leq p \leq 2$，エネルギー距離カーネルに対しては $0 < p < 2$ である．
%
これは，入力測度の任意の集まり $(\al_i)_i$ に対して，行列 $( \Krkhs(\al_i,\al_j) )_{i,j}$ が対称半正定値であることを意味する．これらのカーネルを用いて，「カーネルトリック」を使った様々な機械学習タスク，たとえば回帰，分類（SVM とロジスティック），クラスタリング（K-means），次元削減（PCA）~\citep{Hofmann2008} を実行することが可能である．
%
詳細と応用については~\citet{kolouri2016sliced} を参照されたい．

\section{ベクトルと行列の輸送}

実数値測度 $\al \in \Mm(\Xx)$ は，ベクトル値測度 $\al \in \Mm(\Xx;\VV)$ へ容易に一般化される．ここで $\VV$ はあるベクトル空間である．表記の単純さのために，$\VV$ は Euclid 的であり，ある内積 $\dotp{\cdot}{\cdot}$ を備えていると仮定する（典型的には $\VV=\RR^\dim$ であり，内積は標準的なものである）．この内積のおかげで，ベクトル値測度は連続関数 $g : \Xx \rightarrow \VV$ の双対と同一視される．すなわち任意のそのような $g$ に対して，測度に対するその積分を次のように定義する
\eql{\label{eq-integration-vec-valued}
	\int_\Xx g(x) \d\al(x) \in \RR,
}
これは $g$ と $\al$ に関する線形演算である．離散測度は $\al = \sum_i \a_i \de_{x_i}$ の形をもち，ここで $(x_i,a_i) \in \Xx \times \VV$ であり，積分公式~\eqref{eq-integration-vec-valued} は単に
\eq{
	\int_\Xx g(x) \d\al(x) = \sum_i \dotp{\a_i}{g(x_i)} \in \RR.
}
と書ける．同等に，$\VV=\RR^\dim$ であれば，そのような $\al$ は $\dim$ 個の「古典的な」実数値測度の集まり $(\al_s)_{s=1}^\dim$（その座標）と見なすことができ，
\eq{
	\int_\Xx g(x) \d\al(x) =  \sum_{s=1}^\dim \int_\Xx g_s(x) \d\al_s(x),
}
と書ける．ここで $g(x)=(g_s(x))_{s=1}^\dim$ は標準基底における $g$ の座標である．

%%%
\paragraph{双対ノルム．}

OT 距離をこのような一般的な設定へ拡張することは自明でなく，実際一般には不可能である．正と負の両方の値をとる実数値測度を扱うことさえ困難である．唯一の単純な選択肢は，\S\ref{sec-dual-norms} で定義したように双対ノルムを考えることである．実際，公式~\eqref{eq-w1-cont} は，$B$ を $\Cc(\Xx;\VV)$ の部分集合と考えることによって $\Mm(\Xx;\VV)$ へ容易に拡張される．したがって特に，$\Ww_1$，フラットノルム，および MMD ノルムは，ベクトル値測度に対して計算できる．

%%%
\paragraph{錐値測度上の OT．}

$\al$ が部分集合 $\Mm(\Xx;\Vv) \subset \Mm(\Xx;\VV)$ に制限される場合，より高度な OT 距離を定義することが可能である．集合 $\Vv$ は $\VV$ の正に 1 次同次な凸錐であるべきである
\eq{
	\Vv \eqdef \enscond{ \la u }{ \la \in \RR^+,  u \in \Vv_0, }
}
ここで $\Vv_0$ はコンパクトな凸集合である．
%
典型的な例は，$\Vv=\RR_+^\dim$ である正測度の集合である．このような錐上の OT の動的な凸定式化が提案されている．~\citep{MatthesPositive} を参照されたい．これは化学成分の分布をモデル化することに応用されている．
%
もう一つの重要な例は，正の対称行列の集合 $\Vv=\Ss_+^\dim \subset \RR^{\dim \times \dim}$ である．もちろん，行列をベクトルとして扱うことで，この空間上で双対ノルムを用いることが可能である．たとえば~\citep{Ning2014metrics} を参照されたい．このような錐上の OT に対する動的な凸定式化が提供されている~\citep{Chen2016,JiangSpectral}．いくつかの静的（Kantorovich 型）定式化も提案されている~\citep{ning2015matrix,2016-peyre-qot} が，数学的に健全な理論的枠組みはまだ欠けている．特に，これらの静的なアプローチがベクトル値測度に対する距離を定義するのか，またそれらが何らかの動的な定式化と関連するのかは不明である．Figure~\ref{fig-tensors} は，テンソル場を扱うために量子相対エントロピー~\eqref{eq-quantum-entropy} を用いた Sinkhorn アルゴリズムの一般化を提案する~\citep{2016-peyre-qot} で詳述される方法を用いて得られたテンソル補間の例である．

%%%
\paragraph{正行列上の OT．}

関連するがかなり異なる設定は，離散測度，すなわちヒストグラム $\a \in \Si_n$ を，$\tr(A)=1$ であるトレースが単位の正行列 $A \in \Ss_n^+$ に置き換えることである．根拠は，$A$ の固有値 $\la(A) \in \Si_n$ がヒストグラムの役割を果たすことであるが，固有ベクトルの回転にも注意を払わなければならず，したがってこの問題はより複雑である．

\S\ref{sec-phi-div} で導入したいくつかのダイバージェンスをこの設定へ拡張できる．たとえば，Bures 計量~\eqref{eq-bure-defn} は Hellinger 距離（Remark~\ref{exmp-hellinger} で定義される）の一般化である．なぜなら，それらは正の対角行列上で等しいからである．
%
Kullback--Leibler ダイバージェンス~\eqref{eq-kl-defn}（Remark~\ref{ex_KLdiv} も参照）も拡張でき，正行列に対しては次のように一般化される
\eql{\label{eq-quantum-entropy}
	\KLD(A|B) \eqdef \tr\pa{ P \log(P)-P\log(Q)-P+Q, }
}
ここで $\log(\cdot)$ は行列対数である．この行列 $\KLD$ は，その両方の引数に関して凸である．

このような行列間の OT 距離を定義するために，凸な動的定式化を解くことが可能である~\citep{Carlen2014,Chen2016,ChenGangbo17}．Sinkhorn のアルゴリズムの同等物も存在し，それは~\citet{gurvits2004classical} によるもので，~\citep{georgiou2015positive} で広範に研究されている．レビュー論文~\citep{ReviewSinkhorn} も参照されたい．それはいくつかの場合にのみ収束することが知られているが，経験的には常に機能するように見える．

\section{Gromov--Wasserstein 距離}

形状マッチングのような一部の応用において，最適輸送距離の重要な弱点は，それらがスケール変換，平行移動，回転といった重要な不変性の族に対して不変でないという事実にある．このような大域的変換を扱うための OT の非凸変種がいくつか提案されており~\citep{cohen1999earth,pele2013tangent}，最近では言語横断的な単語埋め込みのアラインメント~\citep{grave2018unsupervised,alvarez2018towards,grave2018unsupervised} のような問題に応用されているが，これらの方法は，関連性をもたせるために，おそらく異なる距離空間の間で，まず不変性の部分集合を指定することを要求する．本節では，不変性の族の定義を要求することなく，異なる空間上で定義された測度を扱える，OT のより一般的で非常に自然な拡張を述べる．

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Hausdorff 距離}
\label{sec-hausdorff}

ある計量 $\dist_\Z$ に対する二つの集合 $A,B \subset \Z$ の間の Hausdorff 距離は
\eq{
	\Hh_\Zz(A,B) \eqdef \max\Big(
  		\sup_{a \in A} \inf_{b \in B} d_{\Zz}(a,b),
  		\sup_{b \in B} \inf_{a \in A} d_{\Zz}(a,b)
	\Big),
}
である．Figure~\ref{fig-hausdorff} を参照されたい．
%
これは $\Z$ のコンパクト集合 $\Kk(\Z)$ の間の距離を定義し，$\Z$ がコンパクトであれば $(\Kk(\Z),\Hh_\Zz)$ 自体もコンパクトである．~\citep{burago2001course} を参照されたい．

\begin{figure}[h!]
\centering
\includegraphics[width=.26\linewidth]{gromov-hausdorff/hausdorff-1}
\qquad
\includegraphics[width=.26\linewidth]{gromov-hausdorff/hausdorff-2}
\caption{\label{fig-hausdorff}
$\RR^2$ における Hausdorff 距離の計算．
}
\end{figure}

\citet{memoli-2011} に従い，集合 $(A,B)$ の間のこの距離が，測度間の Wasserstein 距離（測度は何らかの意味で「重み付き」集合と理解されるべきである）と同様に定義できることに気づく．測度のカップリング~\eqref{eq-coupling-generic} を集合のカップリング
\eq{
	\Rr(A,B) \eqdef \enscond{ R \in \X \times \Y }{
	\begin{array}{l}
		\foralls a \in A, \exists b \in B, (a,b) \in R \\
		\foralls b \in B, \exists a \in A, (a,b) \in R
	\end{array}
	}.
}
に置き換える．Kantorovich 問題~\eqref{eq-mk-generic} に対しては，（測度にアクセスできないので）積分を最大化に置き換えるべきであり，
\eql{\label{eq-hausf-couplings}
	\Hh_\Zz(A,B) = \uinf{ R \in \Rr(A,B) } \sup_{(a,b) \in R} d(a,b).
}
となる．
%
測度カップリング $\pi \in \Couplings(\al,\be)$ の台は，台の間の集合カップリングであること，すなわち $\Supp(\pi) \in \Rr( \Supp(\al),\Supp(\be) )$ であることに注意されたい．
%
したがって Hausdorff 距離は $\infty$-Wasserstein 距離（Remark~\ref{rem-p-inf} を参照）と関連しており，台が $(A,B)$ である任意の測度 $(\al,\be)$ に対して $\Hh(A,B) \leq \Wass_\infty(\al,\be)$ が成り立つ．


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Gromov--Hausdorff 距離}

Gromov--Hausdorff（GH）距離~\citep{gromov-2001}（~\citep{edwards1975structure} も参照）は，二つの距離空間 $(\X,d_\X), (\Y,d_\Y)$ の間の距離を，それらが互いに等長であることからどれだけ離れているかを定量化することによって測る方法である．Figure~\ref{fig-gh} を参照されたい．これは，二つの空間を第三の空間に等長埋め込みするあらゆる可能な方法の間の最小 Hausdorff 距離として定義される，
\eq{
	\Gg\Hh(d_\Xx,d_\Yy) \eqdef
	\inf_{\Zz,f,g} \enscond{
		\Hh_\Zz(f(\Xx), g(\Yy))
	}{
		\begin{array}{l}
		f : \Xx \overset{\text{isom}}{\longrightarrow} \Zz\\
		g : \Yy \overset{\text{isom}}{\longrightarrow} \Zz
		\end{array}
	}.
}
ここで制約は，$f$ が等長埋め込みでなければならないこと，すなわち任意の $(x,x') \in \X^2$ に対して $d_\Zz(f(x),f(x'))=d_\Xx(x,x')$ であること（$g$ についても同様）である．$\Gg\Hh$ が等長を除いてコンパクト距離空間の間の距離を定義することを示すことができ，特に $\Gg\Hh(d_\Xx,d_\Yy)=0$ であることは，等長 $h : \Xx \rightarrow \Yy$ が存在すること，すなわち $h$ が全単射であり任意の $(x,x') \in \Xx^2$ に対して $d_\Yy(h(x),h(x'))=d_\Xx(x,x')$ であることと同値である．

\begin{figure}[h!]
\centering
\includegraphics[width=.45\linewidth]{gromov-hausdorff/gh-concept}
\caption{\label{fig-gh}
二つの距離空間を比較する GH アプローチ．
}
\end{figure}

\eqref{eq-hausf-couplings} と同様に，そして~\citep{memoli-2011} で説明されているように，GH 距離をカップリングを用いて次のように同等に書き直すことが可能である：
\eq{
	\Gg\Hh(d_\Xx,d_\Yy) = \frac{1}{2} \uinf{ R \in \Rr(\X,\Y) } \sup_{ ((x,y),(x',y')) \in R^2 }
	|d_\Xx(x,x')-d_\Xx(y,y')|.
}
距離行列 $\distD = ( d_{\Xx}(x_i,x_{i'}) )_{i,i'} \in \RR^{n \times n}$，$\distD' = ( d_{\Yy}(y_j,y_{j'}) )_{j,j'} \in \RR^{m \times m}$ を用いて表される離散空間 $\Xx = (x_i)_{i=1}^n, \Yy = (y_j)_{j=1}^m$ に対して，集合カップリング $R$ の台を示す二値行列 $\VectMode{R} \in \{0,1\}^{n \times m}$ を用いて，この最適化を次のように書き直すことができる：
\eql{\label{eq-gh-coupling}
	\text{GH}(\distD,\distD') = \frac{1}{2} \uinf{ \VectMode{R} \ones >0,\VectMode{R}^\top \ones>0  } \max_{ (i,i',j,j') }
	\VectMode{R}_{i,j} \VectMode{R}_{j,j'} |\distD_{i,i'}-\distD_{j,j'}'|.
}
GH 距離の本来の動機は，Figure~\ref{fig-gh-limit} に示されるように，距離空間の極限を定義し研究することであり，詳細については~\citep{burago2001course} を参照されたい．
%
GH 距離の測地線には明示的な記述があり~\citep{chowdhury2016constructing}，これは Remark~\ref{rem-geod-gw} で詳述される Gromov--Wasserstein 空間における測地線と非常によく似ている．

\begin{figure}[h!]
\centering
\includegraphics[width=1\linewidth]{gromov-hausdorff/gh-convergence}
\caption{\label{fig-gh-limit}
距離空間の列の GH 極限．
}
\end{figure}

根底にある最適化問題~\eqref{eq-gh-coupling} は高度に非凸であり，大域的最小値を計算するのは扱いにくい．これは近似スキームを用いて数値的に取り組まれており，視覚や図形学において形状マッチング~\citep{memoli2005theoretical,bronstein2006generalized} への応用を見出している．

最大化を積分に置き換えることによって Hausdorff 距離の定義を「滑らかにする」ことがしばしば望ましい．これは今度は測度の導入を必要とし，次節における GW 距離の定義の動機の一つである．

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Gromov--Wasserstein 距離}

最適輸送はヒストグラム $(\a,\b)$ を比較するために基礎コスト $\C$ を必要とし，したがってそれらのヒストグラムのビンが同じ基礎空間上で定義されていない場合，あるいは第一と第二のヒストグラムのビンの任意の対の間の基礎コストを定義するためにこれらの空間を事前に位置合わせできない場合には用いることができない．
%
この制限に対処するために，代わりにより弱い仮定のみを置くことができる．すなわち，二つの行列 $\distD \in \RR^{n \times n}$ と $\distD' \in \RR^{m \times m}$ が，ヒストグラムが定義される点の間の類似性関係を定量化するという仮定である．典型的なシナリオは，これらの行列が距離行列（のべき）である場合である．
%
GW 問題は次のように書ける
\eql{\label{eq-gw-def}
	\GWD( (\a,\distD), (\b,\distD') )^2 \eqdef \umin{ \P \in \CouplingsD(\a,\b) }
		\Ee_{\distD,\distD'}(\P)
}
\eq{
	\qwhereq
	\Ee_{\distD,\distD'}(\P) \eqdef
		\sum_{i,j,i',j'} |\distD_{i,i'} - \distD'_{j,j'}|^2 \P_{i,j}\P_{i',j'},
}
Figure~\ref{fig-gw} を参照されたい．
%
この問題は，最大化を和に，集合カップリングを測度カップリングに置き換えるとき，GH 問題~\eqref{eq-gh-coupling} と類似している．
%
これは非凸問題であり，二次割当問題~\citep{loiola-2007} として再構成でき，一般には任意の入力に対して解くのが NP 困難である．
%
それは実際，特定のコストに対するグラフマッチング問題~\citep{lyzinski-2015} と同値である．\todoK{detail QAP/graph matching}



\begin{figure}[h!]
\centering
\includegraphics[width=.7\linewidth]{gw/gw-concept}
\caption{\label{fig-gw}
二つの距離測度空間を比較する GW アプローチ．
}
\end{figure}

$\GWD$ が三角不等式を満たすことを示すことができ，実際それは，確率分布（ここでは定義~\eqref{eq-gw-def} において離散であると仮定される）を備えた距離空間の間の距離を，測度を保つ等長を除いて定義する．
%
この距離は~\citet{memoli-2011} によって導入され詳細に研究された．詳細な数学的解説（特にその測地線構造と勾配フロー）は~\citep{SturmGW} に与えられている．コンピュータビジョンへの応用については~\citep{schmitzer2013modelling} も参照されたい．
%
この距離はまた，距離空間の間の GH 距離~\citep{gromov-2001} と密接に関連しており，それは形状マッチング~\citep{memoli-2007,bronstein-2010} に用いられてきた．



%%%%%%%%%%%%%%%%%%%%%
\begin{rem2}{Gromov--Wasserstein 距離}
	一般の設定は，距離測度空間 $(\X,\dist_\X,\al_\X)$ と $(\Y,\dist_\Y,\al_\Y)$ の間のカップリングを計算することに対応する．ここで $(\dist_\X,\dist_\Y)$ は距離であり，$\al_\X$ と $\al_\Y$ はそれぞれの空間上の測度である．
	%
	次を定義する
	\begin{equation}
		\label{eq-gw-generic}
		\begin{split}
		&\GW( (\al_\X,\dist_\X), (\al_\Y,\dist_\Y) )^2 \eqdef \\
		&\umin{ \pi \in \Couplings(\al_\X,\al_\Y) }
		\int_{\X^2 \times \Y^2}
		| \dist_\X(x,x')-\dist_\Y(y,y') |^2
		\d\pi(x,y)\d\pi(x',y').
		\end{split}
	\end{equation}
	$\GW$ は，等長を除いて距離測度空間の間の距離を定義する．ここで，$\phi_{\sharp}\al_\X=\al_\Y$ かつ $\dist_\Y(\phi(x),\phi(x'))=\dist_\X(x,x')$ である全単射 $\phi : \X \rightarrow \Y$ が存在するとき，$(\X,\al_\X,\dist_\X)$ と $(\Y,\al_\Y,\dist_\Y)$ は等長であるという．
\end{rem2}
%%%%%%%%%%%%%%%%%%%%%



%%%%%%%%%%%%%%%%%%%%%
\begin{rem2}{Gromov--Wasserstein 測地線}\label{rem-geod-gw}
この $\GW$ 距離~\eqref{eq-gw-generic} を備えた（等長を除く）距離空間の空間は，測地線構造をもつ．\citet{SturmGW} は，$(\X_0,\dist_{\X_0},\al_0)$ と $(\X_1,\dist_{\X_1},\al_1)$ の間の測地線が
$t \in [0,1] \mapsto (\X_0 \times \X_1,\dist_t,\pi^\star),$ として選べることを示している．ここで $\pi^\star$ は~\eqref{eq-gw-generic} の解であり，すべての $((x_0,x_1), (x_0',x_1')) \in (\X_0 \times \X_1)^2$ に対して，
\eq{
	\dist_t((x_0,x_1), (x_0',x_1')) \eqdef
	(1-t)\dist_{\X_0}(x_0,x_0') + t\dist_{\X_1}(x_1,x_1').
}
である．この公式は，距離空間に関わる汎関数を最小化する勾配フローを定義し解析することを可能にする．~\citet{SturmGW} を参照されたい．しかしながら，それは積空間 $\X_0 \times \X_1$ 上の計算を伴うため，数値的に扱うのは難しい．
%
~\citep{peyre2016gromov} では，関わる空間の濃度を課しつつ，以下で詳述するエントロピー平滑化~\eqref{eq-gw-entropy} を用いて，距離測度空間の測地線とバリセンターを定義するために発見的アプローチが用いられている．
\end{rem2}

\subsection{エントロピー正則化}

$\GWD$ の計算を近似するため，また最小化スキームのより良い最小値への収束を助けるため，エントロピー正則化された変種
\eql{\label{eq-gw-entropy}
	\umin{ \P \in \CouplingsD(\a,\b) }
		\Ee_{\distD,\distD'}(\P) - \varepsilon \HD(\P).
}
を考えることができる．~\citep{gold-1996,rangarajan-1999} で当初提案され，後に図形学への応用のために~\citep{2016-solomon-gw} で再考されたように，Sinkhorn のアルゴリズムを反復的に用いて~\eqref{eq-gw-entropy} の停留点を漸進的に計算できる．
%
実際，目的関数の逐次的な線形化により，次の更新の連続を考えることになる
\eql{\label{eq-gw-sinkh}
	\itt{\P} \eqdef \umin{ \P \in \CouplingsD(\a,\b) } \dotp{\P}{\it{\C}} - \varepsilon\H(\P)
		\qwhereq
}
\eq{
		\it{\C} \eqdef \nabla \Ee_{\distD,\distD'}(\it{\P}) = -\distD \it{\P} \distD',
}
これはミラー降下スキーム~\citep{2016-solomon-gw} として解釈できる．したがって各更新は，コスト $\it{\C}$ をもつ Sinkhorn 反復~\eqref{eq-sinkhorn} を用いて解くことができる．
