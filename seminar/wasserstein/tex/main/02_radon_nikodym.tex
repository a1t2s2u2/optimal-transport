\chapter{Radon--Nikodym の定理}
\label{ch:radon-nikodym}

第\ref{ch:disintegration}章の測度分解定理では，
二つの有限測度の間の密度（Radon--Nikodym 微分）の存在を用いる．
本章ではこれを証明する．
主張は北別府悠『最適輸送理論入門〜兼非標準的関数解析入門〜』
Corollary 4.6.2（p.~57）に従う．
資料では Hilbert 空間 $L^2$ 上の Riesz の表現定理から Lebesgue 分解定理
（Theorem 4.6.1）を経て導いているが，
本資料では $L^2$ の完備性を用いずに済む，正集合の補題による証明を採用する
（例えば Royden--Fitzpatrick \emph{Real Analysis}（第4版）の Hahn の補題と同じ構成）．

以下，$(\Omega,\mathcal{A})$ を可測空間とする．

\section{絶対連続性}

\begin{definition}{測度の絶対連続性（北別府 p.~57）}{meas-absolute-continuity}
  $(\Omega,\mathcal{A})$ 上の有限測度 $\nu,\alpha$ を考える．
  $\alpha$ が $\nu$ に\textbf{絶対連続}であるとは，
  任意の $\varepsilon>0$ に対してある $\delta>0$ が存在し，
  \[
    \nu(B)<\delta\quad\Longrightarrow\quad\alpha(B)<\varepsilon
    \qquad(B\in\mathcal{A})
  \]
  が成り立つことをいう．このとき $\alpha\ll\nu$ と書く．
  北別府の資料では，この定義を実数値の集合関数 $\psi$ に対して
  $\abs{\psi(B)}<\varepsilon$ の形で与えている．
  本資料で用いるのは有限測度の場合である．
\end{definition}

\begin{remark}{零集合による特徴付けとの関係}{rn-null-characterization}
  文献によっては（例えば Folland (1999) §3.2），
  「$\nu(B)=0$ ならば $\alpha(B)=0$」を絶対連続性の定義とする．
  有限測度の場合，この二つは同値である（定理~\ref{thm:meas-radon-nikodym}）．
\end{remark}

\section{正集合の補題}

二つの有限測度 $\alpha,\beta$ に対し，集合関数
\[
  \lambda(B):=\alpha(B)-\beta(B)\qquad(B\in\mathcal{A})
\]
を考える．$\lambda$ は実数値であり，互いに素な $B_1,B_2,\ldots\in\mathcal{A}$ に対して
\[
  \lambda\left(\bigcup_kB_k\right)=\sum_k\lambda(B_k)
\]
が成り立つ．実際，右辺の級数は
$\sum_k\abs{\lambda(B_k)}\leq\sum_k(\alpha(B_k)+\beta(B_k))\leq\alpha(\Omega)+\beta(\Omega)<\infty$
より絶対収束し，$\alpha,\beta$ それぞれの可算加法性の差を取ればよい．

\begin{lemma}{正集合の存在}{rn-positive-set}
  $\alpha,\beta$ を $(\Omega,\mathcal{A})$ 上の有限測度とし，$\lambda:=\alpha-\beta$ とおく．
  $\lambda(\Omega)>0$ ならば，ある $P\in\mathcal{A}$ で
  \[
    \lambda(P)>0,\qquad
    \lambda(C)\geq0\quad(C\in\mathcal{A},\ C\subset P)
  \]
  をみたすものが存在する．
\end{lemma}

\begin{proof}
  $\lambda$ の値が負になる部分集合を，できるだけ大きく負のものから順に取り除いていく．

  $A_0:=\Omega$ とおく．$A_{k-1}$ まで定まったとする．
  $A_{k-1}$ のすべての可測部分集合 $C$ で $\lambda(C)\geq0$ ならば，ここで操作を止める．
  そうでなければ，
  \[
    n_k:=\min\{n\in\N\mid\text{ある }C\in\mathcal{A},\ C\subset A_{k-1}\text{ で }\lambda(C)\leq-1/n\}
  \]
  とし，$C_k\subset A_{k-1}$ を $\lambda(C_k)\leq-1/n_k$ となるように選んで，
  $A_k:=A_{k-1}\setminus C_k$ とおく．このとき
  \[
    \lambda(A_k)=\lambda(A_{k-1})-\lambda(C_k)\geq\lambda(A_{k-1})+\frac1{n_k}
  \]
  なので，$\lambda(A_k)$ は $k$ について増加する．

  操作がある $k$ で止まれば，$P:=A_{k-1}$ は $\lambda(P)\geq\lambda(\Omega)>0$ をみたし，求める集合である．

  止まらない場合，$C_1,C_2,\ldots$ は互いに素であり，
  $P:=\Omega\setminus\bigcup_kC_k$ とおくと，$\lambda$ の可算加法性より
  \[
    \lambda(P)=\lambda(\Omega)-\sum_k\lambda(C_k)\geq\lambda(\Omega)>0.
  \]
  また $\sum_k1/n_k\leq\sum_k\abs{\lambda(C_k)}<\infty$ より $n_k\to\infty$ である．
  ある $C\in\mathcal{A}$，$C\subset P$ で $\lambda(C)<0$ と仮定し，
  $\lambda(C)\leq-1/n$ となる $n\in\N$ を取る．
  $n_k>n$ となる $k$ を取ると，$C\subset P\subset A_{k-1}$ だから，
  $n_k$ の最小性に反する．よって $P$ の可測部分集合 $C$ はすべて $\lambda(C)\geq0$ をみたす．
\end{proof}

\section{Radon--Nikodym の定理}

\begin{lemma}{密度の一意性}{rn-density-uniqueness}
  $\nu$ を $(\Omega,\mathcal{A})$ 上の有限測度とし，
  $h,k\colon\Omega\to\R$ を $\nu$-可積分な可測関数とする．すべての $B\in\mathcal{A}$ で
  \[
    \int_Bh\,\d\nu=\int_Bk\,\d\nu
  \]
  が成り立つならば，$h=k$ が $\nu$-a.e. に成り立つ．
\end{lemma}

\begin{proof}
  $B:=\{h>k\}$ とおくと $\int_B(h-k)\,\d\nu=0$ であり，
  $(h-k)\mathbf{1}_B$ は非負なので，$\nu$-a.e. に $0$ である．
  $B$ 上では $h-k>0$ だから $\nu(B)=0$ である．
  $h,k$ を入れ替えて $\nu(\{k>h\})=0$ も得られ，$h=k$ が $\nu$-a.e. に成り立つ．
\end{proof}

\begin{theorem}{Radon--Nikodym の定理（北別府 Cor 4.6.2）}{meas-radon-nikodym}
  $(\Omega,\mathcal{A})$ 上の二つの有限測度 $\nu,\alpha$ に対し，次の三つは同値である．
  \begin{enumerate}
    \item $\alpha\ll\nu$，すなわち $\alpha$ は $\nu$ に絶対連続である．
    \item すべての $B\in\mathcal{A}$ に対して，$\nu(B)=0$ ならば $\alpha(B)=0$ である．
    \item 非負の $\nu$-可積分な可測関数 $h\colon\Omega\to[0,\infty)$ が存在し，
      \[
        \alpha(B)=\int_Bh\,\d\nu\qquad(B\in\mathcal{A})
      \]
      が成り立つ．
  \end{enumerate}
  条件 3 の $h$ は $\nu$-a.e. の意味で一意であり（補題~\ref{lem:rn-density-uniqueness}），
  \textbf{Radon--Nikodym 微分}とよんで $\d\alpha/\d\nu$ と書く．
\end{theorem}

\begin{proof}
  \textbf{$1\Rightarrow2$．}
  $\nu(B)=0$ とする．任意の $\varepsilon>0$ に対し，対応する $\delta>0$ について
  $\nu(B)<\delta$ だから $\alpha(B)<\varepsilon$ である．$\varepsilon$ は任意なので $\alpha(B)=0$ である．

  \textbf{$3\Rightarrow1$．}（北別府の証明に従う．）
  $\varepsilon>0$ を任意に取る．$n\in\N$ に対し $B_n:=\{h>n\}$ とおくと，
  $h$ は実数値なので $h-h\mathbf{1}_{B_n}=h\mathbf{1}_{\{h\leq n\}}\uparrow h$（各点）である．
  単調収束定理（定理~\ref{thm:meas-mct}）と $\int_\Omega h\,\d\nu<\infty$ より
  \[
    \alpha(B_n)=\int_{B_n}h\,\d\nu\longrightarrow0\qquad(n\to\infty).
  \]
  そこで $\alpha(B_n)<\varepsilon/2$ となる $n$ を取り，$\delta:=\varepsilon/(2n)$ とおく．
  $\nu(B)<\delta$ ならば，$B\setminus B_n$ 上で $h\leq n$ なので
  \[
    \alpha(B)=\alpha(B\cap B_n)+\int_{B\setminus B_n}h\,\d\nu
    \leq\alpha(B_n)+n\,\nu(B)
    <\frac\varepsilon2+n\delta=\varepsilon.
  \]

  \textbf{$2\Rightarrow3$．}
  \[
    \mathcal{G}:=\left\{g\colon\Omega\to[0,\infty]\ \middle|\
    g\text{ は可測で，すべての }B\in\mathcal{A}\text{ について }\int_Bg\,\d\nu\leq\alpha(B)\right\}
  \]
  とおく．$0\in\mathcal{G}$ である．

  \emph{(a) $\mathcal{G}$ は最大について閉じている．}
  $g_1,g_2\in\mathcal{G}$ とし，$B\in\mathcal{A}$ を
  $B_1:=B\cap\{g_1\geq g_2\}$，$B_2:=B\setminus B_1$ に分けると
  \[
    \int_B\max\{g_1,g_2\}\,\d\nu
    =\int_{B_1}g_1\,\d\nu+\int_{B_2}g_2\,\d\nu
    \leq\alpha(B_1)+\alpha(B_2)=\alpha(B).
  \]

  \emph{(b) 積分が最大となる元を取る．}
  $s:=\sup_{g\in\mathcal{G}}\int_\Omega g\,\d\nu\leq\alpha(\Omega)<\infty$ とおく．
  $\int_\Omega g_n\,\d\nu\to s$ となる $g_n\in\mathcal{G}$ を取り，
  $\tilde g_n:=\max\{g_1,\dots,g_n\}$ とおくと，(a) より $\tilde g_n\in\mathcal{G}$ であり，
  $\tilde g_n$ は単調増加である．その各点極限を $g$ とすると，
  単調収束定理（定理~\ref{thm:meas-mct}）より，すべての $B$ で
  $\int_Bg\,\d\nu=\lim_n\int_B\tilde g_n\,\d\nu\leq\alpha(B)$ となり，
  $g\in\mathcal{G}$ かつ $\int_\Omega g\,\d\nu=s$ である．
  $\int_\Omega g\,\d\nu<\infty$ より $\nu(\{g=\infty\})=0$ なので，
  $h:=g\,\mathbf{1}_{\{g<\infty\}}$ とおけば $h$ は実数値で，$h\in\mathcal{G}$，$\int_\Omega h\,\d\nu=s$ である．

  \emph{(c) $h$ が求める密度であること．}
  \[
    \beta(B):=\alpha(B)-\int_Bh\,\d\nu\qquad(B\in\mathcal{A})
  \]
  とおくと，$h\in\mathcal{G}$ より $\beta\geq0$ であり，$\beta$ は有限測度である
  （可算加法性は単調収束定理による）．$\beta(\Omega)=0$ を示せばよい．

  $\beta(\Omega)>0$ と仮定する．$\nu(\Omega)=0$ なら仮定 2 から $\alpha(\Omega)=0$，
  したがって $\beta(\Omega)=0$ となるので，$\nu(\Omega)>0$ である．
  $t>0$ を $\beta(\Omega)-t\,\nu(\Omega)>0$ となるように取り，
  補題~\ref{lem:rn-positive-set}を $\lambda:=\beta-t\nu$ に適用して $P$ を得る．
  $P$ の可測部分集合 $C$ について $t\,\nu(C)\leq\beta(C)$ であるから，
  任意の $B\in\mathcal{A}$ で
  \[
    \int_B(h+t\mathbf{1}_P)\,\d\nu
    =\int_Bh\,\d\nu+t\,\nu(B\cap P)
    \leq\int_Bh\,\d\nu+\beta(B\cap P)
    \leq\int_Bh\,\d\nu+\beta(B)=\alpha(B).
  \]
  よって $h+t\mathbf{1}_P\in\mathcal{G}$ である．
  一方，$\nu(P)=0$ なら仮定 2 から $\alpha(P)=0$，したがって $\beta(P)=0$ となり，
  $\lambda(P)=\beta(P)-t\nu(P)>0$ に反する．よって $\nu(P)>0$ であり，
  \[
    \int_\Omega(h+t\mathbf{1}_P)\,\d\nu=s+t\,\nu(P)>s
  \]
  となって $s$ の定義に反する．したがって $\beta(\Omega)=0$ であり，
  すべての $B$ で $\alpha(B)=\int_Bh\,\d\nu$ が成り立つ．
  $\int_\Omega h\,\d\nu=\alpha(\Omega)<\infty$ より $h$ は $\nu$-可積分である．
\end{proof}

\begin{corollary}{密度による積分の書き換え}{rn-density-integral}
  定理~\ref{thm:meas-radon-nikodym}の条件 3 のもとで，
  任意の非負可測関数 $g\colon\Omega\to[0,\infty]$ に対し
  \[
    \int_\Omega g\,\d\alpha=\int_\Omega g\,h\,\d\nu
  \]
  が成り立つ．
\end{corollary}

\begin{proof}
  $g=\mathbf{1}_B$ のときは条件 3 そのものである．
  積分の線形性から非負単関数について成り立ち，
  非負単関数の列 $s_n\uparrow g$（注意~\ref{rem:meas-measurable-ops}）を取れば
  $s_nh\uparrow gh$ なので，両辺に単調収束定理（定理~\ref{thm:meas-mct}）を用いて一般の場合を得る．
\end{proof}
