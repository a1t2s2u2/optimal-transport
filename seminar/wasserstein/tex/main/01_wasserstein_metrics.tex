\chapter{Wasserstein 距離 \texorpdfstring{$W_p$}{Wp}}
\label{ch:wasserstein-metrics}

\section{設定と定義}
\label{sec:wp-definition}

$(\mathcal{X},d)$ を Polish 空間（定義~\ref{def:meas-polish}）とする．
Borel $\sigma$-代数 $\Borel(\mathcal{X})$ 上の確率測度全体を
\[
  \Prob(\mathcal{X})
  :=
  \left\{
    \mu\colon\Borel(\mathcal{X})\to[0,1]
    \;\middle|\;
    \mu\text{ は測度},\;\mu(\mathcal{X})=1
  \right\}
\]
とおく．

\begin{definition}{有限 $p$ 次モーメントをもつ確率測度（Villani Def 6.4）}{wp-p-space}
  任意に点 $x_0\in \mathcal{X}$ を一つ固定する．$1\leq p<\infty$ に対し
  \[
    \Pp{p}(\mathcal{X})
    :=
    \left\{
      \mu\in\Prob(\mathcal{X})
      \;\middle|\;
      \int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty
    \right\}
  \]
  と定める．積分 $\int_{\mathcal{X}}d(x,x_0)^p\,\d\mu(x)$ を
  $\mu$ の（$x_0$ を基点とする）\textbf{$p$ 次モーメント}という．
\end{definition}

\begin{lemma}{$\Pp{p}(\mathcal{X})$ の $x_0$ 非依存性（Villani Def 6.4）}{wp-p-indep}
  $1\leq p<\infty$ のとき，$\Pp{p}(\mathcal{X})$ は点 $x_0$ の選び方によらない．
  すなわち，任意の $x_0,x_0'\in \mathcal{X}$ と $\mu\in\Prob(\mathcal{X})$ に対して
  \[
    \int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty
    \iff
    \int_\mathcal{X} d(x,x_0')^p\,\d\mu(x)<\infty
  \]
  が成り立つ．
\end{lemma}

\begin{proof}
  基点を $x_0$ から $x_0'$ に変えても，距離は高々 $d(x_0,x_0')$ しか増えない．
  三角不等式と，凸性から従う評価
  $(a+b)^p\leq2^{p-1}(a^p+b^p)$（$a,b\geq0$；注意~\ref{rem:meas-real-facts}）により，
  \[
    \int_\mathcal{X}d(x,x_0')^p\,\d\mu(x)
    \leq
    2^{p-1}\left(\int_\mathcal{X}d(x,x_0)^p\,\d\mu(x)+d(x_0,x_0')^p\right).
  \]
  したがって，$x_0$ を基点とする積分が有限なら，$x_0'$ を基点とする積分も有限である．
  $x_0,x_0'$ を入れ替えれば逆も従う．
\end{proof}

\begin{definition}{Coupling（Villani Def 1.1）}{wp-coupling}
  距離空間 $\mathcal{X},\mathcal{Y}$ 上の確率測度
  $\mu\in\Prob(\mathcal{X})$，$\nu\in\Prob(\mathcal{Y})$ に対し
  \[
    \Couplings(\mu,\nu)
    :=
    \left\{
      \pi\in\Prob(\mathcal{X}\times\mathcal{Y})
      \;\middle|\;
      \begin{aligned}
        \pi(A\times\mathcal{Y})&=\mu(A) &&(\forall A\in\Borel(\mathcal{X})),\\
        \pi(\mathcal{X}\times B)&=\nu(B) &&(\forall B\in\Borel(\mathcal{Y}))
      \end{aligned}
    \right\}
  \]
  と定め，$\pi\in\Couplings(\mu,\nu)$ を $\mu$ と $\nu$ の
  \textbf{coupling} という．
  二つの条件は，送り元の分布が $\mu$，送り先の分布が $\nu$ であることを表す．
\end{definition}

\begin{definition}{Wasserstein 距離（Villani Def 6.1）}{wp-wasserstein}
  $1\leq p<\infty$ と $\mu,\nu\in\Pp{p}(\mathcal{X})$ に対し
  \[
    \Wass_p(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \left(
      \int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    \right)^{1/p}
  \]
  と定める．
\end{definition}

\begin{remark}{距離関数の可測性}{wp-welldef}
  距離関数 $d\colon \mathcal{X}\times \mathcal{X}\to[0,\infty)$ は，三角不等式から従う評価
  \[
    \abs{d(x,y)-d(x',y')}\leq d(x,x')+d(y,y')
  \]
  により連続，したがって Borel 可測である
  （注意~\ref{rem:meas-measurable-ops}）．
\end{remark}

\begin{lemma}{$\Wass_p$ の有限性（Villani Def 6.4）}{wp-finite}
  $1\leq p<\infty$，$\mu,\nu\in\Pp{p}(\mathcal{X})$ とすると，
  $\Wass_p(\mu,\nu)<\infty$ である．
\end{lemma}

\begin{proof}
  輸送コストを，送り元と送り先から基点 $x_0$ までの距離で抑える．
  三角不等式と補題~\ref{lem:wp-p-indep}の証明で用いた凸性の評価から，
  \[
    d(x,y)^p\leq2^{p-1}\bigl(d(x,x_0)^p+d(x_0,y)^p\bigr).
  \]
  直積確率測度 $\pi_0:=\mu\otimes\nu$ は $\mu,\nu$ の coupling である
  （定理~\ref{thm:meas-product}）．その周辺分布は $\mu,\nu$ なので，
  周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  \[
    \begin{aligned}
      \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi_0(x,y)
      &\leq2^{p-1}\left(
        \int_\mathcal{X}d(x,x_0)^p\,\d\mu(x)
        +\int_\mathcal{X}d(x_0,y)^p\,\d\nu(y)
      \right)\\
      &<\infty.
    \end{aligned}
  \]
  最後の不等式は $\mu,\nu\in\Pp{p}(\mathcal{X})$ による．
  こうして有限のコストをもつ coupling が一つ得られたので，
  その下限から定まる $\Wass_p(\mu,\nu)$ も有限である．
\end{proof}

\section{最適 coupling の存在}
\label{sec:wp-existence}

この節では，コストが下限に近づく coupling の列を取り，
その極限で下限を達成する coupling を得る．
証明は次の四段階からなる．
\begin{enumerate}
  \item コストが下限に近づく coupling の列 $(\pi_k)$ を取る．
  \item coupling の tightness と Prokhorov の定理を用いて，
    弱収束部分列 $\pi_{k_j}\weakto\pi^*$ を取る．
  \item 弱極限 $\pi^*$ の二つの周辺分布が $\mu,\nu$ のままであることを示す．
  \item 輸送コストの下半連続性を用いて，$\pi^*$ が下限を達成することを示す．
\end{enumerate}
以下では，2に必要な tightness と，4に必要な下半連続性を先に準備する．
以下，積分域が明らかなときは
$\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)$ を
$\int d^p\,\d\pi$ と略記する．また，弱収束の定義
（定義~\ref{def:meas-weak-convergence}）をここで再掲すると，
\[
  \pi_k\weakto\pi
  \quad\overset{\mathrm{def}}{\Longleftrightarrow}\quad
  \int F\,\d\pi_k\longrightarrow\int F\,\d\pi
  \qquad
  (\forall F\in C_b(\mathcal{X}\times\mathcal{X})).
\]

\begin{lemma}{$p$ 次輸送コストの下半連続性（Villani Lem 4.3）}{wp-cost-lsc}
  $1\leq p<\infty$ とする．
  $\Prob(\mathcal{X}\times\mathcal{X})$ の列 $(\pi_k)$ が
  $\pi\in\Prob(\mathcal{X}\times\mathcal{X})$ に弱収束，すなわち
  $\pi_k\weakto\pi$ とする．このとき
  \[
    \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    \leq
    \liminf_{k\to\infty}
    \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi_k(x,y).
  \]
\end{lemma}

\begin{proof}
  弱収束で積分の収束が分かるのは有界連続関数なので，
  まず $d^p$ の大きい値を $m\in\N$ で切り捨て，
  \[
    c_m(x,y):=\min\{d(x,y)^p,m\}
  \]
  とおく．$d$ は連続（注意~\ref{rem:wp-welldef}）だから $c_m$ は有界連続である．
  各 $m$ について，弱収束の定義（定義~\ref{def:meas-weak-convergence}）と
  $c_m\leq d^p$ より
  \[
    \int c_m\,\d\pi
    =\lim_{k\to\infty}\int c_m\,\d\pi_k
    \leq\liminf_{k\to\infty}\int d^p\,\d\pi_k.
  \]
  $m$ を増やすと $c_m$ は単調に増加して $d^p$ に近づく．
  右辺は $m$ によらないので，単調収束定理（定理~\ref{thm:meas-mct}）から
  \[
    \int d^p\,\d\pi
    =\lim_{m\to\infty}\int c_m\,\d\pi
    \leq\liminf_{k\to\infty}\int d^p\,\d\pi_k
  \]
  を得る．
\end{proof}

\begin{remark}{記法：測度の族}{wp-family-symbol}
  Villani は測度の族を $\mathcal{P},\mathcal{Q}$ と書くが，
  本資料では確率測度全体を $\Prob(\mathcal{X})$ と書いているため
  記号が衝突する．そこで以下では族を $A,B$ で表す．
\end{remark}

\begin{lemma}{輸送計画の tightness（Villani Lem 4.4）}{wp-plans-tight}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間とし，
  $A\subset\Prob(\mathcal{X})$，$B\subset\Prob(\mathcal{Y})$ を
  tight（定義~\ref{def:meas-tight}）な確率測度の族とする．
  周辺分布がそれぞれ $A$，$B$ に属する coupling の全体を
  \[
    \Couplings(A,B)
    :=\bigcup_{\mu\in A,\,\nu\in B}\Couplings(\mu,\nu)
  \]
  と定めると，$\Couplings(A,B)$ は $\Prob(\mathcal{X}\times\mathcal{Y})$
  で tight である．
\end{lemma}

\begin{proof}
  $\varepsilon>0$ とする．$A,B$ の tightness より，compact 集合
  $K\subset\mathcal{X}$，$L\subset\mathcal{Y}$ を
  \[
    \eta(\mathcal{X}\setminus K)\leq\frac{\varepsilon}{2}
    \quad(\forall\eta\in A),
    \qquad
    \rho(\mathcal{Y}\setminus L)\leq\frac{\varepsilon}{2}
    \quad(\forall\rho\in B)
  \]
  となるように選ぶ．
  任意の $\pi\in\Couplings(A,B)$ を取り，その二つの周辺分布を
  $\eta\in A$，$\rho\in B$ とする．
  $(x,y)$ が $K\times L$ の外にあるなら，$x$ が $K$ の外にあるか，
  $y$ が $L$ の外にある．したがって
  \[
    \begin{aligned}
      \pi\bigl((\mathcal{X}\times\mathcal{Y})\setminus(K\times L)\bigr)
      &\leq\eta(\mathcal{X}\setminus K)+\rho(\mathcal{Y}\setminus L)\\
      &\leq\varepsilon.
    \end{aligned}
  \]
  同じ compact 集合 $K\times L$ で，すべての $\pi\in\Couplings(A,B)$ に
  対してこの評価が成り立つので，$\Couplings(A,B)$ は tight である．
\end{proof}

\begin{theorem}{$\Wass_p$ の最適 coupling の存在（Villani Thm 4.1；pp.~43--45）}{wp-optimal-coupling-existence}
  $\mathcal{X}$ を Polish 空間，$1\leq p<\infty$ とする．
  任意の $\mu,\nu\in\Pp{p}(\mathcal{X})$ に対して，
  ある $\pi^*\in\Couplings(\mu,\nu)$ が存在し，
  \[
    \Wass_p(\mu,\nu)
    =
    \left(
      \int_{\mathcal{X}\times\mathcal{X}}
      d(x,y)^p\,\d\pi^*(x,y)
    \right)^{1/p}
  \]
  が成り立つ．
  すなわち，$\Wass_p$ の定義に現れる下限は最小値として達成される．
  この $\pi^*$ を $\Wass_p$ の\textbf{最適 coupling}という．
\end{theorem}

\begin{proof}
  輸送コストとその下限を
  \[
    J(\pi):=\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y),
    \qquad
    I:=\inf_{\pi\in\Couplings(\mu,\nu)}J(\pi)
  \]
  とおく．補題~\ref{lem:wp-finite}の証明より，coupling は存在し，$I<\infty$ である．
  $t\mapsto t^{1/p}$ は連続かつ単調増加なので，$I=\Wass_p(\mu,\nu)^p$ である．

  \textbf{1. コストを下限に近づける．}
  下限の定義より，各 $k\in\N$ に対して
  \[
    \pi_k\in\Couplings(\mu,\nu),
    \qquad
    I\leq J(\pi_k)<I+\frac1k
  \]
  となる $\pi_k$ を取る．これで $J(\pi_k)\to I$ となる．

  \textbf{2. 収束する部分列を取る．}
  $\{\mu\}$ の列は定数列なので，Prokhorov の定理
  （定理~\ref{thm:meas-prokhorov}）より $\{\mu\}$ は tight である．
  $\{\nu\}$ も同様であり，補題~\ref{lem:wp-plans-tight}から
  $\Couplings(\mu,\nu)$ は tight である．
  $\mathcal{X}\times\mathcal{X}$ は Polish 空間なので，再び Prokhorov の定理により，
  ある部分列と $\pi^*\in\Prob(\mathcal{X}\times\mathcal{X})$ に対して
  \[
    \pi_{k_j}\weakto\pi^*
  \]
  となる．

  \textbf{3. 極限でも送り元と送り先の分布は変わらない．}
  任意の有界連続関数 $\varphi\in C_b(\mathcal{X})$ に対し，
  $(x,y)\mapsto\varphi(x)$ も有界連続である．弱収束と周辺積分公式
  （補題~\ref{lem:meas-marginal-integral}）から
  \[
    \begin{aligned}
      \int_{\mathcal{X}\times\mathcal{X}}\varphi(x)\,\d\pi^*(x,y)
      &=\lim_{j\to\infty}
        \int_{\mathcal{X}\times\mathcal{X}}\varphi(x)\,\d\pi_{k_j}(x,y)\\
      &=\int_\mathcal{X}\varphi(x)\,\d\mu(x).
    \end{aligned}
  \]
  有界連続関数による測度の判定（定理~\ref{thm:meas-cb-determines}）より，
  $\pi^*$ の第1周辺分布は $\mu$ である．$y$ だけの関数で同様に調べれば，
  第2周辺分布は $\nu$ である．したがって $\pi^*\in\Couplings(\mu,\nu)$ である．

  \textbf{4. 極限のコストが下限に一致する．}
  $\pi^*$ も coupling なので $I\leq J(\pi^*)$ である．
  一方，コストの下半連続性（補題~\ref{lem:wp-cost-lsc}）と $J(\pi_{k_j})\to I$ より
  \[
    I\leq J(\pi^*)\leq\liminf_{j\to\infty}J(\pi_{k_j})=I.
  \]
  よって $J(\pi^*)=I=\Wass_p(\mu,\nu)^p$ となり，$\pi^*$ が最適 coupling である．
\end{proof}

\section{Wasserstein 距離の距離性}
\label{sec:metric-property}

前節の最適 coupling の存在を用いて，$\Wass_p$ の距離性を示す．
三角不等式の鍵は，共通の周辺分布をもつ二つの coupling を
一つの三変数の確率測度にまとめる接着補題である．
まずその準備として測度分解定理を証明し，接着補題を経て距離の公理を確かめる．

\subsection{測度分解定理}

接着補題では，二変数の分布 $\pi$ を，$y$ の分布 $\nu$ と，
$y$ を固定したときの $x$ の分布 $\pi_y$ に分ける．
証明は北別府悠『最適輸送理論入門〜兼非標準的関数解析入門〜』
Theorem 5.3.1 の証明（提供資料）に沿って，
まず compact な場合に測度を構成し，次に一般の Polish 空間へ拡張する．
用いる標準定理は，付録の Radon--Nikodym の定理（定理~\ref{thm:meas-radon-nikodym}）と
Riesz の表現定理（定理~\ref{thm:meas-riesz-representation}）である．
$C(K)$ の記法と可分性は注意~\ref{rem:meas-compact-continuous}にまとめた．

\begin{theorem}{測度分解定理（周辺分布を用いた積分表示）}{meas-disintegration}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間，$\pi\in\Prob(\mathcal{X}\times\mathcal{Y})$ とし，
  $\nu(B):=\pi(\mathcal{X}\times B)$ を第2周辺分布とする．
  このとき，確率測度の族 $(\pi_y)_{y\in\mathcal{Y}}$ で，次をみたすものが存在する．
  \begin{enumerate}
    \item すべての $y\in\mathcal{Y}$ で $\pi_y\in\Prob(\mathcal{X})$ であり，
      各 $A\in\Borel(\mathcal{X})$ に対して $y\mapsto\pi_y(A)$ は Borel 可測である．
    \item 任意の $A\in\Borel(\mathcal{X})$，$B\in\Borel(\mathcal{Y})$ に対して
      \[
        \pi(A\times B)=\int_B\pi_y(A)\,\d\nu(y).
      \]
  \end{enumerate}
  さらに，任意の非負 Borel 可測関数
  $F\colon\mathcal{X}\times\mathcal{Y}\to[0,+\infty]$ に対し，
  $y\mapsto\int_{\mathcal{X}}F(x,y)\,\d\pi_y(x)$ は Borel 可測で，
  \[
    \int_{\mathcal{X}\times\mathcal{Y}}F(x,y)\,\d\pi(x,y)
    =\int_{\mathcal{Y}}\left(\int_{\mathcal{X}}F(x,y)\,\d\pi_y(x)\right)\d\nu(y).
  \]
  $F$ が $\pi$-可積分な実数値 Borel 関数の場合も，内側の積分は
  $\nu$-a.e. に絶対収束し，同じ公式が成り立つ．
  同じ条件をみたす族 $(\widetilde\pi_y)$ があれば，ある Borel 集合
  $N\subset\mathcal{Y}$，$\nu(N)=0$ が存在し，
  \[
    \pi_y=\widetilde\pi_y\qquad(\text{すべての }y\notin N)
  \]
  が成り立つ．すなわち，例外集合は $A$ によらず共通に取れる．
  第1周辺分布を用いた表示も，二つの座標を入れ替えれば得られる．
\end{theorem}

\begin{proof}
  \textbf{1. $\mathcal{X}$ が compact の場合：積分の候補を作る．}
  この段階では $\pi$ は零でない有限測度でもよい．
  $f\in C(\mathcal{X})$ に対して
  \[
    \nu_f(B):=\int_{\mathcal{X}\times B}f(x)\,\d\pi(x,y)
    \qquad(B\in\Borel(\mathcal{Y}))
  \]
  とおく．$\nu_f$ は有限符号付き測度であり，
  $\abs{\nu_f(B)}\leq\lVert f\rVert_\infty\nu(B)$ より $\nu_f\ll\nu$ である．
  Radon--Nikodym の定理（定理~\ref{thm:meas-radon-nikodym}）から，
  Borel 可測関数 $l_f$ で $\nu_f(B)=\int_B l_f\,\d\nu$ となるものを取る．

  注意~\ref{rem:meas-compact-continuous}の可算な稠密部分空間
  $\mathcal{C}\subset C(\mathcal{X})$ を取る．
  Radon--Nikodym 微分の一意性と積分の単調性より，
  各 $f,g\in\mathcal{C}$，$a,b\in\mathbb{Q}$ について，$\nu$-a.e. に
  \[
    \begin{gathered}
      l_{af+bg}=a l_f+b l_g,\qquad l_1=1,\\
      f\geq0\ \Longrightarrow\ l_f\geq0,\qquad
      \abs{l_f}\leq\lVert f\rVert_\infty
    \end{gathered}
  \]
  が成り立つ．例えば最初の等式は $\nu_{af+bg}=a\nu_f+b\nu_g$ による．
  条件は可算個しかないので，その例外集合を合併すれば，
  一つの Borel 零集合 $N$ の外ですべてが同時に成り立つ．

  \textbf{2. 積分の候補から測度を得る．}
  $y\notin N$ を固定する．$f\mapsto l_f(y)$ は $\mathcal{C}$ 上で有理数について線形で，
  \[
    \abs{l_f(y)-l_g(y)}\leq\lVert f-g\rVert_\infty
  \]
  をみたす．一様近似により $C(\mathcal{X})$ 上の実線形汎関数 $L_y$ に一意に延長できる．
  この延長も正である：$f\geq0$，$g_n\in\mathcal{C}$，
  $\lVert g_n-f\rVert_\infty\to0$ なら，
  $r_n>\lVert g_n-f\rVert_\infty$，$r_n\to0$ となる有理数を選ぶと
  $g_n+r_n\geq0$ なので，$L_y(f)=\lim_n l_{g_n+r_n}(y)\geq0$ となる．
  また $L_y(1)=1$ である．
  Riesz の表現定理（定理~\ref{thm:meas-riesz-representation}）により
  \[
    L_y(f)=\int_{\mathcal{X}}f\,\d\pi_y\qquad(f\in C(\mathcal{X}))
  \]
  をみたす $\pi_y\in\Prob(\mathcal{X})$ が得られる．
  $y\in N$ では，固定した $x_0\in\mathcal{X}$ に対する
  Dirac 測度 $\pi_y:=\delta_{x_0}$，$\delta_{x_0}(A):=\mathbf{1}_A(x_0)$ と定める．

  $f\in\mathcal{C}$ について $\int f\,\d\pi_y=l_f(y)$ が $N$ の外で成り立つから，
  左辺は $y$ の Borel 可測関数であり，
  \[
    \int_B\left(\int_{\mathcal{X}}f\,\d\pi_y\right)\d\nu(y)
    =\int_{\mathcal{X}\times B}f(x)\,\d\pi(x,y)
    \qquad(B\in\Borel(\mathcal{Y}))
  \]
  をみたす．一様近似により，同じことがすべての $f\in C(\mathcal{X})$ に成り立つ．
  閉集合 $A\neq\emptyset$ の指示関数は
  $f_m(x):=\max\{1-m\,d(x,A),0\}\to\mathbf{1}_A(x)$ と近似できるので，
  有界収束定理から $y\mapsto\pi_y(A)$ の可測性と
  $\pi(A\times B)=\int_B\pi_y(A)\,\d\nu(y)$ が従う（$A=\emptyset$ も明らか）．
  この二つの性質をみたす $A$ の全体は，補集合と互いに素な可算合併で閉じた
  $\lambda$-系である．閉集合は Borel 集合を生成する $\pi$-系なので，
  Dynkin の定理（定理~\ref{thm:meas-pi-lambda}）から，すべての Borel 集合 $A$ で成り立つ．

  \textbf{3. 一般の Polish 空間へ拡張する．}
  第1周辺分布 $\mu(A):=\pi(A\times\mathcal{Y})$ は tight である
  （定理~\ref{thm:meas-prokhorov}を $\{\mu\}$ に適用）．
  したがって，増大する compact 集合列 $(K_n)$ で
  $\mu(\bigcup_nK_n)=1$ となるものが取れる．
  $K_0:=\emptyset$，$E_n:=K_n\setminus K_{n-1}$ とおき，
  $\pi$ を $E_n\times\mathcal{Y}$ に制限した有限測度を
  $K_n\times\mathcal{Y}$ 上の測度 $\pi^{(n)}$ とみなす．
  その第2周辺測度は $\nu_n(B):=\pi(E_n\times B)$ であり，$\sum_n\nu_n=\nu$ である．
  零測度となる項は以下の和から除く．

  1--2 を $\pi^{(n)}$ に適用して得られる族を，$K_n$ の外では質量 $0$ として
  $\mathcal{X}$ 上に延長し，$(\rho_y^{(n)})$ と書く．すると
  \[
    \pi\bigl((A\cap E_n)\times B\bigr)
    =\int_B\rho_y^{(n)}(A)\,\d\nu_n(y).
  \]
  $\nu_n\ll\nu$ なので非負の Borel 可測関数 $h_n:=\d\nu_n/\d\nu$ を取ると，
  単調収束定理と微分の一意性から $\sum_nh_n=1$ が $\nu$-a.e. に成り立つ．
  この等式が成り立つ $y$ では
  \[
    \pi_y(A):=\sum_n h_n(y)\rho_y^{(n)}(A)
  \]
  と定め，残りの Borel 零集合上では，固定した $x_0\in\mathcal{X}$ に対して
  $\pi_y:=\delta_{x_0}$ とする．
  これは各 $y$ で確率測度を定め，$y\mapsto\pi_y(A)$ は Borel 可測である．
  $\d\nu_n=h_n\,\d\nu$ と単調収束定理より
  \[
    \begin{aligned}
      \int_B\pi_y(A)\,\d\nu
      &=\sum_n\int_B h_n(y)\rho_y^{(n)}(A)\,\d\nu(y)\\
      &=\sum_n\int_B\rho_y^{(n)}(A)\,\d\nu_n(y)\\
      &=\sum_n\pi\bigl((A\cap E_n)\times B\bigr)
      =\pi(A\times B).
    \end{aligned}
  \]
  ここで密度による積分の置き換えは，指示関数に対する
  $\nu_n(B)=\int_Bh_n\,\d\nu$ を単関数近似で拡張したものである．

  \textbf{4. 積分表示と一意性．}
  $D\in\Borel(\mathcal{X}\times\mathcal{Y})$ に対して
  $y\mapsto\int_{\mathcal{X}}\mathbf{1}_D(x,y)\,\d\pi_y(x)$ が可測であり，
  その $\nu$ による積分が $\pi(D)$ に等しくなることを示す．
  この性質をもつ $D$ の全体は，単調収束定理により $\lambda$-系をなし，
  2 の矩形上の公式（一般の場合は 3）によりすべての矩形を含む．
  よって Dynkin の定理と注意~\ref{rem:meas-borel-product}から，すべての Borel 集合で成り立つ．
  非負単関数による近似と単調収束定理で，主張した $F$ の積分表示を得る．
  可積分な $F$ については，$\abs{F}$ にこの公式を適用すれば内側の絶対可積分性が分かり，
  $F$ の正部分・負部分の公式の差を取ればよい．

  最後に，同じ条件をみたす族 $(\widetilde\pi_y)$ を取る．
  $\mathcal{X}$ の可算稠密集合を中心とする有理数半径の開球の有限交叉と
  $\mathcal{X}$ 全体からなる族 $\mathcal{G}$ は，
  $\Borel(\mathcal{X})$ を生成する可算な $\pi$-系である．
  各 $A\in\mathcal{G}$ について，$\pi_y(A)$ と $\widetilde\pi_y(A)$ はともに
  $B\mapsto\pi(A\times B)$ の $\nu$ に関する Radon--Nikodym 微分なので，$\nu$-a.e. に一致する．
  $\mathcal{G}$ は可算だから，共通の Borel 零集合 $N$ の外で
  すべての $A\in\mathcal{G}$ について一致する．
  各 $y\notin N$ で有限測度の一致定理（定理~\ref{thm:meas-uniqueness}）を適用すれば，
  $\pi_y=\widetilde\pi_y$ を得る．
\end{proof}

\begin{corollary}{Borel 写像による測度分解（北別府 Theorem 5.3.1）}{meas-disintegration-map}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間，$\mu\in\Prob(\mathcal{X})$，
  $T\colon\mathcal{X}\to\mathcal{Y}$ を Borel 可測写像，$\nu:=\mu\circ T^{-1}$ とする．
  このとき，$\mu_y\in\Prob(\mathcal{X})$ の族 $(\mu_y)_{y\in\mathcal{Y}}$ で，
  各 $A\in\Borel(\mathcal{X})$ に対して $y\mapsto\mu_y(A)$ が Borel 可測であり，
  \[
    \mu_y\bigl(T^{-1}(\{y\})\bigr)=1\quad(\nu\text{-a.e. }y),
  \]
  \[
    \int_{\mathcal{X}}f\,\d\mu
    =\int_{\mathcal{Y}}\left(\int_{\mathcal{X}}f\,\d\mu_y\right)\d\nu(y)
  \]
  をみたすものが存在する（$f$ は任意の非負 Borel 可測関数）．
  これを $\mu$ の $T$ による\textbf{測度分解}という．
  この族は，共通の $\nu$-零集合の外で一意である．
\end{corollary}

\begin{proof}
  $G(x):=(x,T(x))$ による像測度 $\pi:=\mu\circ G^{-1}$ に
  定理~\ref{thm:meas-disintegration}を適用し，得られる族を $(\mu_y)$ とする．
  $F(x,y)=f(x)$ に対する積分表示が，主張の積分公式を与える．
  $T$ のグラフ $\Gamma_T:=\{(x,y)\mid T(x)=y\}$ は
  Borel 関数 $(x,y)\mapsto d_{\mathcal{Y}}(T(x),y)$ の零点集合なので Borel 集合である．
  $\pi(\Gamma_T)=1$ より，$F=\mathbf{1}_{\Gamma_T}$ に対する積分表示から
  \[
    1=\int_{\mathcal{Y}}\mu_y\bigl(T^{-1}(\{y\})\bigr)\,\d\nu(y)
  \]
  を得る．被積分関数は $[0,1]$ に値を取るので，$\nu$-a.e. に $1$ である．
  また，任意の分解について，$T^{-1}(\{y\})$ に集中することと
  $f=\mathbf{1}_{A\cap T^{-1}(B)}$ に対する公式から
  $\mu(A\cap T^{-1}(B))=\int_B\mu_y(A)\,\d\nu(y)$ を得る．
  したがって一意性も定理~\ref{thm:meas-disintegration}から従う．
\end{proof}

\begin{remark}{参考資料との対応}{meas-disintegration-references}
  北別府の証明の「Radon--Nikodym 微分，可算個の条件の同時成立，
  Riesz の表現定理，compact な部分への分割」という流れを用いた．
  ここでは接着補題に必要な直積上の形を先に証明したので，
  compact な場合にも $\mathcal{Y}$ の compact 性を要さず，
  一般の場合への拡張には第1周辺分布の tightness だけで足りる．
  一般の Borel 写像による形は直前の系で得られる．
  桑江ほか (2015) 補題 1.32 も測度分解定理であり，
  Villani (2009) p.~12 は接着補題の説明でこの分解を用いている．
\end{remark}

\subsection{接着補題}

\begin{lemma}{接着補題（gluing lemma；Villani (2009) pp.~11--12）}{wp-gluing}
  $\mathcal{X},\mathcal{Y},\mathcal{Z}$ を Polish 空間とする．
  $\mu\in\Prob(\mathcal{X})$，$\nu\in\Prob(\mathcal{Y})$，
  $\xi\in\Prob(\mathcal{Z})$，$\pi\in\Couplings(\mu,\nu)$，
  $\sigma\in\Couplings(\nu,\xi)$ とする．
  このとき $\gamma\in\Prob(\mathcal{X}\times\mathcal{Y}\times\mathcal{Z})$ で，任意の
  $A\in\Borel(\mathcal{X})$，$B\in\Borel(\mathcal{Y})$，
  $C\in\Borel(\mathcal{Z})$ に対し
  \[
    \begin{aligned}
      \gamma(A\times B\times\mathcal{Z})&=\pi(A\times B),\\
      \gamma(\mathcal{X}\times B\times C)&=\sigma(B\times C)
    \end{aligned}
  \]
  をみたすものが存在する．
  すなわち，$\gamma$ の第 $1,2$ 成分の周辺分布が $\pi$ に，
  第 $2,3$ 成分の周辺分布が $\sigma$ に一致する．

  \demohint{gluing}
\end{lemma}

\begin{proof}
  共通の変数 $y$ に沿って二つの測度を分解し，得られた分布を直積にしてから $\nu$ で平均する．
  各 $y\in\mathcal{Y}$ について，次のように確率測度 $\pi_y,\sigma_y$ を取る．
  $\pi$ の第2周辺分布は $\nu$ なので，定理~\ref{thm:meas-disintegration}により
  \[
    \pi(A\times B)=\int_B\pi_y(A)\,\d\nu(y)
    \qquad(A\in\Borel(\mathcal{X}),\,B\in\Borel(\mathcal{Y}))
  \]
  をみたす $\pi_y\in\Prob(\mathcal{X})$ が取れる．
  同様に，$\sigma$ の第1周辺分布は $\nu$ なので，同じ定理により
  \[
    \sigma(B\times C)=\int_B\sigma_y(C)\,\d\nu(y)
    \qquad(B\in\Borel(\mathcal{Y}),\,C\in\Borel(\mathcal{Z}))
  \]
  をみたす $\sigma_y\in\Prob(\mathcal{Z})$ が取れる．
  同定理により，$y\mapsto\pi_y(A)$ と $y\mapsto\sigma_y(C)$ はともに Borel 可測である．

  そこで，$\gamma\in\Prob(\mathcal{X}\times\mathcal{Y}\times\mathcal{Z})$ を
  \[
    \gamma(A\times B\times C):=\int_B\pi_y(A)\,\sigma_y(C)\,\d\nu(y)
  \]
  と定める．このような確率測度がただ一つ存在することは
  定理~\ref{thm:meas-kernel-composition}による．
  $\sigma_y(\mathcal{Z})=1$ より，$C=\mathcal{Z}$ とすると
  \[
    \gamma(A\times B\times\mathcal{Z})=\int_B\pi_y(A)\,\d\nu(y)=\pi(A\times B)
  \]
  である．同様に $\pi_y(\mathcal{X})=1$ より，$A=\mathcal{X}$ とすると
  \[
    \gamma(\mathcal{X}\times B\times C)=\int_B\sigma_y(C)\,\d\nu(y)=\sigma(B\times C)
  \]
  である．
\end{proof}

\subsection{距離の公理の確認}

\begin{proposition}{Wasserstein 距離は距離である（Villani Def 6.4）}{wp-metric}
  $1\leq p<\infty$ に対し，
  $\Wass_p$ は $\Pp{p}(\mathcal{X})$ 上の距離である．
\end{proposition}

\begin{proof}
  $\mu,\nu\in\Pp{p}(\mathcal{X})$ を任意に取る．
  距離の公理（定義~\ref{def:meas-metric-space}）を順に確かめる．

  \textbf{1. 非負性．}
  すべての $x,y\in\mathcal{X}$ で $d(x,y)\geq0$ なので，
  任意の $\pi\in\Couplings(\mu,\nu)$ に対し
  $\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)\geq0$ であり，
  $\Wass_p(\mu,\nu)\geq0$ である．
  なお，$\pi$ として直積測度 $\mu\otimes\nu$ を用いれば
  $\Wass_p(\mu,\nu)<\infty$ も従う（補題~\ref{lem:wp-finite}）．

  \textbf{2. 一致の公理．}
  $\Wass_p(\mu,\nu)=0\iff\mu=\nu$ を示す．
  対角集合を $\Delta:=\{(x,x)\mid x\in\mathcal{X}\}$ とおく．
  $\Delta=\{(x,y)\in\mathcal{X}\times\mathcal{X}\mid d(x,y)=0\}$ であり，
  $d$ は連続なので（注意~\ref{rem:wp-welldef}），$\Delta$ は閉集合，特に Borel 集合である．

  ($\Leftarrow$) $\mu=\nu$ とする．
  $\pi(\Delta)=1$ となる（$\Delta$ に集中する）$\pi\in\Couplings(\mu,\mu)$ を取る．
  実際，$T(x):=(x,x)$ による像測度 $\pi:=\mu\circ T^{-1}$ がそのような coupling である．
  $A\in\Borel(\mathcal{X})$ に対し $T^{-1}(A\times\mathcal{X})=T^{-1}(\mathcal{X}\times A)=A$
  より $\pi(A\times\mathcal{X})=\pi(\mathcal{X}\times A)=\mu(A)$ であり，
  $T^{-1}(\Delta)=\mathcal{X}$ より $\pi(\Delta)=1$ である．
  $\Delta$ 上で $d(x,y)=0$ なので，
  \[
    \Wass_p(\mu,\mu)^p
    \leq\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)=0.
  \]
  1 の非負性とあわせて $\Wass_p(\mu,\mu)=0$ である．

  ($\Rightarrow$) $\Wass_p(\mu,\nu)=0$ とする．
  最適 coupling の存在（定理~\ref{thm:wp-optimal-coupling-existence}）より，
  ある $\pi^*\in\Couplings(\mu,\nu)$ が存在して
  \[
    \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi^*(x,y)=0
  \]
  となる．被積分関数は非負なので，$d(x,y)=0$ が $\pi^*$-a.e. で成り立つ．
  よって $x=y$ が $\pi^*$-a.e. で成り立ち，$\pi^*(\Delta)=1$ である．
  任意の $A\in\Borel(\mathcal{X})$ に対し，
  $A\times(\mathcal{X}\setminus A)$ と $(\mathcal{X}\setminus A)\times A$ は
  $\Delta$ と交わらないので，
  \[
    \begin{aligned}
      \mu(A)&=\pi^*(A\times\mathcal{X})
      =\pi^*(A\times A)+\pi^*\bigl(A\times(\mathcal{X}\setminus A)\bigr)
      =\pi^*(A\times A),\\
      \nu(A)&=\pi^*(\mathcal{X}\times A)
      =\pi^*(A\times A)+\pi^*\bigl((\mathcal{X}\setminus A)\times A\bigr)
      =\pi^*(A\times A).
    \end{aligned}
  \]
  $A$ は任意なので $\mu=\nu$ である．

  \textbf{3. 対称性．}
  $S(x,y):=(y,x)$ とし，$\pi\in\Couplings(\mu,\nu)$ に対し
  $\pi^\top:=\pi\circ S^{-1}$ と定める．
  $\pi^\top(A\times B)=\pi(B\times A)$ より $\pi^\top\in\Couplings(\nu,\mu)$ である．
  $d$ の対称性と変数変換公式（補題~\ref{lem:meas-change-of-variables}）より
  \[
    \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    =\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi^\top(x,y)
  \]
  であるから，$\pi$ について下限を取れば $\Wass_p(\nu,\mu)\leq\Wass_p(\mu,\nu)$ を得る．
  $\mu,\nu$ を入れ替えれば逆向きの不等式も従い，$\Wass_p(\mu,\nu)=\Wass_p(\nu,\mu)$ である．

  \textbf{4. 三角不等式．}
  $\xi\in\Pp{p}(\mathcal{X})$ を取り，
  $\Wass_p(\mu,\xi)\leq\Wass_p(\mu,\nu)+\Wass_p(\nu,\xi)$ を示す．
  最適 coupling の存在（定理~\ref{thm:wp-optimal-coupling-existence}）より，
  最適 coupling $\pi\in\Couplings(\mu,\nu)$，$\sigma\in\Couplings(\nu,\xi)$ を取る．
  接着補題（補題~\ref{lem:wp-gluing}）より，ある $\gamma\in\Prob(\mathcal{X}^3)$ が存在して
  \[
    \gamma(A\times B\times\mathcal{X})=\pi(A\times B),\qquad
    \gamma(\mathcal{X}\times B\times C)=\sigma(B\times C)
    \qquad(A,B,C\in\Borel(\mathcal{X}))
  \]
  をみたす．ここで，$\gamma$ の第1成分と第3成分を取り出した確率測度 $\rho$ を
  \[
    \rho(E):=\gamma\bigl(\{(x,y,z)\in\mathcal{X}^3\mid(x,z)\in E\}\bigr)
    \qquad(E\in\Borel(\mathcal{X}\times\mathcal{X}))
  \]
  と定める．このとき $\rho(A\times C)=\gamma(A\times\mathcal{X}\times C)$ であり，
  \[
    \rho(A\times\mathcal{X})=\gamma(A\times\mathcal{X}\times\mathcal{X})
    =\pi(A\times\mathcal{X})=\mu(A),\qquad
    \rho(\mathcal{X}\times C)=\gamma(\mathcal{X}\times\mathcal{X}\times C)
    =\sigma(\mathcal{X}\times C)=\xi(C)
  \]
  より $\rho\in\Couplings(\mu,\xi)$ である．したがって
  \[
    \begin{aligned}
      \Wass_p(\mu,\xi)
      &\leq\left(\int_{\mathcal{X}^2}d(x,z)^p\,\d\rho(x,z)\right)^{1/p}\\
      &=\left(\int_{\mathcal{X}^3}d(x,z)^p\,\d\gamma(x,y,z)\right)^{1/p}\\
      &\leq\left(\int_{\mathcal{X}^3}\bigl(d(x,y)+d(y,z)\bigr)^p\,\d\gamma(x,y,z)\right)^{1/p}\\
      &\leq
      \left(\int_{\mathcal{X}^3}d(x,y)^p\,\d\gamma(x,y,z)\right)^{1/p}
      +\left(\int_{\mathcal{X}^3}d(y,z)^p\,\d\gamma(x,y,z)\right)^{1/p}\\
      &=\left(\int_{\mathcal{X}^2}d(x,y)^p\,\d\pi(x,y)\right)^{1/p}
      +\left(\int_{\mathcal{X}^2}d(y,z)^p\,\d\sigma(y,z)\right)^{1/p}\\
      &=\Wass_p(\mu,\nu)+\Wass_p(\nu,\xi).
    \end{aligned}
  \]
  順に，下限の定義（$\rho\in\Couplings(\mu,\xi)$），
  $\rho$ の定義と変数変換公式（補題~\ref{lem:meas-change-of-variables}），
  $d$ の三角不等式，Minkowski の不等式（定理~\ref{thm:meas-minkowski}），
  $\gamma$ の第 $1,2$ 成分の周辺分布が $\pi$，第 $2,3$ 成分の周辺分布が $\sigma$ であること
  と周辺積分公式（補題~\ref{lem:meas-marginal-integral}），
  $\pi,\sigma$ の最適性を用いた．

  1--4 より，$(\Pp{p}(\mathcal{X}),\Wass_p)$ は距離空間である．
\end{proof}

\begin{definition}{Wasserstein 空間（Villani Def 6.4）}{wp-wasserstein-space}
  $1\leq p<\infty$ に対し，距離 $\Wass_p$ を備えた
  $\Pp{p}(\mathcal{X})$ を $p$ 次の \textbf{Wasserstein 空間}といい，
  \[
    (\Pp{p}(\mathcal{X}),\Wass_p)
  \]
  と書く．
\end{definition}

\begin{remark}{文献との対応}{wp-metric-references}
  存在定理は Villani (2009) Theorem 4.1 をコスト $d^p$ に特殊化したものであり，
  二つの補題は同 Lemmas 4.3, 4.4 に対応する．
  距離性の証明は同 Chapter 6 に従う．接着補題は同 Chapter 1 の
  Gluing Lemma（桑江ほか (2015) 補題 1.33）である．
  なお Villani (2009) Definition 6.1, 6.4 は
  まず $\Prob(\mathcal{X})$ 上に $\Wass_p$ を定義し，
  その後 $\Pp{p}(\mathcal{X})$ に制限して有限値の距離とするが，
  本資料でははじめから $\Pp{p}(\mathcal{X})$ 上で定義した．
\end{remark}
