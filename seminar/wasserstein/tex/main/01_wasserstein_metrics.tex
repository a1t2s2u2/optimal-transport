\chapter{Wasserstein 距離 \texorpdfstring{$W_p$}{Wp}}
\label{ch:wasserstein-metrics}

\section{設定と定義}
\label{sec:wp-definition}

$(\mathcal{X},d)$ を Polish 空間（定義~\ref{def:meas-polish}）とする．
Borel $\sigma$-代数 $\Borel(\mathcal{X})$ 上の確率測度全体を
\[
  \Prob(\mathcal{X})
  :=
  \left\{
    \mu\colon\Borel(\mathcal{X})\to[0,1]
    \;\middle|\;
    \mu\text{ は測度},\;\mu(\mathcal{X})=1
  \right\}
\]
とおく．

\begin{definition}{有限 $p$ 次モーメントをもつ確率測度（Villani Def 6.4）}{wp-p-space}
  任意に点 $x_0\in \mathcal{X}$ を一つ固定する．$1\leq p<\infty$ に対し
  \[
    \Pp{p}(\mathcal{X})
    :=
    \left\{
      \mu\in\Prob(\mathcal{X})
      \;\middle|\;
      \int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty
    \right\}
  \]
  と定める．積分 $\int_{\mathcal{X}}d(x,x_0)^p\,\d\mu(x)$ を
  $\mu$ の（$x_0$ を基点とする）\textbf{$p$ 次モーメント}という．
\end{definition}

\begin{lemma}{$\Pp{p}(\mathcal{X})$ の $x_0$ 非依存性（Villani Def 6.4）}{wp-p-indep}
  $1\leq p<\infty$ のとき，$\Pp{p}(\mathcal{X})$ は点 $x_0$ の選び方によらない．
  すなわち，任意の $x_0,x_0'\in \mathcal{X}$ と $\mu\in\Prob(\mathcal{X})$ に対して
  \[
    \int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty
    \iff
    \int_\mathcal{X} d(x,x_0')^p\,\d\mu(x)<\infty
  \]
  が成り立つ．
\end{lemma}

\begin{proof}
  基点を $x_0$ から $x_0'$ に変えても，距離は高々 $d(x_0,x_0')$ しか増えない．
  三角不等式と，凸性から従う評価
  $(a+b)^p\leq2^{p-1}(a^p+b^p)$（$a,b\geq0$；注意~\ref{rem:meas-real-facts}）により，
  \[
    \int_\mathcal{X}d(x,x_0')^p\,\d\mu(x)
    \leq
    2^{p-1}\left(\int_\mathcal{X}d(x,x_0)^p\,\d\mu(x)+d(x_0,x_0')^p\right).
  \]
  したがって，$x_0$ を基点とする積分が有限なら，$x_0'$ を基点とする積分も有限である．
  $x_0,x_0'$ を入れ替えれば逆も従う．
\end{proof}

\begin{definition}{Coupling（Villani Def 1.1）}{wp-coupling}
  距離空間 $\mathcal{X},\mathcal{Y}$ 上の確率測度
  $\mu\in\Prob(\mathcal{X})$，$\nu\in\Prob(\mathcal{Y})$ に対し
  \[
    \Couplings(\mu,\nu)
    :=
    \left\{
      \pi\in\Prob(\mathcal{X}\times\mathcal{Y})
      \;\middle|\;
      \begin{aligned}
        \pi(A\times\mathcal{Y})&=\mu(A) &&(\forall A\in\Borel(\mathcal{X})),\\
        \pi(\mathcal{X}\times B)&=\nu(B) &&(\forall B\in\Borel(\mathcal{Y}))
      \end{aligned}
    \right\}
  \]
  と定め，$\pi\in\Couplings(\mu,\nu)$ を $\mu$ と $\nu$ の
  \textbf{coupling} という．
  二つの条件は，送り元の分布が $\mu$，送り先の分布が $\nu$ であることを表す．
\end{definition}

\begin{definition}{Wasserstein 距離（Villani Def 6.1）}{wp-wasserstein}
  $1\leq p<\infty$ と $\mu,\nu\in\Pp{p}(\mathcal{X})$ に対し
  \[
    \Wass_p(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \left(
      \int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    \right)^{1/p}
  \]
  と定める．
\end{definition}

\begin{remark}{距離関数の可測性}{wp-welldef}
  距離関数 $d\colon \mathcal{X}\times \mathcal{X}\to[0,\infty)$ は，三角不等式から従う評価
  \[
    \abs{d(x,y)-d(x',y')}\leq d(x,x')+d(y,y')
  \]
  により連続，したがって Borel 可測である
  （注意~\ref{rem:meas-measurable-ops}）．
\end{remark}

\begin{lemma}{$\Wass_p$ の有限性（Villani Def 6.4）}{wp-finite}
  $1\leq p<\infty$，$\mu,\nu\in\Pp{p}(\mathcal{X})$ とすると，
  $\Wass_p(\mu,\nu)<\infty$ である．
\end{lemma}

\begin{proof}
  輸送コストを，送り元と送り先から基点 $x_0$ までの距離で抑える．
  三角不等式と補題~\ref{lem:wp-p-indep}の証明で用いた凸性の評価から，
  \[
    d(x,y)^p\leq2^{p-1}\bigl(d(x,x_0)^p+d(x_0,y)^p\bigr).
  \]
  直積確率測度 $\pi_0:=\mu\otimes\nu$ は $\mu,\nu$ の coupling である
  （定理~\ref{thm:meas-product}）．その周辺分布は $\mu,\nu$ なので，
  周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  \[
    \begin{aligned}
      \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi_0(x,y)
      &\leq2^{p-1}\left(
        \int_\mathcal{X}d(x,x_0)^p\,\d\mu(x)
        +\int_\mathcal{X}d(x_0,y)^p\,\d\nu(y)
      \right)\\
      &<\infty.
    \end{aligned}
  \]
  最後の不等式は $\mu,\nu\in\Pp{p}(\mathcal{X})$ による．
  こうして有限のコストをもつ coupling が一つ得られたので，
  その下限から定まる $\Wass_p(\mu,\nu)$ も有限である．
\end{proof}

\section{最適 coupling の存在}
\label{sec:wp-existence}

この節では，コストが下限に近づく coupling の列を取り，
その極限で下限を達成する coupling を得る．
証明は次の四段階からなる．
\begin{enumerate}
  \item コストが下限に近づく coupling の列 $(\pi_k)$ を取る．
  \item coupling の tightness と Prokhorov の定理を用いて，
    弱収束部分列 $\pi_{k_j}\weakto\pi^*$ を取る．
  \item 弱極限 $\pi^*$ の二つの周辺分布が $\mu,\nu$ のままであることを示す．
  \item 輸送コストの下半連続性を用いて，$\pi^*$ が下限を達成することを示す．
\end{enumerate}
以下では，2に必要な tightness と，4に必要な下半連続性を先に準備する．
以下，積分域が明らかなときは
$\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)$ を
$\int d^p\,\d\pi$ と略記する．また，弱収束の定義
（定義~\ref{def:meas-weak-convergence}）をここで再掲すると，
\[
  \pi_k\weakto\pi
  \quad\overset{\mathrm{def}}{\Longleftrightarrow}\quad
  \int F\,\d\pi_k\longrightarrow\int F\,\d\pi
  \qquad
  (\forall F\in C_b(\mathcal{X}\times\mathcal{X})).
\]

\begin{lemma}{$p$ 次輸送コストの下半連続性（Villani Lem 4.3）}{wp-cost-lsc}
  $1\leq p<\infty$ とする．
  $\Prob(\mathcal{X}\times\mathcal{X})$ の列 $(\pi_k)$ が
  $\pi\in\Prob(\mathcal{X}\times\mathcal{X})$ に弱収束，すなわち
  $\pi_k\weakto\pi$ とする．このとき
  \[
    \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    \leq
    \liminf_{k\to\infty}
    \int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi_k(x,y).
  \]
\end{lemma}

\begin{proof}
  弱収束で積分の収束が分かるのは有界連続関数なので，
  まず $d^p$ の大きい値を $m\in\N$ で切り捨て，
  \[
    c_m(x,y):=\min\{d(x,y)^p,m\}
  \]
  とおく．$d$ は連続（注意~\ref{rem:wp-welldef}）だから $c_m$ は有界連続である．
  各 $m$ について，弱収束の定義（定義~\ref{def:meas-weak-convergence}）と
  $c_m\leq d^p$ より
  \[
    \int c_m\,\d\pi
    =\lim_{k\to\infty}\int c_m\,\d\pi_k
    \leq\liminf_{k\to\infty}\int d^p\,\d\pi_k.
  \]
  $m$ を増やすと $c_m$ は単調に増加して $d^p$ に近づく．
  右辺は $m$ によらないので，単調収束定理（定理~\ref{thm:meas-mct}）から
  \[
    \int d^p\,\d\pi
    =\lim_{m\to\infty}\int c_m\,\d\pi
    \leq\liminf_{k\to\infty}\int d^p\,\d\pi_k
  \]
  を得る．
\end{proof}

\begin{remark}{記法：測度の族}{wp-family-symbol}
  Villani は測度の族を $\mathcal{P},\mathcal{Q}$ と書くが，
  本資料では確率測度全体を $\Prob(\mathcal{X})$ と書いているため
  記号が衝突する．そこで以下では族を $A,B$ で表す．
\end{remark}

\begin{lemma}{輸送計画の tightness（Villani Lem 4.4）}{wp-plans-tight}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間とし，
  $A\subset\Prob(\mathcal{X})$，$B\subset\Prob(\mathcal{Y})$ を
  tight（定義~\ref{def:meas-tight}）な確率測度の族とする．
  周辺分布がそれぞれ $A$，$B$ に属する coupling の全体を
  \[
    \Couplings(A,B)
    :=\bigcup_{\mu\in A,\,\nu\in B}\Couplings(\mu,\nu)
  \]
  と定めると，$\Couplings(A,B)$ は $\Prob(\mathcal{X}\times\mathcal{Y})$
  で tight である．
\end{lemma}

\begin{proof}
  $\varepsilon>0$ とする．$A,B$ の tightness より，compact 集合
  $K\subset\mathcal{X}$，$L\subset\mathcal{Y}$ を
  \[
    \eta(\mathcal{X}\setminus K)\leq\frac{\varepsilon}{2}
    \quad(\forall\eta\in A),
    \qquad
    \rho(\mathcal{Y}\setminus L)\leq\frac{\varepsilon}{2}
    \quad(\forall\rho\in B)
  \]
  となるように選ぶ．
  任意の $\pi\in\Couplings(A,B)$ を取り，その二つの周辺分布を
  $\eta\in A$，$\rho\in B$ とする．
  $(x,y)$ が $K\times L$ の外にあるなら，$x$ が $K$ の外にあるか，
  $y$ が $L$ の外にある．したがって
  \[
    \begin{aligned}
      \pi\bigl((\mathcal{X}\times\mathcal{Y})\setminus(K\times L)\bigr)
      &\leq\eta(\mathcal{X}\setminus K)+\rho(\mathcal{Y}\setminus L)\\
      &\leq\varepsilon.
    \end{aligned}
  \]
  同じ compact 集合 $K\times L$ で，すべての $\pi\in\Couplings(A,B)$ に
  対してこの評価が成り立つので，$\Couplings(A,B)$ は tight である．
\end{proof}

\begin{theorem}{$\Wass_p$ の最適 coupling の存在（Villani Thm 4.1；pp.~43--45）}{wp-optimal-coupling-existence}
  $\mathcal{X}$ を Polish 空間，$1\leq p<\infty$ とする．
  任意の $\mu,\nu\in\Pp{p}(\mathcal{X})$ に対して，
  ある $\pi^*\in\Couplings(\mu,\nu)$ が存在し，
  \[
    \Wass_p(\mu,\nu)
    =
    \left(
      \int_{\mathcal{X}\times\mathcal{X}}
      d(x,y)^p\,\d\pi^*(x,y)
    \right)^{1/p}
  \]
  が成り立つ．
  すなわち，$\Wass_p$ の定義に現れる下限は最小値として達成される．
  この $\pi^*$ を $\Wass_p$ の\textbf{最適 coupling}という．
\end{theorem}

\begin{proof}
  輸送コストとその下限を
  \[
    J(\pi):=\int_{\mathcal{X}\times\mathcal{X}}d(x,y)^p\,\d\pi(x,y),
    \qquad
    I:=\inf_{\pi\in\Couplings(\mu,\nu)}J(\pi)
  \]
  とおく．補題~\ref{lem:wp-finite}の証明より，coupling は存在し，$I<\infty$ である．
  $t\mapsto t^{1/p}$ は連続かつ単調増加なので，$I=\Wass_p(\mu,\nu)^p$ である．

  \textbf{1. コストを下限に近づける．}
  下限の定義より，各 $k\in\N$ に対して
  \[
    \pi_k\in\Couplings(\mu,\nu),
    \qquad
    I\leq J(\pi_k)<I+\frac1k
  \]
  となる $\pi_k$ を取る．これで $J(\pi_k)\to I$ となる．

  \textbf{2. 収束する部分列を取る．}
  $\{\mu\}$ の列は定数列なので，Prokhorov の定理
  （定理~\ref{thm:meas-prokhorov}）より $\{\mu\}$ は tight である．
  $\{\nu\}$ も同様であり，補題~\ref{lem:wp-plans-tight}から
  $\Couplings(\mu,\nu)$ は tight である．
  $\mathcal{X}\times\mathcal{X}$ は Polish 空間なので，再び Prokhorov の定理により，
  ある部分列と $\pi^*\in\Prob(\mathcal{X}\times\mathcal{X})$ に対して
  \[
    \pi_{k_j}\weakto\pi^*
  \]
  となる．

  \textbf{3. 極限でも送り元と送り先の分布は変わらない．}
  任意の有界連続関数 $\varphi\in C_b(\mathcal{X})$ に対し，
  $(x,y)\mapsto\varphi(x)$ も有界連続である．弱収束と周辺積分公式
  （補題~\ref{lem:meas-marginal-integral}）から
  \[
    \begin{aligned}
      \int_{\mathcal{X}\times\mathcal{X}}\varphi(x)\,\d\pi^*(x,y)
      &=\lim_{j\to\infty}
        \int_{\mathcal{X}\times\mathcal{X}}\varphi(x)\,\d\pi_{k_j}(x,y)\\
      &=\int_\mathcal{X}\varphi(x)\,\d\mu(x).
    \end{aligned}
  \]
  有界連続関数による測度の判定（定理~\ref{thm:meas-cb-determines}）より，
  $\pi^*$ の第1周辺分布は $\mu$ である．$y$ だけの関数で同様に調べれば，
  第2周辺分布は $\nu$ である．したがって $\pi^*\in\Couplings(\mu,\nu)$ である．

  \textbf{4. 極限のコストが下限に一致する．}
  $\pi^*$ も coupling なので $I\leq J(\pi^*)$ である．
  一方，コストの下半連続性（補題~\ref{lem:wp-cost-lsc}）と $J(\pi_{k_j})\to I$ より
  \[
    I\leq J(\pi^*)\leq\liminf_{j\to\infty}J(\pi_{k_j})=I.
  \]
  よって $J(\pi^*)=I=\Wass_p(\mu,\nu)^p$ となり，$\pi^*$ が最適 coupling である．
\end{proof}

\section{Wasserstein 距離の距離性}
\label{sec:metric-property}

前節の最適 coupling の存在を用いて，$\Wass_p$ の距離性を示す．
三角不等式の鍵は，共通の周辺分布をもつ二つの coupling を
一つの三変数の確率測度にまとめる，次の接着補題である．

\begin{lemma}{接着補題（gluing lemma；Villani (2009) pp.~11--12）}{wp-gluing}
  $\mathcal{X},\mathcal{Y},\mathcal{Z}$ を Polish 空間とする．
  $\mu\in\Prob(\mathcal{X})$，$\nu\in\Prob(\mathcal{Y})$，
  $\xi\in\Prob(\mathcal{Z})$，$\pi\in\Couplings(\mu,\nu)$，
  $\sigma\in\Couplings(\nu,\xi)$ とする．
  このとき $\gamma\in\Prob(\mathcal{X}\times\mathcal{Y}\times\mathcal{Z})$ で，任意の
  $A\in\Borel(\mathcal{X})$，$B\in\Borel(\mathcal{Y})$，
  $C\in\Borel(\mathcal{Z})$ に対し
  \[
    \begin{aligned}
      \gamma(A\times B\times\mathcal{Z})&=\pi(A\times B),\\
      \gamma(\mathcal{X}\times B\times C)&=\sigma(B\times C)
    \end{aligned}
  \]
  をみたすものが存在する．
  すなわち，$\gamma$ の第 $1,2$ 成分の周辺分布が $\pi$ に，
  第 $2,3$ 成分の周辺分布が $\sigma$ に一致する．

  \demohint{gluing}
\end{lemma}

\begin{proof}
  まず，$y$ が与えられたときの $x,z$ の分布にあたる確率測度
  $\pi_y\in\Prob(\mathcal{X})$，$\sigma_y\in\Prob(\mathcal{Z})$ を，
  \[
    \begin{aligned}
      \pi(A\times B)&=\int_B\pi_y(A)\,\d\nu(y),\\
      \sigma(B\times C)&=\int_B\sigma_y(C)\,\d\nu(y)
    \end{aligned}
  \]
  をみたすように取る．つまり，$y$ ごとの分布を $\nu$ で平均すると，
  元の $\pi,\sigma$ が得られる．このように，積分ができる形で
  $\pi_y,\sigma_y$ を選べることは，定理~\ref{thm:meas-disintegration}による．

  そこで，$y$ を共有してこの二つの分布の積を取り，
  \[
    \gamma(A\times B\times C)
    :=\int_B\pi_y(A)\,\sigma_y(C)\,\d\nu(y)
  \]
  と定める．これが確率測度を定めることは
  定理~\ref{thm:meas-kernel-composition}による．
  $\pi_y(\mathcal{X})=\sigma_y(\mathcal{Z})=1$ なので，
  $C=\mathcal{Z}$，$A=\mathcal{X}$ をそれぞれ代入すれば，
  求める二つの周辺分布の等式を得る．
\end{proof}

\begin{proposition}{Wasserstein 距離は距離である（Villani Def 6.4）}{wp-metric}
  $1\leq p<\infty$ に対し，
  $\Wass_p$ は $\Pp{p}(\mathcal{X})$ 上の距離である．
\end{proposition}

\begin{proof}
  $\mu,\nu\in\Pp{p}(\mathcal{X})$ を任意に取る．
  非負性は定義から，有限性は補題~\ref{lem:wp-finite}から従う．
  残る距離の公理（定義~\ref{def:meas-metric-space}）を確かめる．

  \textbf{1. 対称性．}
  $\mu$ から $\nu$ への coupling で，送り元 $x$ と送り先 $y$ を入れ替えると，
  $\nu$ から $\mu$ への coupling になる．
  $d(x,y)^p=d(y,x)^p$ なので，輸送コストは変わらない．
  逆向きにも同じことができるから，取りうる輸送コストは同じであり，
  その下限も等しい．したがって
  \[
    \Wass_p(\mu,\nu)=\Wass_p(\nu,\mu).
  \]

  \textbf{2. 一致の公理．}
  $\mu=\nu$ なら，各点の質量を動かさない coupling を取ればよい．
  どの点でも $d(x,x)=0$ なので，
  \[
    0\leq\Wass_p(\mu,\mu)
    \leq\left(\int_\mathcal{X}d(x,x)^p\,\d\mu(x)\right)^{1/p}=0.
  \]

  逆に $\Wass_p(\mu,\nu)=0$ とする．最適 coupling
  $\pi^*\in\Couplings(\mu,\nu)$ を取ると
  （定理~\ref{thm:wp-optimal-coupling-existence}），
  \[
    \int d(x,y)^p\,\d\pi^*(x,y)=\Wass_p(\mu,\nu)^p=0.
  \]
  コストは非負なので，積分が $0$ なら，正の距離を動く質量はない．
  すなわち $x=y$ が $\pi^*$-ほとんど至るところで成り立つ．
  したがって任意の $A\in\Borel(\mathcal{X})$ に対し，周辺積分公式
  （補題~\ref{lem:meas-marginal-integral}）から
  \[
    \mu(A)
    =\int\mathbf{1}_A(x)\,\d\pi^*(x,y)
    =\int\mathbf{1}_A(y)\,\d\pi^*(x,y)
    =\nu(A)
  \]
  となり，$\mu=\nu$ を得る．

  \textbf{3. 三角不等式．}
  $\xi\in\Pp{p}(\mathcal{X})$ を取る．
  $\mu$ から $\nu$，$\nu$ から $\xi$ への輸送をつなぎ，
  $x$ から $z$ への距離を，$x$ から $y$，$y$ から $z$ への距離の和で抑える．
  それぞれの最適 coupling
  $\pi\in\Couplings(\mu,\nu)$，$\sigma\in\Couplings(\nu,\xi)$ を取る．
  接着補題（補題~\ref{lem:wp-gluing}）により，この二つを周辺分布にもつ
  $\gamma\in\Prob(\mathcal{X}^3)$ を取る．
  送り元 $x$ と送り先 $z$ だけを見ると，その分布は $\mu,\xi$ の coupling になる．
  したがって
  \[
    \begin{aligned}
      \Wass_p(\mu,\xi)
      &\leq\left(\int_{\mathcal{X}^3}d(x,z)^p\,\d\gamma\right)^{1/p}\\
      &\leq\left(\int_{\mathcal{X}^3}\bigl(d(x,y)+d(y,z)\bigr)^p\,\d\gamma\right)^{1/p}\\
      &\leq
      \left(\int_{\mathcal{X}^3}d(x,y)^p\,\d\gamma\right)^{1/p}
      +\left(\int_{\mathcal{X}^3}d(y,z)^p\,\d\gamma\right)^{1/p}\\
      &=\Wass_p(\mu,\nu)+\Wass_p(\nu,\xi).
    \end{aligned}
  \]
  順に，$\Wass_p$ の定義，$d$ の三角不等式と積分の単調性，Minkowski の不等式
  （定理~\ref{thm:meas-minkowski}）を用いた．
  最後の等号は周辺積分公式と $\pi,\sigma$ の最適性による．
\end{proof}

\begin{definition}{Wasserstein 空間（Villani Def 6.4）}{wp-wasserstein-space}
  $1\leq p<\infty$ に対し，距離 $\Wass_p$ を備えた
  $\Pp{p}(\mathcal{X})$ を $p$ 次の \textbf{Wasserstein 空間}といい，
  \[
    (\Pp{p}(\mathcal{X}),\Wass_p)
  \]
  と書く．
\end{definition}

\begin{remark}{文献との対応}{wp-metric-references}
  存在定理は Villani (2009) Theorem 4.1 をコスト $d^p$ に特殊化したものであり，
  二つの補題は同 Lemmas 4.3, 4.4 に対応する．
  距離性の証明は同 Chapter 6 に従う．接着補題は同 Chapter 1 の
  Gluing Lemma（桑江ほか (2015) 補題 1.33）である．
  なお Villani (2009) Definition 6.1, 6.4 は
  まず $\Prob(\mathcal{X})$ 上に $\Wass_p$ を定義し，
  その後 $\Pp{p}(\mathcal{X})$ に制限して有限値の距離とするが，
  本資料でははじめから $\Pp{p}(\mathcal{X})$ 上で定義した．
\end{remark}
