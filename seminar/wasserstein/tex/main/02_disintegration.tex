\chapter{測度分解定理}
\label{ch:disintegration}

接着補題（補題~\ref{lem:wp-gluing}）で必要なのは，
二変数の分布 $\pi$ を，変数 $y$ ごとの確率測度 $\lambda_y$ に分け，
その平均として元の分布を復元することである．
北別府悠『最適輸送理論入門〜兼非標準的関数解析入門〜』Theorem 5.3.1 の
構成をもとに，接着補題で使う積空間上の形を示す．
原資料は一般の Borel 可測写像に沿う分解を扱うが，本章では第2変数に沿う分解に限定する．
分解測度 $\lambda_y$ と，その構成に用いる汎関数 $m_y$ の記号は原資料に合わせる．
本章では，まず測度分解定理を述べ，その準備となる二つの定理を証明してから，測度分解の証明に進む．
Radon--Nikodym の定理は有限測度，Riesz--Markov--Kakutani の表現定理は compact 距離空間の場合に限る．
測度分解の証明に用いるのは，この二つの形である．
積分の基本性質・単調収束定理・Dynkin の定理と距離空間の基本事項は既知とする．
Radon--Nikodym の準備として Riesz--Fréchet の表現定理を述べ，その証明は付録に置く．
Riesz--Markov--Kakutani の証明と一般の Polish 空間への拡張では，
第\ref{ch:wasserstein-metrics}章で用いた Prokhorov の定理を再利用する．

発表では，二つの定理の証明から測度分解の構成・一意性へ進む流れを軸とする．
$L^2$ の完備性と Riesz--Fréchet の表現定理，
$C(K)$ の可分性や Borel 集合に関する補助結果には付録で証明を付け，
本文の参照先から必要に応じて確認できるようにした．

\section{測度分解定理の主張}

\begin{theorem}{測度分解定理（接着補題で用いる形）}{meas-disintegration}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間，$\pi\in\Prob(\mathcal{X}\times\mathcal{Y})$ とし，
  第2周辺分布 $\nu\in\Prob(\mathcal{Y})$ を
  \[
    \nu(B):=\pi(\mathcal{X}\times B)\qquad(B\in\Borel(\mathcal{Y}))
  \]
  で定める．
  このとき，確率測度の族 $(\lambda_y)_{y\in\mathcal{Y}}$，$\lambda_y\in\Prob(\mathcal{X})$ で，
  次をみたすものが存在する．
  \begin{enumerate}
    \item 各 Borel 集合 $A\in\Borel(\mathcal{X})$ に対して $y\mapsto\lambda_y(A)$ は Borel 可測である．
    \item 任意の Borel 集合 $A\in\Borel(\mathcal{X})$，$B\in\Borel(\mathcal{Y})$ に対して
      \[
        \pi(A\times B)=\int_B\lambda_y(A)\,\d\nu(y).
      \]
  \end{enumerate}
  さらに，$(\lambda'_y)_{y\in\mathcal{Y}}$ も条件 1，2 をみたすならば，
  $\nu(N)=0$ をみたす $N\in\Borel(\mathcal{Y})$ が存在して，$y\notin N$ ならば $\lambda_y=\lambda'_y$ である．
\end{theorem}

条件 1，2 をみたす族 $(\lambda_y)$ を $\pi$ の $\nu$ に関する\textbf{分解}という．
一意性は，$A$ ごとに例外集合を取るだけでなく，
一つの零集合の外で\emph{測度として一致する}ことを主張している．
まず $\mathcal{X}=K$ が compact な場合を考える．連続関数 $f$ に対して
$\nu_f(B):=\int_{K\times B}f(x)\,\d\pi(x,y)$ とおくと，証明の流れは
\[
  \nu_f(B)=\int_B l_f(y)\,\d\nu(y)
  \quad\longrightarrow\quad
  m_y(f)=\int_K f\,\d\lambda_y
\]
となる．左の密度 $l_f$ を Radon--Nikodym の定理で作り，
共通の零集合の外で $f\mapsto l_f(y)$ を正の線形汎関数 $m_y$ にまとめる．
右の確率測度 $\lambda_y$ を与えるのが Riesz--Markov--Kakutani の表現定理である．
$\nu_f$ は $f\geq0$ のとき測度になるので，まずこの場合を扱い，一般の $f$ は正部分・負部分に分ける．

\section{Radon--Nikodym の定理}
\label{sec:dis-radon-nikodym}

Radon--Nikodym の定理は，一つの測度を別の測度に関する密度で表す定理である．
測度分解の証明では，この密度を使って各 $y$ における積分の値を定める．
有限測度の場合の主張は，北別府悠『最適輸送理論入門〜兼非標準的関数解析入門〜』
Corollary 4.6.2（p.~57）に対応する．原資料では，$L^2$ を用いて Lebesgue 分解定理
（Theorem 4.6.1，pp.~56--57）を示し，その系としてこの定理を得る．
本節では，同じ $L^2$ による構成を $\nu\ll\mu$ の場合に適用し，必要な密度を直接作る．

本節では，北別府 p.~57 の記号に合わせ，$(X,\Sigma)$ を可測空間とする．

\subsection{準備：Riesz--Fréchet の表現定理}

有界線形汎関数を Hilbert 空間の内積で表す定理を用いる．
実 Hilbert 空間・有界線形汎関数の定義は定義~\ref{def:rn-hilbert-space}による．

\begin{theorem}{Riesz--Fréchet の表現定理（実 Hilbert 空間）}{rn-hilbert-riesz}
  $\mathcal{H}$ を実 Hilbert 空間，$L\colon\mathcal{H}\to\R$ を有界線形汎関数とする．
  このとき，ただ一つの $h\in\mathcal{H}$ が存在して
  \[
    L(g)=(h,g)\qquad(g\in\mathcal{H})
  \]
  が成り立つ．
\end{theorem}

この定理は Hilbert 空間版 Riesz の表現定理ともよばれる．証明は付録に記す．

\subsection{絶対連続性と定理の主張}

\begin{definition}{測度の絶対連続性}{meas-absolute-continuity}
  $(X,\Sigma)$ 上の有限測度 $\mu,\nu$ を考える．
  $\nu$ が $\mu$ に\textbf{絶対連続}であるとは，
  \[
    \mu(A)=0\quad\Longrightarrow\quad\nu(A)=0
    \qquad(A\in\Sigma)
  \]
  が成り立つことをいう．このとき $\nu\ll\mu$ と書く．
  この定義は Folland (1999) §3.2 に従う．北別府 p.~57 の $\varepsilon$--$\delta$ による定義との
  同値性は，注意~\ref{rem:rn-null-characterization}で確認する．
\end{definition}

\begin{theorem}{Radon--Nikodym の定理（有限測度の場合；北別府 Cor 4.6.2）}{meas-radon-nikodym}
  $(X,\Sigma)$ を可測空間とし，測度 $\mu,\nu\colon\Sigma\to[0,\infty]$ が
  \[
    \mu(X)<\infty,\qquad\nu(X)<\infty,\qquad\nu\ll\mu
  \]
  をみたすとする．
  このとき，非負の $\mu$-可積分な可測関数 $f\colon X\to[0,\infty)$ が存在して，
  \[
    \nu(A)=\int_Af\,\d\mu\qquad(A\in\Sigma)
  \]
  が成り立つ．$f$ は $\mu$-a.e. の意味で一意である．この密度を
  \textbf{Radon--Nikodym 微分}とよび，
  \[
    f=\frac{\d\nu}{\d\mu}
  \]
  と書く．
\end{theorem}

証明の要点は，和の測度 $\pi:=\mu+\nu$ を基準にして，まず
\[
  \nu(A)=\int_Ah\,\d\pi,\qquad
  \mu(A)=\int_A(1-h)\,\d\pi
\]
という表示を作ることである．次に，$\nu\ll\mu$ により $\{h=1\}$ を零集合として除き，
$f=h/(1-h)$ を得る．補助関数 $h$ と，求める密度 $f=\d\nu/\d\mu$ は区別する．

\subsection{定理の証明と積分への応用}

ここで用いる $L^2(\pi)$ は，$\int_X\abs{g}^2\,\d\pi<\infty$ をみたす実数値可測関数を，
$\pi$-a.e. に等しいもの同士で同一視した空間である．
内積 $(g,h)_\pi=\int_Xgh\,\d\pi$ とノルム
$\norm{g}_{L^2(\pi)}=(\int_X\abs{g}^2\,\d\pi)^{1/2}$ により Hilbert 空間となる
（補題~\ref{lem:rn-l2-complete}）．

\begin{proof}[定理~\ref{thm:meas-radon-nikodym}の証明]
  \textbf{1. $\pi=\mu+\nu$ に関する表示を作る．}
  $\pi$ は有限測度である．$L^2(\pi)$ 上の線形汎関数を
  \[
    L(g):=\int_Xg\,\d\nu
  \]
  で定める．$\nu\leq\pi$ なので代表元によらず，Cauchy--Schwarz の不等式
  （定理~\ref{thm:meas-holder}の $r=r'=2$ の場合）より
  \[
    \abs{L(g)}\leq\int_X\abs{g}\,\d\nu
    \leq\nu(X)^{1/2}\left(\int_X\abs{g}^2\,\d\nu\right)^{1/2}
    \leq\nu(X)^{1/2}\norm{g}_{L^2(\pi)}.
  \]
  したがって $L$ は有界であり，Riesz--Fréchet の表現定理
  （定理~\ref{thm:rn-hilbert-riesz}）から，ある $h\in L^2(\pi)$ について
  \[
    \int_Xg\,\d\nu=(h,g)_\pi=\int_Xgh\,\d\pi\qquad(g\in L^2(\pi))
  \]
  と表される．$\pi$ が有限なので，任意の $A\in\Sigma$ について $\mathbf{1}_A\in L^2(\pi)$ であり，
  \[
    \nu(A)=\int_Ah\,\d\pi,\qquad
    \mu(A)=\pi(A)-\nu(A)=\int_A(1-h)\,\d\pi.
  \]
  前者に $A=\{h<0\}$，後者に $A=\{h>1\}$ を代入すると，左辺の非負性から
  $0\leq h\leq1$ が $\pi$-a.e. に成り立つ．
  この可測な例外集合上で $h=0$ と定め直し，以下では各点で $0\leq h\leq1$ とする．

  \textbf{2. $\mu$ に関する密度 $f$ を取り出す．}
  $B:=\{h=1\}$ とおくと，前段の表示から $\mu(B)=0$ であり，
  $\nu\ll\mu$ により $\nu(B)=0$，したがって $\pi(B)=0$ でもある．そこで
  \[
    f(x):=\begin{cases}
      \dfrac{h(x)}{1-h(x)},&h(x)<1,\\
      0,&h(x)=1
    \end{cases}
  \]
  と定める．これは非負の有限値可測関数である．
  ここで，集合上の表示 $\mu(A)=\int_A(1-h)\,\d\pi$ から，任意の非負可測関数 $u$ に対して
  \[
    \int_Xu\,\d\mu=\int_Xu(1-h)\,\d\pi
  \]
  が従う．実際，指示関数については表示そのものであり，線形性から非負単関数について成り立つ．
  非負単関数の列 $s_n\uparrow u$ を取り，両辺に単調収束定理を用いればよい．
  この議論は集合上の表示だけから従い，Radon--Nikodym の定理を前提としない．

  $f(1-h)=h$ は $X\setminus B$ 上で成り立ち，$\pi(B)=0$ である．
  したがって，上の書き換えを $u=\mathbf{1}_A f$ に適用すると
  \[
    \int_Af\,\d\mu=\int_Af(1-h)\,\d\pi=\int_Ah\,\d\pi=\nu(A)
    \qquad(A\in\Sigma)
  \]
  を得る．
  特に $\int_Xf\,\d\mu=\nu(X)<\infty$ であり，$f$ は $\mu$-可積分である．

  \textbf{3. 一意性．}
  一般に，$\mu$-可積分な可測関数 $u,v\colon X\to\R$ がすべての $A\in\Sigma$ で
  \[
    \int_Au\,\d\mu=\int_Av\,\d\mu
  \]
  をみたすとする．$A:=\{u>v\}$ では $u-v>0$ である一方，
  $\int_A(u-v)\,\d\mu=0$ なので $\mu(A)=0$ である．
  $u,v$ を入れ替えて $\mu(\{v>u\})=0$ も得るので，$u=v$ が $\mu$-a.e. に成り立つ．
  特に，密度 $f$ の一意性が従う．
\end{proof}

\begin{corollary}{密度による積分の書き換え}{rn-density-integral}
  定理~\ref{thm:meas-radon-nikodym}の密度を $f=\d\nu/\d\mu$ とする．
  任意の非負可測関数 $g\colon X\to[0,\infty]$ に対し
  \[
    \int_X g\,\d\nu=\int_X g\,f\,\d\mu
  \]
  が成り立つ．
\end{corollary}

\begin{proof}
  $g=\mathbf{1}_A$ のときは密度の定義そのものである．
  積分の線形性から非負単関数について成り立ち，
  非負単関数の列 $s_n\uparrow g$（注意~\ref{rem:meas-measurable-ops}）を取れば
  $s_nf\uparrow gf$ なので，両辺に単調収束定理（定理~\ref{thm:meas-mct}）を用いて一般の場合を得る．
\end{proof}

\section{Riesz--Markov--Kakutani の表現定理}
\label{sec:dis-riesz-representation}

連続関数に対する正の線形汎関数を，測度による積分として表す．
前節の Riesz--Fréchet の表現定理が Hilbert 空間の元を与えるのに対し，こちらは測度を構成する定理である．
主張は北別府 Theorem 5.2.5（p.~75）のうち，測度分解に必要な正の汎関数と確率測度の場合に対応する．
証明は発表ノートに沿い，離散的な確率測度を作り，既知とする Prokhorov の定理で極限を取る．
$C(K)$ と一様ノルムの記法は注意~\ref{rem:meas-compact-continuous}による．

\begin{theorem}{Riesz--Markov--Kakutani の表現定理（正の汎関数の場合；北別府 Thm 5.2.5）}{meas-riesz-representation}
  $K$ を空でない compact 距離空間とする．
  実線形写像 $L\colon C(K)\to\R$ が
  \[
    f\geq0\ \Longrightarrow\ L(f)\geq0,\qquad L(1)=1
  \]
  をみたすならば，ある唯一の $\rho\in\Prob(K)$ により
  \[
    L(f)=\int_Kf\,\d\rho\qquad(f\in C(K))
  \]
  と表せる．さらに，$\rho$ は正則である
  （正則性の定義と証明は補題~\ref{lem:meas-compact-regular}）．
  条件 $f\geq0\Rightarrow L(f)\geq0$ をみたす実線形写像を
  \textbf{正の線形汎関数}という．
\end{theorem}

正値性から $f\leq g$ ならば $L(f)\leq L(g)$ である．特に $L(1)=1$ より
\[
  a\leq f\leq b\quad\Longrightarrow\quad a\leq L(f)\leq b
  \qquad(a,b\in\R)
\]
となる．この評価から $L$ の一様収束に関する連続性も従うため，連続性を別に仮定する必要はない．

\begin{proof}[定理~\ref{thm:meas-riesz-representation}の証明]
  \textbf{1. 近似する確率測度を作る．}
  $n\in\N$ に対し $r_n:=1/(4n)$ とおく．compact 性により，有限個の点 $x_{n,1},\dots,x_{n,m_n}\in K$ で
  \[
    K=\bigcup_{i=1}^{m_n}B_{r_n}(x_{n,i})
  \]
  となるものを選べる．次のように連続関数を作り，和が $1$ になるように正規化する：
  \[
    \psi_{n,i}(x):=\max\{r_n-d(x,x_{n,i}),0\},\qquad
    \varphi_{n,i}(x):=\frac{\psi_{n,i}(x)}{\sum_{j=1}^{m_n}\psi_{n,j}(x)}.
  \]
  開球による被覆から分母は各点で正であり，$\varphi_{n,i}\in C(K)$，$\varphi_{n,i}\geq0$，
  $\sum_i\varphi_{n,i}=1$ である．また，$\varphi_{n,i}(x)>0$ ならば $d(x,x_{n,i})<r_n$ である．
  Dirac 測度（注意~\ref{rem:meas-dirac}）を用いて
  \[
    \rho_n:=\sum_{i=1}^{m_n}L(\varphi_{n,i})\,\delta_{x_{n,i}}
  \]
  とおく．各係数は非負で，その和は $L(\sum_i\varphi_{n,i})=L(1)=1$ なので，$\rho_n\in\Prob(K)$ である．

  \textbf{2. 積分が $L(f)$ に近づくことを示す．}
  $f\in C(K)$ を固定し，
  \[
    M_n:=\sup\{\abs{f(x)-f(z)}\mid x,z\in K,\ d(x,z)\leq r_n\}
  \]
  とおく．$f$ は一様連続なので $M_n\to0$ である．
  $\varphi_{n,i}(x)>0$ となる項だけを見れば，すべての $x\in K$ で
  \[
    \left|f(x)-\sum_{i=1}^{m_n}f(x_{n,i})\varphi_{n,i}(x)\right|
    \leq\sum_{i=1}^{m_n}M_n\varphi_{n,i}(x)=M_n.
  \]
  線形性と冒頭の評価より
  \[
    \left|L(f)-\int_Kf\,\d\rho_n\right|
    =\left|L\left(f-\sum_{i=1}^{m_n}f(x_{n,i})\varphi_{n,i}\right)\right|
    \leq M_n\longrightarrow0.
  \]

  \textbf{3. Prokhorov の定理で極限を取る．}
  compact 距離空間 $K$ は Polish 空間である（注意~\ref{rem:meas-compact-facts}）．
  さらに，$K$ 自身が compact で $\rho_n(K)=1$ なので，族 $\{\rho_n\mid n\in\N\}$ は tight である．
  Prokhorov の定理（定理~\ref{thm:meas-prokhorov}）より，ある部分列 $(\rho_{n_j})$ と
  $\rho\in\Prob(K)$ が存在して $\rho_{n_j}\weakto\rho$ となる．この部分列は $f$ に依存せずに選ぶ．
  任意の $f\in C(K)=C_b(K)$ に対して，前段の収束と弱収束の定義から
  \[
    L(f)=\lim_j\int_Kf\,\d\rho_{n_j}=\int_Kf\,\d\rho.
  \]
  また，補題~\ref{lem:meas-compact-regular}より $\rho$ は正則である．

  \textbf{4. 一意性．}
  $\rho'\in\Prob(K)$ も同じ表示をみたすなら，すべての $f\in C(K)=C_b(K)$ の積分が一致する．
  定理~\ref{thm:meas-cb-determines}より $\rho=\rho'$ である．
\end{proof}

\section{測度分解定理の証明}

まず第1の空間が compact な場合に分解を構成し，tightness によって一般の場合へ拡張した後，一意性を示す．

\subsection{第1の空間が compact な場合}

構成は二段階である．Radon--Nikodym の定理で連続関数の積分を $y$ の関数として表し，
各 $y$ で得られる正の線形汎関数に Riesz--Markov--Kakutani の表現定理を適用する．
一般の場合への拡張に備え，ここでは有限測度を扱う．

\begin{lemma}{compact な場合の分解}{dis-compact}
  $K$ を空でない compact 距離空間，$\mathcal{Y}$ を距離空間とし，
  $\pi$ を $K\times\mathcal{Y}$ 上の零でない有限 Borel 測度，
  $\nu(B):=\pi(K\times B)$ とする．
  このとき，$\lambda_y\in\Prob(K)$ の族で，各 $A\in\Borel(K)$ に対し $y\mapsto\lambda_y(A)$ が Borel 可測となり，
  \[
    \pi(A\times B)=\int_B\lambda_y(A)\,\d\nu(y)
    \qquad(A\in\Borel(K),\ B\in\Borel(\mathcal{Y}))
  \]
  をみたすものが存在する．
\end{lemma}

\begin{proof}
  \textbf{1. Radon--Nikodym 微分と共通の零集合．}
  非負の $g\in C(K)$ に対し
  \[
    \nu_g(B):=\int_{K\times B}g(x)\,\d\pi(x,y)
    \qquad(B\in\Borel(\mathcal{Y}))
  \]
  とおく．単調収束定理より $\nu_g$ は有限測度であり，
  $\nu_g(B)\leq\norm{g}_\infty\nu(B)$ である．
  したがって $\nu_g\ll\nu$ であり，Radon--Nikodym の定理（定理~\ref{thm:meas-radon-nikodym}）から，
  非負で $\nu$-可積分な Borel 可測関数 $l_g=\d\nu_g/\d\nu$ で
  $\nu_g(B)=\int_Bl_g\,\d\nu$ となるものを取る．
  この適用では，定理の基準測度 $\mu$ が第2周辺分布 $\nu$ に当たる．
  $\nu_0=0$，$\nu_1=\nu$ なので $l_0:=0$，$l_1:=1$ と選び，
  一般の $f\in C(K)$ には $l_f:=l_{f^+}-l_{f^-}$ と定める
  （$f^\pm$ は注意~\ref{rem:meas-compact-continuous}）．
  この選び方では $f\geq0$ ならば $l_f\geq0$ が各点で成り立ち，積分の線形性より
  \[
    \int_Bl_f\,\d\nu=\int_{K\times B}f(x)\,\d\pi(x,y)
  \]
  がすべての $f\in C(K)$ と $B\in\Borel(\mathcal{Y})$ で成り立つ．

  密度の等式は一般に $\nu$-a.e. にしか成り立たず，例外集合は関数に依存する．
  そこで，共通の零集合を取るため，まず関数を可算個に絞る．
  $1$ を含む可算な稠密 $\mathbb{Q}$-ベクトル部分空間 $\mathcal{C}\subset C(K)$ を取る
  （定理~\ref{thm:meas-ck-separable}）．
  $f,g\in\mathcal{C}$，$a,b\in\mathbb{Q}$ について，$l_{af+bg}$ と $al_f+bl_g$ は
  上の積分公式により任意の $B$ で同じ積分値をもつので，
  一意性の議論（定理~\ref{thm:meas-radon-nikodym}の証明の3）より $\nu$-a.e. に一致する．
  条件は可算個なので，例外集合の合併は Borel 零集合 $N$ となり，$y\notin N$ で
  \[
    l_{af+bg}(y)=a\,l_f(y)+b\,l_g(y)\qquad(f,g\in\mathcal{C},\ a,b\in\mathbb{Q})
  \]
  が同時に成り立つ．
  $y\notin N$，$f\in\mathcal{C}$ とし，有理数 $r>\norm{f}_\infty$ を取ると，
  $r\pm f\geq0$ かつ $r\pm f\in\mathcal{C}$ より $r\pm l_f(y)=l_{r\pm f}(y)\geq0$ である．
  $r$ を $\norm{f}_\infty$ に近づけて $\abs{l_f(y)}\leq\norm{f}_\infty$ を得る．

  $x_0\in K$ を固定し，各 $f\in\mathcal{C}$ について $N$ 上では $l_f(y):=f(x_0)$ と置き直す．
  Borel 可測性と積分公式は変わらず，$\mathbb{Q}$-線形性，正値性，$l_1=1$，
  $\abs{l_f}\leq\norm{f}_\infty$ がすべての $y$ で成り立つ．

  \textbf{2. 各 $y$ に対応する確率測度の構成．}
  $y\in\mathcal{Y}$ を固定する．$f\mapsto l_f(y)$ は $\mathcal{C}$ 上で $\mathbb{Q}$-線形であり，
  1 より $\abs{l_f(y)-l_g(y)}=\abs{l_{f-g}(y)}\leq\norm{f-g}_\infty$ をみたす．
  $f\in C(K)$ に対し，$\norm{f_k-f}_\infty\to0$ となる $f_k\in\mathcal{C}$ を取ると
  $(l_{f_k}(y))_k$ は Cauchy 列なので，
  \[
    m_y(f):=\lim_{k\to\infty}l_{f_k}(y)
  \]
  と定める．別の近似列 $g_k\to f$ に対しても
  \[
    \abs{l_{f_k}(y)-l_{g_k}(y)}\leq\norm{f_k-g_k}_\infty\to0
  \]
  なので，極限は近似列によらない．$f\in\mathcal{C}$ なら $m_y(f)=l_f(y)$ であり，
  極限を取ることで $\mathbb{Q}$-線形性と $\abs{m_y(f)-m_y(g)}\leq\norm{f-g}_\infty$ が保たれる．
  実数 $a$ に有理数列 $q_j$ を近づけると，この連続性より
  \[
    m_y(af)=\lim_jm_y(q_jf)=\lim_jq_jm_y(f)=am_y(f)
  \]
  となるので，$m_y$ は実線形である．また $m_y(1)=1$，$\abs{m_y(f)}\leq\norm{f}_\infty$ である．
  $f\geq0$ なら，$m:=\norm{f}_\infty$ とおくと $\norm{m-f}_\infty\leq m$ なので，
  \[
    m_y(f)=m-m_y(m-f)\geq0.
  \]
  したがって Riesz--Markov--Kakutani の表現定理（定理~\ref{thm:meas-riesz-representation}）により
  $m_y(f)=\int_Kf\,\d\lambda_y$（$f\in C(K)$）となる $\lambda_y\in\Prob(K)$ が得られる．

  \textbf{3. 連続関数に対する積分公式．}
  $f\in C(K)$ と上の近似列 $f_k\in\mathcal{C}$ を取る．
  すべての $y$ で $\int_Kf\,\d\lambda_y=\lim_kl_{f_k}(y)$ なので，
  $y\mapsto\int_Kf\,\d\lambda_y$ は Borel 可測である．
  さらに $\abs{\int_Kf\,\d\lambda_y-l_{f_k}(y)}\leq\norm{f-f_k}_\infty$ は $y$ によらず $0$ に近づく．
  $\nu,\pi$ は有限測度なので，この一様収束と $f_k\to f$ の一様収束から
  \[
    \int_B\left(\int_Kf\,\d\lambda_y\right)\d\nu(y)
    =\lim_k\int_Bl_{f_k}\,\d\nu
    =\lim_k\int_{K\times B}f_k(x)\,\d\pi(x,y)
    =\int_{K\times B}f(x)\,\d\pi(x,y).
  \]

  \textbf{4. Borel 集合への拡張．}
  空でない閉集合 $A\subset K$ に対し，補題~\ref{lem:meas-closed-approx}の
  $\varphi_k\downarrow\mathbf{1}_A$ を 3 の $f$ として有界収束定理を用いると，
  $y\mapsto\lambda_y(A)$ は可測で，主張の積分公式が成り立つ（$A=\emptyset$ では明らか）．
  可測性とすべての $B$ に対する積分公式をみたす $A$ の全体は $\lambda$-系である．
  実際，$K$ を含み，補集合については $\lambda_y(K\setminus A)=1-\lambda_y(A)$ を，
  互いに素な可算合併については可算加法性と単調収束定理を使えばよい．
  閉集合の全体は $\Borel(K)$ を生成する $\pi$-系なので，
  Dynkin の定理（定理~\ref{thm:meas-pi-lambda}）によりすべての $A\in\Borel(K)$ で成り立つ．
\end{proof}

\subsection{一般の Polish 空間の場合}

\begin{proof}[定理~\ref{thm:meas-disintegration}の存在の証明]
  $x_0\in\mathcal{X}$ を固定する．

  \textbf{1. compact 集合の列による分割．}
  第1周辺分布を $\mu(A):=\pi(A\times\mathcal{Y})$ とおく．
  $\{\mu\}$ 内の列は定数列で $\mu$ に弱収束するので，既出の Prokhorov の定理
  （定理~\ref{thm:meas-prokhorov}，$1\Rightarrow2$）より $\{\mu\}$ は tight である．
  $K_0:=\emptyset$ とし，tightness で得た compact 集合と $K_{n-1}$ の合併を取って，
  $K_{n-1}\subset K_n$，$\mu(\mathcal{X}\setminus K_n)\leq1/n$ となる compact 集合の列を作る．
  $K:=\bigcup_nK_n$，$E_n:=K_n\setminus K_{n-1}$ とおけば，$\mu(\mathcal{X}\setminus K)=0$ であり，
  $E_n$ は $K$ を互いに素に分割する．したがって，$\nu_n(B):=\pi(E_n\times B)$ とおくと
  \[
    \pi(A\times B)=\sum_n\pi((A\cap E_n)\times B),\qquad
    \nu(B)=\sum_n\nu_n(B).
  \]

  \textbf{2. 各部分での分解．}
  $\nu_n\neq0$ の各 $n$ では，$\pi$ の $E_n\times\mathcal{Y}$ への制限
  $C\mapsto\pi(C\cap(E_n\times\mathcal{Y}))$ を $K_n\times\mathcal{Y}$ 上の有限測度とみて，
  補題~\ref{lem:dis-compact}を適用する．得た確率測度を $K_n$ の外で $0$ として
  $\mathcal{X}$ 上に延長し，$\rho_y^{(n)}$ と書く
  （注意~\ref{rem:meas-subspace-borel}）．このとき
  $y\mapsto\rho^{(n)}_y(A)$ は Borel 可測で，
  \[
    \pi((A\cap E_n)\times B)=\int_B\rho_y^{(n)}(A)\,\d\nu_n(y)
    \qquad(A\in\Borel(\mathcal{X}),\ B\in\Borel(\mathcal{Y}))
  \]
  である．$\nu_n=0$ の $n$ では $\rho_y^{(n)}:=\delta_{x_0}$ とすれば，上式は両辺 $0$ で成り立つ．

  \textbf{3. 重みを付けて足し合わせる．}
  $\nu_n(B)\leq\nu(B)$ なので，Radon--Nikodym の定理（定理~\ref{thm:meas-radon-nikodym}）から
  非負の Borel 可測な密度 $h_n:=\d\nu_n/\d\nu$ を取る．
  単調収束定理より $\int_B\sum_nh_n\,\d\nu=\sum_n\nu_n(B)=\nu(B)$ である．
  特に $\sum_nh_n$ は可積分である．零集合上の無限値を $0$ に置き換え，
  一意性の議論（定理~\ref{thm:meas-radon-nikodym}の証明の3）を適用すると，$\sum_nh_n=1$ が $\nu$-a.e. に成り立つ．
  Borel 零集合 $N_0:=\{\sum_nh_n\neq1\}$ を用いて
  \[
    \lambda_y(A):=
    \begin{cases}
      \sum_nh_n(y)\,\rho_y^{(n)}(A) & (y\notin N_0),\\
      \delta_{x_0}(A) & (y\in N_0)
    \end{cases}
  \]
  と定める．$y\notin N_0$ では，確率測度を和が $1$ の非負の重みで足し合わせているので，
  $\lambda_y$ は確率測度である（可算加法性は非負二重級数の和の交換による）．
  各項と $N_0$ の可測性から条件 1 が成り立つ．
  系~\ref{cor:rn-density-integral}と単調収束定理より
  \[
    \begin{aligned}
      \int_B\lambda_y(A)\,\d\nu(y)
      &=\sum_n\int_B\rho_y^{(n)}(A)\,h_n(y)\,\d\nu(y)
      =\sum_n\int_B\rho_y^{(n)}(A)\,\d\nu_n(y)\\
      &=\sum_n\pi((A\cap E_n)\times B)=\pi(A\times B).
    \end{aligned}
  \]
  よって条件 2 も成り立つ．
\end{proof}

\subsection{一意性}

\begin{proof}[定理~\ref{thm:meas-disintegration}の一意性の証明]
  $(\lambda_y)$，$(\lambda'_y)$ がともに条件 1，2 をみたすとする．
  $A\in\Borel(\mathcal{X})$ を固定すると，$\lambda_y(A)$，$\lambda'_y(A)$ は $y$ の有界可測関数であり，
  条件 2 よりすべての $B$ で同じ積分値をもつ．
  一意性の議論（定理~\ref{thm:meas-radon-nikodym}の証明の3）より，
  $N_A:=\{y\mid\lambda_y(A)\neq\lambda'_y(A)\}$ は Borel 零集合である．

  補題~\ref{lem:meas-countable-generator}により，
  $\Borel(\mathcal{X})$ を生成する可算な $\pi$-系 $\mathcal{E}$ を取り，
  $N:=\bigcup_{A\in\mathcal{E}}N_A$ とおく．$N$ は $\nu$-零集合である．
  $y\notin N$ では $\lambda_y,\lambda'_y$ は $\mathcal{E}$ 上で一致する確率測度なので，
  一致定理（定理~\ref{thm:meas-uniqueness}）より $\lambda_y=\lambda'_y$ である．
\end{proof}
