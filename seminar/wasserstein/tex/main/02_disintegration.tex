\chapter{測度分解定理}
\label{ch:disintegration}

接着補題（補題~\ref{lem:wp-gluing}）で必要なのは，
二変数の分布 $\pi$ を，変数 $y$ ごとの確率測度 $\lambda_y$ に分け，
その平均として元の分布を復元することである．
北別府悠『最適輸送理論入門〜兼非標準的関数解析入門〜』Theorem 5.3.1 の
構成をもとに，接着補題で使う積空間上の形を示す．
原資料は一般の Borel 可測写像に沿う分解を扱うが，本章では第2変数に沿う分解に限定する．
記号と証明の流れは発表ノート（2026年11月25日版）に合わせ，
分解測度を $\lambda_y$，その構成に用いる汎関数を $m_y$ と書く（原資料と同じ記号）．
本章では，まず測度分解定理を述べ，その準備となる二つの定理を証明してから，測度分解の証明に進む．
Radon--Nikodym の定理は有限測度，Riesz--Markov--Kakutani の表現定理は compact 距離空間の場合に限る．
測度分解の証明に用いるのは，この二つの形である．
積分の基本性質・単調収束定理・Dynkin の定理と距離空間の基本事項は既知とする．
Radon--Nikodym の準備として Riesz--Fréchet の表現定理を述べ，その証明は付録に置く．
Riesz--Markov--Kakutani の証明と一般の Polish 空間への拡張では，
第\ref{ch:wasserstein-metrics}章で用いた Prokhorov の定理を再利用する．

発表では，二つの定理の証明から測度分解の構成・一意性へ進む流れを軸とする．
$L^2$ の完備性と Riesz--Fréchet の表現定理，
$C(K)$ の可分性や Borel 集合に関する補助結果には付録で証明を付け，
本文の参照先から必要に応じて確認できるようにした．

\section{測度分解定理の主張}

\begin{theorem}{測度分解定理（接着補題で用いる形）}{meas-disintegration}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間，$\pi\in\Prob(\mathcal{X}\times\mathcal{Y})$ とし，
  第2周辺分布 $\nu\in\Prob(\mathcal{Y})$ を
  \[
    \nu(B):=\pi(\mathcal{X}\times B)\qquad(B\in\Borel(\mathcal{Y}))
  \]
  で定める．
  このとき，確率測度の族 $(\lambda_y)_{y\in\mathcal{Y}}$，$\lambda_y\in\Prob(\mathcal{X})$ で，
  次をみたすものが存在する．
  \begin{enumerate}
    \item 各 Borel 集合 $A\in\Borel(\mathcal{X})$ に対して $y\mapsto\lambda_y(A)$ は Borel 可測である．
    \item 任意の Borel 集合 $A\in\Borel(\mathcal{X})$，$B\in\Borel(\mathcal{Y})$ に対して
      \[
        \pi(A\times B)=\int_B\lambda_y(A)\,\d\nu(y).
      \]
  \end{enumerate}
  さらに，この族は $\nu$-a.e. の意味で一意である．すなわち，別の族 $(\lambda'_y)$ も
  条件 1，2 をみたすならば，ある Borel 集合 $N$ が存在して，
  $\nu(N)=0$ かつすべての $y\notin N$ で $\lambda_y=\lambda'_y$ となる．
\end{theorem}

条件 1，2 をみたす族 $(\lambda_y)$ を $\pi$ の $\nu$ に関する\textbf{分解}という．
一意性では，一つの $\nu$-零集合の外で測度として一致することを要求する．
まず $\mathcal{X}$ が compact な場合を考える．連続関数 $f$ に対して
$\nu_f(B):=\int_{\mathcal{X}\times B}f(x)\,\d\pi(x,y)$ とおくと，証明の流れは
\[
  \nu_f(B)=\int_B l_f(y)\,\d\nu(y)
  \quad\longrightarrow\quad
  m_y(f)=\int_{\mathcal{X}} f\,\d\lambda_y
\]
となる．左の密度 $l_f$ を Radon--Nikodym の定理で作り，
可算稠密な関数族上で共通の零集合を除き，$f\mapsto l_f(y)$ を
$C(\mathcal{X})$ 上の正の線形汎関数 $m_y$ に延長する．
右の確率測度 $\lambda_y$ を与えるのが Riesz--Markov--Kakutani の表現定理である．
本原稿の Radon--Nikodym の定理は非負の測度についての定理なので，
$\nu_f$ には $f$ の正部分・負部分に分けて適用する．

\section{Radon--Nikodym の定理}
\label{sec:dis-radon-nikodym}

Radon--Nikodym の定理は，一つの測度を別の測度に関する密度で表す定理である．
測度分解の証明では，この密度を使って各 $y$ における積分の値を定める．
有限測度の場合の主張は，北別府悠『最適輸送理論入門〜兼非標準的関数解析入門〜』
Corollary 4.6.2（p.~57）に対応する．原資料では，$L^2$ を用いて Lebesgue 分解定理
（Theorem 4.6.1，pp.~56--57）を示し，その系としてこの定理を得る．
本節では，同じ $L^2$ による構成を $\nu\ll\mu$ の場合に適用し，必要な密度を直接作る．

本節では，北別府 p.~57 の記号に合わせ，$(X,\Sigma)$ を可測空間とする．

\subsection{準備：Riesz--Fréchet の表現定理}

有界線形汎関数を Hilbert 空間の内積で表す定理を用いる．
実 Hilbert 空間・有界線形汎関数の定義は定義~\ref{def:rn-hilbert-space}による．

\begin{theorem}{Riesz--Fréchet の表現定理（実 Hilbert 空間）}{rn-hilbert-riesz}
  $\mathcal{H}$ を実 Hilbert 空間，$L\colon\mathcal{H}\to\R$ を有界線形汎関数とする．
  このとき，ただ一つの $h\in\mathcal{H}$ が存在して
  \[
    L(g)=(h,g)\qquad(g\in\mathcal{H})
  \]
  が成り立つ．
\end{theorem}

この定理は Hilbert 空間版 Riesz の表現定理ともよばれる．証明は付録に記す．

\subsection{絶対連続性と定理の主張}

\begin{definition}{測度の絶対連続性}{meas-absolute-continuity}
  $(X,\Sigma)$ 上の有限測度 $\mu,\nu$ を考える．
  $\nu$ が $\mu$ に\textbf{絶対連続}であるとは，
  \[
    \mu(A)=0\quad\Longrightarrow\quad\nu(A)=0
    \qquad(A\in\Sigma)
  \]
  が成り立つことをいう．このとき $\nu\ll\mu$ と書く．
  この定義は Folland (1999) §3.2 に従う．北別府 p.~57 の $\varepsilon$--$\delta$ による定義との
  同値性は，注意~\ref{rem:rn-null-characterization}で確認する．
\end{definition}

\begin{theorem}{Radon--Nikodym の定理（有限測度の場合；北別府 Cor 4.6.2）}{meas-radon-nikodym}
  $(X,\Sigma)$ を可測空間とし，測度 $\mu,\nu\colon\Sigma\to[0,\infty]$ が
  \[
    \mu(X)<\infty,\qquad\nu(X)<\infty,\qquad\nu\ll\mu
  \]
  をみたすとする．
  このとき，非負の $\mu$-可積分な可測関数 $f\colon X\to[0,\infty)$ が存在して，
  \[
    \nu(A)=\int_Af\,\d\mu\qquad(A\in\Sigma)
  \]
  が成り立つ．$f$ は $\mu$-a.e. の意味で一意である．この密度を
  \textbf{Radon--Nikodym 微分}とよび，
  \[
    f=\frac{\d\nu}{\d\mu}
  \]
  と書く．
\end{theorem}

証明の要点は，和の測度 $\pi:=\mu+\nu$ を基準にして，まず
\[
  \nu(A)=\int_Ah\,\d\pi,\qquad
  \mu(A)=\int_A(1-h)\,\d\pi
\]
という表示を作ることである．次に，$\nu\ll\mu$ により $\{h=1\}$ を零集合として除き，
$f=h/(1-h)$ を得る．補助関数 $h$ と，求める密度 $f=\d\nu/\d\mu$ は区別する．

\subsection{定理の証明と積分への応用}

ここで用いる $L^2(\pi)$ は，$\int_X\abs{g}^2\,\d\pi<\infty$ をみたす実数値可測関数を，
$\pi$-a.e. に等しいもの同士で同一視した空間である．
内積 $(g,h)_\pi=\int_Xgh\,\d\pi$ とノルム
$\norm{g}_{L^2(\pi)}=(\int_X\abs{g}^2\,\d\pi)^{1/2}$ により Hilbert 空間となる
（補題~\ref{lem:rn-l2-complete}）．

証明の第2段では，集合上の積分表示を一般の非負関数の積分に書き換える次の補題を使う．

\begin{lemma}{密度による積分公式}{rn-integral-formula}
  $(X,\Sigma)$ 上の測度 $\mu,\pi$ と非負可測関数 $w\colon X\to[0,\infty]$ が
  \[
    \mu(A)=\int_Aw\,\d\pi\qquad(A\in\Sigma)
  \]
  をみたすとする．このとき，任意の非負可測関数 $u\colon X\to[0,\infty]$ に対して
  \[
    \int_Xu\,\d\mu=\int_Xuw\,\d\pi
  \]
  が成り立つ．
\end{lemma}

\begin{proof}
  $u=\mathbf{1}_A$ のときは仮定そのものである．
  積分の線形性から非負単関数について成り立つ．
  一般の $u$ に対しては非負単関数の列 $s_n\uparrow u$（注意~\ref{rem:meas-measurable-ops}）を取る．
  $s_nw\uparrow uw$ なので，両辺に単調収束定理（定理~\ref{thm:meas-mct}）を用いればよい．
  この証明は Radon--Nikodym の定理を用いない．
\end{proof}

\begin{proof}[定理~\ref{thm:meas-radon-nikodym}の証明]
  \textbf{1. $\pi=\mu+\nu$ に関する表示を作る．}
  $\pi$ は有限測度である．$L^2(\pi)$ 上の線形汎関数を
  \[
    L(g):=\int_Xg\,\d\nu
  \]
  で定める．$\nu\leq\pi$ なので代表元によらず，Cauchy--Schwarz の不等式
  （定理~\ref{thm:meas-holder}の $r=r'=2$ の場合）より
  \[
    \abs{L(g)}\leq\int_X\abs{g}\,\d\nu
    \leq\nu(X)^{1/2}\left(\int_X\abs{g}^2\,\d\nu\right)^{1/2}
    \leq\nu(X)^{1/2}\norm{g}_{L^2(\pi)}.
  \]
  したがって $L$ は有界であり，Riesz--Fréchet の表現定理
  （定理~\ref{thm:rn-hilbert-riesz}）から，ある $h\in L^2(\pi)$ について
  \[
    \int_Xg\,\d\nu=(h,g)_\pi=\int_Xgh\,\d\pi\qquad(g\in L^2(\pi))
  \]
  と表される．$\pi$ が有限なので，任意の $A\in\Sigma$ について $\mathbf{1}_A\in L^2(\pi)$ であり，
  \[
    \nu(A)=\int_Ah\,\d\pi,\qquad
    \mu(A)=\pi(A)-\nu(A)=\int_A(1-h)\,\d\pi.
  \]
  前者に $A=\{h<0\}$，後者に $A=\{h>1\}$ を代入すると，左辺の非負性から
  $0\leq h\leq1$ が $\pi$-a.e. に成り立つ．
  この可測な例外集合上で $h=0$ と定め直し，以下では各点で $0\leq h\leq1$ とする．

  \textbf{2. $\mu$ に関する密度 $f$ を取り出す．}
  $B:=\{h=1\}$ とおくと，前段の表示から $\mu(B)=0$ であり，
  $\nu\ll\mu$ により $\nu(B)=0$，したがって $\pi(B)=0$ でもある．そこで
  \[
    f(x):=\begin{cases}
      \dfrac{h(x)}{1-h(x)},&h(x)<1,\\
      0,&h(x)=1
    \end{cases}
  \]
  と定める．これは非負の有限値可測関数である．
  前段の表示 $\mu(A)=\int_A(1-h)\,\d\pi$ と，各点で $1-h\geq0$ であることから，
  補題~\ref{lem:rn-integral-formula}を $w=1-h$ として使える．
  $f(1-h)=h$ は $X\setminus B$ 上で成り立ち，$\pi(B)=0$ である．
  したがって，補題を $u=\mathbf{1}_A f$ に適用すると
  \[
    \int_Af\,\d\mu=\int_Af(1-h)\,\d\pi=\int_Ah\,\d\pi=\nu(A)
    \qquad(A\in\Sigma)
  \]
  を得る．
  特に $\int_Xf\,\d\mu=\nu(X)<\infty$ であり，$f$ は $\mu$-可積分である．

  \textbf{3. 一意性．}
  一般に，$\mu$-可積分な可測関数 $u,v\colon X\to\R$ がすべての $A\in\Sigma$ で
  \[
    \int_Au\,\d\mu=\int_Av\,\d\mu
  \]
  をみたすとする．$A:=\{u>v\}$ では $u-v>0$ である一方，
  $\int_A(u-v)\,\d\mu=0$ なので $\mu(A)=0$ である．
  $u,v$ を入れ替えて $\mu(\{v>u\})=0$ も得るので，$u=v$ が $\mu$-a.e. に成り立つ．
  特に，密度 $f$ の一意性が従う．
\end{proof}

\begin{corollary}{密度による積分の書き換え}{rn-density-integral}
  定理~\ref{thm:meas-radon-nikodym}の密度を $f=\d\nu/\d\mu$ とする．
  任意の非負可測関数 $g\colon X\to[0,\infty]$ に対し
  \[
    \int_X g\,\d\nu=\int_X g\,f\,\d\mu
  \]
  が成り立つ．
\end{corollary}

\begin{proof}
  $\nu(A)=\int_Af\,\d\mu$ $(A\in\Sigma)$ なので，
  補題~\ref{lem:rn-integral-formula}を $\mu,\pi,w$ をそれぞれ $\nu,\mu,f$ として適用すればよい．
\end{proof}

\section{Riesz--Markov--Kakutani の表現定理}
\label{sec:dis-riesz-representation}

連続関数に対する正の線形汎関数を，測度による積分として表す．
前節の Riesz--Fréchet の表現定理が Hilbert 空間の元を与えるのに対し，こちらは測度を構成する定理である．
主張は北別府 Theorem 5.2.5（p.~75）のうち，測度分解に必要な正の汎関数と確率測度の場合に対応する．
証明は発表ノートに沿い，離散的な確率測度を作り，既知とする Prokhorov の定理で極限を取る．
$C(K)$ と一様ノルムの記法は注意~\ref{rem:meas-compact-continuous}による．

\begin{theorem}{Riesz--Markov--Kakutani の表現定理（正の汎関数の場合；北別府 Thm 5.2.5）}{meas-riesz-representation}
  $K$ を空でない compact 距離空間とする．
  実線形写像 $L\colon C(K)\to\R$ が
  \[
    f\geq0\ \Longrightarrow\ L(f)\geq0,\qquad L(1)=1
  \]
  をみたすならば，ある唯一の $\mu\in\Prob(K)$ により
  \[
    L(f)=\int_Kf\,\d\mu\qquad(f\in C(K))
  \]
  と表せる．
  条件 $f\geq0\Rightarrow L(f)\geq0$ をみたす実線形写像を
  \textbf{正の線形汎関数}という．
\end{theorem}

正値性から $f\leq g$ ならば $L(f)\leq L(g)$ である．特に $L(1)=1$ より
\[
  a\leq f\leq b\quad\Longrightarrow\quad a\leq L(f)\leq b
  \qquad(a,b\in\R)
\]
となる．この評価から $L$ の一様収束に関する連続性も従うため，連続性を別に仮定する必要はない．

\begin{proof}[定理~\ref{thm:meas-riesz-representation}の証明]
  \textbf{1. 近似する確率測度を作る．}
  $n\in\N$ に対し $r_n:=1/(4n)$ とおく．compact 性により，有限個の点 $x_{n,1},\dots,x_{n,m_n}\in K$ で
  \[
    K=\bigcup_{i=1}^{m_n}B_{r_n}(x_{n,i})
  \]
  となるものを選べる．次のように連続関数を作り，和が $1$ になるように正規化する：
  \[
    \psi_{n,i}(x):=\max\{r_n-d(x,x_{n,i}),0\},\qquad
    \varphi_{n,i}(x):=\frac{\psi_{n,i}(x)}{\sum_{j=1}^{m_n}\psi_{n,j}(x)}.
  \]
  開球による被覆から分母は各点で正であり，$\varphi_{n,i}\in C(K)$，$\varphi_{n,i}\geq0$，
  $\sum_i\varphi_{n,i}=1$ である．また，$\varphi_{n,i}(x)>0$ ならば $d(x,x_{n,i})<r_n$ である．
  Dirac 測度（注意~\ref{rem:meas-dirac}）を用いて
  \[
    \mu_n:=\sum_{i=1}^{m_n}L(\varphi_{n,i})\,\delta_{x_{n,i}}
  \]
  とおく．各係数は非負で，その和は $L(\sum_i\varphi_{n,i})=L(1)=1$ なので，$\mu_n\in\Prob(K)$ である．

  \textbf{2. 積分が $L(f)$ に近づくことを示す．}
  $f\in C(K)$ を固定し，
  \[
    M_n:=\sup\{\abs{f(x)-f(z)}\mid x,z\in K,\ d(x,z)\leq r_n\}
  \]
  とおく．$f$ は一様連続なので $M_n\to0$ である．
  $\varphi_{n,i}(x)>0$ となる項だけを見れば，すべての $x\in K$ で
  \[
    \left|f(x)-\sum_{i=1}^{m_n}f(x_{n,i})\varphi_{n,i}(x)\right|
    \leq\sum_{i=1}^{m_n}M_n\varphi_{n,i}(x)=M_n.
  \]
  線形性と冒頭の評価より
  \[
    \left|L(f)-\int_Kf\,\d\mu_n\right|
    =\left|L\left(f-\sum_{i=1}^{m_n}f(x_{n,i})\varphi_{n,i}\right)\right|
    \leq M_n\longrightarrow0.
  \]

  \textbf{3. Prokhorov の定理で極限を取る．}
  compact 距離空間 $K$ は Polish 空間である（注意~\ref{rem:meas-compact-facts}）．
  さらに，$K$ 自身が compact で $\mu_n(K)=1$ なので，族 $\{\mu_n\mid n\in\N\}$ は tight である．
  Prokhorov の定理（定理~\ref{thm:meas-prokhorov}）より，ある部分列 $(\mu_{n_j})$ と
  $\mu\in\Prob(K)$ が存在して $\mu_{n_j}\weakto\mu$ となる．この部分列は $f$ に依存せずに選ぶ．
  任意の $f\in C(K)=C_b(K)$ に対して，前段の収束と弱収束の定義から
  \[
    L(f)=\lim_j\int_Kf\,\d\mu_{n_j}=\int_Kf\,\d\mu.
  \]

  \textbf{4. 一意性．}
  $\mu'\in\Prob(K)$ も同じ表示をみたすなら，すべての $f\in C(K)=C_b(K)$ の積分が一致する．
  定理~\ref{thm:meas-cb-determines}より $\mu=\mu'$ である．
\end{proof}

北別府 Theorem 5.2.5 は表現する測度の正則性も主張する．
本原稿の $\mu$ も正則である（補題~\ref{lem:meas-compact-regular}）．
この性質は compact 距離空間上の有限 Borel 測度に一般に成り立つもので，
上の構成にも測度分解の証明にも用いない．

\section{測度分解定理の証明}

まず第1の空間が compact な場合に分解を構成し，tightness によって一般の場合へ拡張した後，一意性を示す．

\subsection{第1の空間が compact な場合}

構成は二段階である．Radon--Nikodym の定理で連続関数の積分を $y$ の関数として表し，
各 $y$ で得られる正の線形汎関数に Riesz--Markov--Kakutani の表現定理を適用する．
一般の場合への拡張に備え，ここでは有限測度を扱う．

$C(K)$，$\norm{f}_\infty$，$f^\pm$ の記法は注意~\ref{rem:meas-compact-continuous}による．

\begin{lemma}{compact な場合の分解}{dis-compact}
  $K$ を空でない compact 距離空間，$\mathcal{Y}$ を距離空間とし，
  $\pi$ を $K\times\mathcal{Y}$ 上の零でない有限 Borel 測度，
  $\nu(B):=\pi(K\times B)$ とする．
  このとき，$\lambda_y\in\Prob(K)$ の族で，各 $A\in\Borel(K)$ に対し $y\mapsto\lambda_y(A)$ が Borel 可測となり，
  \[
    \pi(A\times B)=\int_B\lambda_y(A)\,\d\nu(y)
    \qquad(A\in\Borel(K),\ B\in\Borel(\mathcal{Y}))
  \]
  をみたすものが存在する．
\end{lemma}

\begin{proof}
  \textbf{1. Radon--Nikodym の定理による $l_f$ の構成．}
  $f\in C(K)$ に対し
  \[
    \nu_f(B):=\int_{K\times B}f(x)\,\d\pi(x,y)
    \qquad(B\in\Borel(\mathcal{Y}))
  \]
  とおく．$f$ は有界で $\pi$ は有限測度なので，$\nu_f(B)$ は有限値である．
  $-\norm{f}_\infty\leq f\leq\norm{f}_\infty$ と $\pi(K\times B)=\nu(B)$ より
  \[
    -\norm{f}_\infty\,\nu(B)\leq\nu_f(B)\leq\norm{f}_\infty\,\nu(B)
    \qquad(B\in\Borel(\mathcal{Y})).\tag{1}
  \]
  $f\geq0$ ならば，単調収束定理より $\nu_f$ は有限測度であり，
  (1) から $\nu(B)=0$ ならば $\nu_f(B)=0$，すなわち $\nu_f\ll\nu$ である．
  Radon--Nikodym の定理（定理~\ref{thm:meas-radon-nikodym}）を，
  基準測度 $\mu$ を $\nu$ として適用し，非負で $\nu$-可積分な
  有限値 Borel 可測関数 $l_f=\d\nu_f/\d\nu$ を一つずつ選ぶ．
  特に零関数に対しては $l_0=0$ を選ぶ．
  一般の $f\in C(K)$ では $\nu_f=\nu_{f^+}-\nu_{f^-}$ なので，$l_f:=l_{f^+}-l_{f^-}$ とおけば
  \[
    \nu_f(B)=\int_Bl_f(y)\,\d\nu(y)\qquad(B\in\Borel(\mathcal{Y}))\tag{2}
  \]
  が成り立つ．
  (2) をみたす $\nu$-可積分関数は $\nu$-a.e. の意味で一意である
  （定理~\ref{thm:meas-radon-nikodym}の証明の3）．

  \textbf{2. $f\mapsto l_f$ の性質．}
  $f,g\in C(K)$，$a,b\in\R$ に対し，次が $\nu$-a.e. に成り立つ．
  \begin{enumerate}
    \item[(a)] 線形性：$l_{af+bg}=a\,l_f+b\,l_g$．
    \item[(b)] 定数関数の値：$l_1=1$．
    \item[(c)] 有界性：$\abs{l_f}\leq\norm{f}_\infty$．
    \item[(d)] 正値性：$f\geq0$ ならば $l_f\geq0$．
  \end{enumerate}
  (a) 積分の線形性から $\nu_{af+bg}=a\nu_f+b\nu_g$ である．
  したがって $l_{af+bg}$ と $al_f+bl_g$ はすべての $B$ で同じ積分値をもち，
  1 の一意性から $\nu$-a.e. に一致する．
  (b) $\nu_1(B)=\pi(K\times B)=\nu(B)=\int_B1\,\d\nu$ なので，同じく一意性による．
  (c) $n\in\N$ に対し $A_n:=\{y\mid l_f(y)\geq\norm{f}_\infty+1/n\}$ とおく．
  $\nu(A_n)>0$ と仮定すると，(2) より
  \[
    \nu_f(A_n)=\int_{A_n}l_f\,\d\nu\geq\Bigl(\norm{f}_\infty+\frac1n\Bigr)\nu(A_n)>\norm{f}_\infty\,\nu(A_n)
  \]
  となり，(1) の右側に矛盾する．よって $\nu(A_n)=0$ であり，
  $\{l_f>\norm{f}_\infty\}=\bigcup_nA_n$ は可算劣加法性から $\nu$-零集合である．
  $A'_n:=\{y\mid l_f(y)\leq-\norm{f}_\infty-1/n\}$ について同様に (1) の左側を用いると，
  $\{l_f<-\norm{f}_\infty\}$ も $\nu$-零集合である．
  (d) $f\geq0$ ならばすべての $B$ で $\nu_f(B)\geq0$ である．
  $A_n:=\{y\mid l_f(y)\leq-1/n\}$ について $\nu(A_n)>0$ と仮定すると
  $\nu_f(A_n)=\int_{A_n}l_f\,\d\nu\leq-\nu(A_n)/n<0$ となり矛盾する．
  よって $\{l_f<0\}=\bigcup_nA_n$ は $\nu$-零集合である．

  \textbf{3. 可算稠密集合と共通の零集合．}
  (a)--(d) の例外集合は関数に依存するので，関数を可算個に絞る．
  定理~\ref{thm:meas-ck-separable}により，$C(K)$ の可算稠密集合 $S=\{f_i\mid i\in\N\}$ を取る．
  $\mathbb{Q}$ を有理数全体とし，
  \[
    D:=\Bigl\{q_0\,1+\sum_{i=1}^nq_if_i\Bigm| n\in\N,\ q_0,\dots,q_n\in\mathbb{Q}\Bigr\}
  \]
  とおく．各 $n$ で係数の選び方は可算個なので $D$ は可算であり，
  $S\subset D\subset C(K)$ より $D$ は稠密である．
  また，$D$ は定数関数 $0,1$ を含み，有理係数の線形結合について閉じている．

  $f,g\in D$，$a,b\in\mathbb{Q}$ に対して，例外集合を
  \[
    \begin{aligned}
      N^{(a)}_{f,g,a,b}&:=\{y\mid l_{af+bg}(y)\neq a\,l_f(y)+b\,l_g(y)\},\\
      N^{(b)}&:=\{y\mid l_1(y)\neq1\},\\
      N^{(c)}_f&:=\{y\mid\abs{l_f(y)}>\norm{f}_\infty\},\\
      N^{(d)}_f&:=\{y\mid l_f(y)<0\}\qquad(f\geq0)
    \end{aligned}
  \]
  と定める．これらは $l_f$ の Borel 可測性から Borel 集合であり，2 より $\nu$-零集合である．
  \[
    N:=\bigcup_{\substack{f,g\in D\\a,b\in\mathbb{Q}}}N^{(a)}_{f,g,a,b}
      \cup N^{(b)}\cup\bigcup_{f\in D}N^{(c)}_f
      \cup\bigcup_{\substack{f\in D\\f\geq0}}N^{(d)}_f
  \]
  は可算個の Borel 零集合の合併なので，$N\in\Borel(\mathcal{Y})$，$\nu(N)=0$ である．
  したがって，$y\notin N$ では (a)--(d) が $D$ 上で同時に成り立つ．

  \textbf{4. 線形汎関数 $m_y$ の構成と $C(K)$ 全体への拡張．}
  $y\in\mathcal{Y}\setminus N$ を固定し，$m_y\colon D\to\R$ を $m_y(f):=l_f(y)$ で定める．
  (a) より $m_y$ は有理係数について線形であり，(c) より
  \[
    \abs{m_y(f)}\leq\norm{f}_\infty\qquad(f\in D).\tag{3}
  \]
  $f\in C(K)$ に対し，$\norm{f_k-f}_\infty\to0$ となる $f_k\in D$ を取る．
  $f_k-f_j\in D$ なので，(3) と三角不等式より
  \[
    \abs{m_y(f_k)-m_y(f_j)}=\abs{m_y(f_k-f_j)}\leq\norm{f_k-f}_\infty+\norm{f-f_j}_\infty\longrightarrow0
  \]
  となる．したがって $(m_y(f_k))_k$ は $\R$ の Cauchy 列であり，$\R$ の完備性から収束する．そこで
  \[
    m_y(f):=\lim_{k\to\infty}m_y(f_k)
  \]
  と定める．別の近似列 $g_k\to f$ に対しても
  $\abs{m_y(f_k)-m_y(g_k)}\leq\norm{f_k-f}_\infty+\norm{f-g_k}_\infty\to0$ なので，
  極限は近似列によらない．$f\in D$ なら定数列を取れるので，元の値と一致する．
  極限を取ることで，有理係数についての線形性と
  $\abs{m_y(f)-m_y(g)}\leq\norm{f-g}_\infty$（$f,g\in C(K)$）が保たれる．
  実数 $a$ に有理数列 $q_j\to a$ を取れば
  $\abs{m_y(af)-q_jm_y(f)}=\abs{m_y(af)-m_y(q_jf)}\leq\abs{a-q_j}\norm{f}_\infty\to0$ なので，
  $m_y\colon C(K)\to\R$ は実線形である．

  (b) より $m_y(1)=l_1(y)=1$ であり，(3) の極限から
  $\abs{m_y(f)}\leq\norm{f}_\infty$ はすべての $f\in C(K)$ で成り立つ．
  $f\geq0$ として $M:=\norm{f}_\infty$ とおくと，$0\leq M\,1-f\leq M\,1$ なので，
  \[
    m_y(f)=M-m_y(M\,1-f)\geq M-\norm{M\,1-f}_\infty\geq0.
  \]
  よって $m_y$ は正値である．この議論では $D$ が最大・最小について閉じている必要はない．

  \textbf{5. Riesz--Markov--Kakutani の表現定理による測度の構成．}
  以上より，$m_y$ は $C(K)$ 上の正の線形汎関数で $m_y(1)=1$ をみたす．
  Riesz--Markov--Kakutani の表現定理（定理~\ref{thm:meas-riesz-representation}）により，
  \[
    m_y(f)=\int_{K}f\,\d\lambda_y\qquad(f\in C(K))
  \]
  となる $\lambda_y\in\Prob(K)$ がただ一つ存在する．
  $y\in N$ に対しては，$x_0\in K$ を一つ固定して $\lambda_y:=\delta_{x_0}$ と定める．

  ここで $m_y(f)=l_f(y)$ を $y\notin N$ のすべてで使えるのは，まず $f\in D$ に限る．
  一般の $f\in C(K)$ について，最初に選んだ $l_f$ との一致を同じ $N$ の外で仮定してはならない．
  残りは，この族 $(\lambda_y)$ が補題の二つの条件をみたすことの確認である．

  \textbf{6. 連続関数に対する可測性と積分公式．}
  まず $f\in D$ ならば
  \[
    \int_Kf\,\d\lambda_y=
    \begin{cases}
      l_f(y)&(y\notin N),\\
      f(x_0)&(y\in N).
    \end{cases}
  \]
  $l_f$ と $N$ の Borel 可測性から左辺は $y$ の Borel 可測関数であり，
  $\nu(N)=0$ と (2) より
  \[
    \int_B\left(\int_K f\,\d\lambda_y\right)\d\nu(y)
    =\nu_f(B)=\int_{K\times B}f(x)\,\d\pi(x,y).
  \]
  次に $f\in C(K)$ に対し，$f_k\in D$，$\norm{f_k-f}_\infty\to0$ を取る．
  すべての $y$ で $\lambda_y(K)=1$ なので，
  \[
    \left|\int_K f_k\,\d\lambda_y-\int_Kf\,\d\lambda_y\right|
    \leq\norm{f_k-f}_\infty.
  \]
  よって $y\mapsto\int_Kf\,\d\lambda_y$ は可測関数の各点極限として Borel 可測である．
  任意の $B\in\Borel(\mathcal{Y})$ について，さらに
  \[
    \begin{aligned}
      \left|\int_B\left(\int_K(f_k-f)\,\d\lambda_y\right)\d\nu(y)\right|
      &\leq\norm{f_k-f}_\infty\,\nu(B)\longrightarrow0,\\
      \left|\int_{K\times B}(f_k(x)-f(x))\,\d\pi(x,y)\right|
      &\leq\norm{f_k-f}_\infty\,\nu(B)\longrightarrow0.
    \end{aligned}
  \]
  ここでは $\nu(B)<\infty$ を用いた．したがって
  \[
    \int_B\left(\int_Kf\,\d\lambda_y\right)\d\nu(y)
    =\int_{K\times B}f(x)\,\d\pi(x,y).
  \]

  \textbf{7. 閉集合と Borel 集合への拡張．}
  空でない閉集合 $A\subset K$ に対して
  \[
    u_n(x):=\max\{1-n\,d(x,A),0\}
  \]
  とおく．補題~\ref{lem:meas-closed-approx}より $u_n\in C(K)$，
  $0\leq u_n\leq1$，$u_n\downarrow\mathbf{1}_A$ である．
  $\lambda_y(K)=1$ なので有界収束定理から
  $\int_Ku_n\,\d\lambda_y\to\lambda_y(A)$ となり，$y\mapsto\lambda_y(A)$ は Borel 可測である．
  6 の積分公式に $f=u_n$ を代入し，
  $\pi(K\times B)=\nu(B)<\infty$ と $0\leq\int_Ku_n\,\d\lambda_y\leq1$ を用いて
  両辺で再び有界収束定理を適用すれば
  \[
    \pi(A\times B)=\int_B\lambda_y(A)\,\d\nu(y)
  \]
  を得る．$A=\emptyset$ では可測性もこの等式も明らかである．

  可測性とすべての $B$ に対する積分公式をみたす $A\in\Borel(K)$ の全体を $\mathcal{D}$ とする．
  $\lambda_y(K)=1$ と $\pi(K\times B)=\nu(B)$ より $K\in\mathcal{D}$ である．
  $A\in\mathcal{D}$ ならば $\lambda_y(K\setminus A)=1-\lambda_y(A)$ は可測であり，
  \[
    \pi((K\setminus A)\times B)
    =\nu(B)-\pi(A\times B)=\int_B(1-\lambda_y(A))\,\d\nu(y)
  \]
  より $K\setminus A\in\mathcal{D}$ である．
  互いに素な $A_n\in\mathcal{D}$ に対し，
  $\lambda_y(\bigcup_nA_n)=\sum_n\lambda_y(A_n)$ は可測であり，可算加法性と単調収束定理より
  \[
    \pi\left(\left(\bigcup_nA_n\right)\times B\right)
    =\sum_n\int_B\lambda_y(A_n)\,\d\nu(y)
    =\int_B\lambda_y\left(\bigcup_nA_n\right)\,\d\nu(y).
  \]
  したがって $\mathcal{D}$ は $\lambda$-系（Dynkin 系）である．
  閉集合の族 $\mathcal{F}$ は有限共通部分について閉じているので $\pi$-系であり，
  $\sigma(\mathcal{F})=\Borel(K)$ である．前段より $\mathcal{F}\subset\mathcal{D}$ だから，
  Dynkin の定理（定理~\ref{thm:meas-pi-lambda}）により
  $\Borel(K)=\sigma(\mathcal{F})\subset\mathcal{D}\subset\Borel(K)$ を得る．
\end{proof}

\subsection{一般の Polish 空間の場合}

\begin{proof}[定理~\ref{thm:meas-disintegration}の存在の証明]
  $x_0\in\mathcal{X}$ を固定する．

  \textbf{1. compact 集合の列による分割．}
  第1周辺分布を $\mu(A):=\pi(A\times\mathcal{Y})$ とおく．
  $\{\mu\}$ 内の列は定数列で $\mu$ に弱収束するので，既出の Prokhorov の定理
  （定理~\ref{thm:meas-prokhorov}，$1\Rightarrow2$）より $\{\mu\}$ は tight である．
  $K_0:=\emptyset$ とし，tightness で得た compact 集合と $K_{n-1}$ の合併を取って，
  $K_{n-1}\subset K_n$，$\mu(\mathcal{X}\setminus K_n)\leq1/n$ となる compact 集合の列を作る．
  $K:=\bigcup_nK_n$，$E_n:=K_n\setminus K_{n-1}$ とおけば，$\mu(\mathcal{X}\setminus K)=0$ であり，
  $E_n$ は $K$ を互いに素に分割する．したがって，$\nu_n(B):=\pi(E_n\times B)$ とおくと
  \[
    \pi(A\times B)=\sum_n\pi((A\cap E_n)\times B),\qquad
    \nu(B)=\sum_n\nu_n(B).
  \]

  \textbf{2. 各部分での分解．}
  $\nu_n\neq0$ の各 $n$ では，$\pi$ の $E_n\times\mathcal{Y}$ への制限
  $C\mapsto\pi(C\cap(E_n\times\mathcal{Y}))$ を $K_n\times\mathcal{Y}$ 上の有限測度とみて，
  補題~\ref{lem:dis-compact}を $K=K_n$ として適用する．
  $E_n$ 自身は compact とは限らないので，測度を載せる空間には $K_n$ を使う．
  この補題は有限測度について示したので，正規化は不要である．
  得た $K_n$ 上の確率測度を $\widetilde{\lambda}_y^{(n)}$ とし，
  $\lambda_y^{(n)}(A):=\widetilde{\lambda}_y^{(n)}(A\cap K_n)$ によって
  $\mathcal{X}$ 上に延長する（注意~\ref{rem:meas-subspace-borel}）．このとき
  $y\mapsto\lambda_y^{(n)}(A)$ は Borel 可測で，
  \[
    \pi((A\cap E_n)\times B)=\int_B\lambda_y^{(n)}(A)\,\d\nu_n(y)
    \qquad(A\in\Borel(\mathcal{X}),\ B\in\Borel(\mathcal{Y}))
  \]
  である．$\nu_n=0$ の $n$ では $\lambda_y^{(n)}:=\delta_{x_0}$ とすれば，上式は両辺 $0$ で成り立つ．

  \textbf{3. 重みを付けて足し合わせる．}
  $\nu_n(B)\leq\nu(B)$ なので，Radon--Nikodym の定理（定理~\ref{thm:meas-radon-nikodym}）から
  非負の Borel 可測な密度 $h_n:=\d\nu_n/\d\nu$ を取る．
  単調収束定理より $\int_B\sum_nh_n\,\d\nu=\sum_n\nu_n(B)=\nu(B)$ である．
  $s:=\sum_nh_n$ とおくと，特に $\int_{\mathcal{Y}}s\,\d\nu=1$ なので
  $N_\infty:=\{s=\infty\}$ は Borel 零集合である．
  $N_\infty$ 上だけ値を $0$ にした有限値関数 $\widetilde{s}$ についても
  $\int_B\widetilde{s}\,\d\nu=\nu(B)=\int_B1\,\d\nu$ である．
  一意性の議論（定理~\ref{thm:meas-radon-nikodym}の証明の3）より $\widetilde{s}=1$ が $\nu$-a.e. に成り立つ．
  したがって，元の和についても $s=1$ が $\nu$-a.e. に成り立つ．
  Borel 零集合 $N_0:=\{s\neq1\}$ を用いて
  \[
    \lambda_y(A):=
    \begin{cases}
      \sum_nh_n(y)\,\lambda_y^{(n)}(A) & (y\notin N_0),\\
      \delta_{x_0}(A) & (y\in N_0)
    \end{cases}
  \]
  と定める．$y\notin N_0$ では，確率測度を和が $1$ の非負の重みで足し合わせているので，
  $\lambda_y$ は確率測度である（可算加法性は非負二重級数の和の交換による）．
  各項と $N_0$ の可測性から条件 1 が成り立つ．
  $\nu(N_0)=0$，系~\ref{cor:rn-density-integral}と単調収束定理より
  \[
    \begin{aligned}
      \int_B\lambda_y(A)\,\d\nu(y)
      &=\sum_n\int_B\lambda_y^{(n)}(A)\,h_n(y)\,\d\nu(y)
      =\sum_n\int_B\lambda_y^{(n)}(A)\,\d\nu_n(y)\\
      &=\sum_n\pi((A\cap E_n)\times B)=\pi(A\times B).
    \end{aligned}
  \]
  よって条件 2 も成り立つ．
\end{proof}

\subsection{一意性}

$A$ ごとに例外集合を取るだけでなく，一つの零集合の外で\emph{測度として一致する}ことを示す．

\begin{proof}[定理~\ref{thm:meas-disintegration}の一意性の証明]
  $(\lambda_y)$，$(\lambda'_y)$ がともに条件 1，2 をみたすとする．
  $A\in\Borel(\mathcal{X})$ を固定すると，$\lambda_y(A)$，$\lambda'_y(A)$ は $y$ の有界可測関数であり，
  条件 2 よりすべての $B$ で同じ積分値をもつ．
  一意性の議論（定理~\ref{thm:meas-radon-nikodym}の証明の3）より，
  $N_A:=\{y\mid\lambda_y(A)\neq\lambda'_y(A)\}$ は Borel 零集合である．

  補題~\ref{lem:meas-countable-generator}により，
  $\Borel(\mathcal{X})$ を生成する可算な $\pi$-系 $\mathcal{E}$ を取り，
  $N:=\bigcup_{A\in\mathcal{E}}N_A$ とおく．$\mathcal{E}$ は可算なので $N$ は Borel 零集合である．
  $y\notin N$ では $\lambda_y,\lambda'_y$ は $\mathcal{E}$ 上で一致する確率測度なので，
  一致定理（定理~\ref{thm:meas-uniqueness}）より $\lambda_y=\lambda'_y$ である．
\end{proof}
