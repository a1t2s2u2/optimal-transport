\chapter{Riesz の表現定理}
\label{ch:riesz-representation}

第\ref{ch:disintegration}章の測度分解定理では，
$C(K)$ 上の正の線形汎関数を確率測度による積分として表す．
本章では compact 距離空間の場合に Riesz--Markov--Kakutani の表現定理を証明する．
証明の方針は Rudin \emph{Real and Complex Analysis}（第3版）Theorem 2.14 に従い，
開集合の測度を汎関数の値の上限で定める．
ただし距離空間であることを用いて，
測度の構成を Carathéodory の定理（定理~\ref{thm:meas-caratheodory}）に帰着させて簡略化する．
$C(K)$ と $\lVert\cdot\rVert_\infty$ の記法は注意~\ref{rem:meas-compact-continuous}による．

\begin{theorem}{Riesz--Markov--Kakutani の表現定理（compact 距離空間の場合）}{meas-riesz-representation}
  $K$ を空でない compact 距離空間とする．
  実線形写像 $L\colon C(K)\to\R$ が
  \[
    f\geq0\ \Longrightarrow\ L(f)\geq0,\qquad L(1)=1
  \]
  をみたすならば，ある唯一の $\rho\in\Prob(K)$ により
  \[
    L(f)=\int_Kf\,\d\rho\qquad(f\in C(K))
  \]
  と表せる．条件 $f\geq0\Rightarrow L(f)\geq0$ をみたす実線形写像を
  \textbf{正の線形汎関数}という．
\end{theorem}

本章の残りでは $K$ と $L$ を定理~\ref{thm:meas-riesz-representation}のとおりとし，
証明をいくつかの補題に分けて与える．

\section{準備}

\begin{lemma}{正の線形汎関数の単調性と有界性}{rz-monotone}
  $f,g\in C(K)$ に対し，$f\leq g$ ならば $L(f)\leq L(g)$ であり，
  $\abs{L(f)}\leq\lVert f\rVert_\infty$ である．
\end{lemma}

\begin{proof}
  $g-f\geq0$ より $L(g)-L(f)=L(g-f)\geq0$ である．
  $-\lVert f\rVert_\infty\leq f\leq\lVert f\rVert_\infty$ に単調性と $L(1)=1$ を用いれば後半を得る．
\end{proof}

\begin{definition}{開集合に従属する関数}{rz-subordinate}
  開集合 $U\subset K$ と $g\in C(K)$ について，
  \[
    0\leq g\leq1,\qquad g=0\ \text{（$K\setminus U$ 上）}
  \]
  が成り立つとき，$g\prec U$ と書く．
\end{definition}

\begin{lemma}{1 の分割}{rz-partition}
  $F\subset K$ を閉集合，$U_1,\dots,U_m\subset K$ を $F\subset U_1\cup\cdots\cup U_m$ をみたす開集合とし，
  $g\in C(K)$ は $0\leq g\leq1$ かつ $K\setminus F$ 上で $g=0$ をみたすとする．
  このとき $g_i\prec U_i$（$i=1,\dots,m$）で $g=g_1+\cdots+g_m$ となるものが存在する．
\end{lemma}

\begin{proof}
  $F=\emptyset$ なら $g=0$ なので $g_i:=0$ とすればよい．$F\neq\emptyset$ とする．
  各 $i$ について，$U_i\neq K$ なら $\phi_i(x):=d(x,K\setminus U_i)$，
  $U_i=K$ なら $\phi_i:=1$ とおく．
  補題~\ref{lem:meas-distance-to-set}より $\phi_i$ は連続で，
  $\phi_i(x)>0\iff x\in U_i$ である．
  $\Phi:=\phi_1+\cdots+\phi_m$ は $F$ 上で正であり，
  $F$ は compact（注意~\ref{rem:meas-compact-facts}）なので
  $c:=\min_F\Phi>0$ である．
  \[
    g_i:=\frac{g\,\phi_i}{\max\{\Phi,c\}}
  \]
  とおくと，$g_i$ は連続で非負であり，$U_i$ の外では $\phi_i=0$ より $g_i=0$ である．
  $F$ 上では $\max\{\Phi,c\}=\Phi$ なので $\sum_ig_i=g$ であり，
  $F$ の外では $g=0$ なので $\sum_ig_i=0=g$ である．
  特に $0\leq g_i\leq g\leq1$ となり，$g_i\prec U_i$ である．
\end{proof}

\begin{lemma}{切り落とし}{rz-cutoff}
  $g\in C(K)$，$0\leq g\leq1$，$0<\varepsilon<1$ に対し
  $g_\varepsilon:=\max\{g-\varepsilon,0\}$ とおくと，
  $g_\varepsilon\in C(K)$，$0\leq g_\varepsilon\leq g$，
  $L(g)-\varepsilon\leq L(g_\varepsilon)$ であり，
  $g_\varepsilon$ は閉集合 $F:=\{g\geq\varepsilon\}$ の外で $0$ である．
\end{lemma}

\begin{proof}
  $0\leq g-g_\varepsilon\leq\varepsilon$ と補題~\ref{lem:rz-monotone}より
  $L(g)-L(g_\varepsilon)\leq\varepsilon$ である．
  $F$ は連続関数による閉集合の逆像なので閉集合であり，
  $F$ の外では $g<\varepsilon$ より $g_\varepsilon=0$ である．
\end{proof}

\section{外測度の構成}

\begin{definition}{$L$ から定まる外測度}{rz-outer-measure}
  開集合 $U\subset K$ に対し
  \[
    \rho^*(U):=\sup\{L(g)\mid g\in C(K),\ g\prec U\}
  \]
  と定め，一般の $E\subset K$ に対し
  \[
    \rho^*(E):=\inf\{\rho^*(U)\mid U\text{ は開集合で }E\subset U\}
  \]
  と定める．
\end{definition}

開集合 $U\subset V$ について $g\prec U$ ならば $g\prec V$ なので，
開集合上で $\rho^*$ は単調であり，二つの定義は開集合上で一致する．
また $g\prec U$ なら $0\leq L(g)\leq L(1)=1$ なので $0\leq\rho^*(E)\leq1$ であり，
$g\prec\emptyset$ は $g=0$ に限るので $\rho^*(\emptyset)=0$，
$1\prec K$ より $\rho^*(K)=1$ である．

\begin{lemma}{開集合上の可算劣加法性}{rz-open-subadditive}
  開集合 $U_1,U_2,\ldots$ と $U:=\bigcup_nU_n$ に対し
  $\rho^*(U)\leq\sum_n\rho^*(U_n)$ である．
\end{lemma}

\begin{proof}
  $g\prec U$ と $0<\varepsilon<1$ を任意に取り，補題~\ref{lem:rz-cutoff}の $g_\varepsilon$，$F$ を考える．
  $g\prec U$ より $F\subset U$ である．
  $F$ は compact（注意~\ref{rem:meas-compact-facts}）なので，
  相異なる有限個 $n_1,\dots,n_m$ で $F\subset U_{n_1}\cup\cdots\cup U_{n_m}$ となる．
  補題~\ref{lem:rz-partition}で $g_\varepsilon=\sum_ig_i$，$g_i\prec U_{n_i}$ と分ければ
  \[
    L(g)-\varepsilon\leq L(g_\varepsilon)=\sum_{i=1}^mL(g_i)
    \leq\sum_{i=1}^m\rho^*(U_{n_i})\leq\sum_n\rho^*(U_n).
  \]
  $\varepsilon\to0$ とし，$g\prec U$ について上限を取ればよい．
\end{proof}

\begin{lemma}{$\rho^*$ は外測度である}{rz-is-outer-measure}
  $\rho^*$ は $K$ 上の外測度（定義~\ref{def:meas-outer-measure}）である．
\end{lemma}

\begin{proof}
  $\rho^*(\emptyset)=0$ と単調性は定義から明らかである．
  $E=\bigcup_nE_n$ とし，$\varepsilon>0$ を取る．
  各 $n$ について開集合 $U_n\supset E_n$ を
  $\rho^*(U_n)\leq\rho^*(E_n)+\varepsilon2^{-n}$ となるように選ぶ．
  $\bigcup_nU_n$ は $E$ を含む開集合なので，補題~\ref{lem:rz-open-subadditive}より
  \[
    \rho^*(E)\leq\rho^*\left(\bigcup_nU_n\right)\leq\sum_n\rho^*(U_n)
    \leq\sum_n\rho^*(E_n)+\varepsilon.
  \]
  $\varepsilon$ は任意なので可算劣加法性を得る．
\end{proof}

\begin{lemma}{開集合の可測性}{rz-open-measurable}
  任意の開集合 $U\subset K$ は $\rho^*$-可測である．
\end{lemma}

\begin{proof}
  任意の $E\subset K$ について $\rho^*(E)\geq\rho^*(E\cap U)+\rho^*(E\setminus U)$ を示す．

  \emph{$E=W$ が開集合の場合．}
  $\varepsilon\in(0,1)$ を取る．$W\cap U$ は開集合なので，
  $g\prec W\cap U$ を $L(g)\geq\rho^*(W\cap U)-\varepsilon$ となるように取り，
  補題~\ref{lem:rz-cutoff}の $g_\varepsilon$，$F$ を考える．
  $F\subset W\cap U$ であり，$W\setminus F$ は $W\setminus U$ を含む開集合である．
  そこで $k\prec W\setminus F$ を $L(k)\geq\rho^*(W\setminus F)-\varepsilon$ となるように取る．
  $g_\varepsilon$ は $F$ の外で，$k$ は $F$ 上で $0$ なので，
  $0\leq g_\varepsilon+k\leq1$ であり，
  $W$ の外では両者とも $0$ である．よって $g_\varepsilon+k\prec W$ であり，
  \[
    \rho^*(W)\geq L(g_\varepsilon)+L(k)
    \geq\rho^*(W\cap U)-2\varepsilon+\rho^*(W\setminus F)-\varepsilon
    \geq\rho^*(W\cap U)+\rho^*(W\setminus U)-3\varepsilon.
  \]
  $\varepsilon\to0$ として主張を得る．

  \emph{一般の場合．}
  $E\subset W$ となる開集合 $W$ について，前段と単調性より
  \[
    \rho^*(W)\geq\rho^*(W\cap U)+\rho^*(W\setminus U)\geq\rho^*(E\cap U)+\rho^*(E\setminus U).
  \]
  $W$ について下限を取ればよい．
\end{proof}

\section{表現定理の証明}

以上の補題を用いて定理~\ref{thm:meas-riesz-representation}を証明する．

\begin{proof}[定理~\ref{thm:meas-riesz-representation}の証明]
  \textbf{1. 測度の構成．}
  補題~\ref{lem:rz-is-outer-measure}と Carathéodory の定理（定理~\ref{thm:meas-caratheodory}）より，
  $\rho^*$-可測集合の全体 $\mathcal{M}$ は $\sigma$-代数であり，$\rho^*$ は $\mathcal{M}$ 上の測度である．
  補題~\ref{lem:rz-open-measurable}より $\mathcal{M}$ は開集合をすべて含むので
  $\Borel(K)\subset\mathcal{M}$ である．
  $\rho$ を $\rho^*$ の $\Borel(K)$ への制限とすれば，$\rho(K)=1$ より $\rho\in\Prob(K)$ である．
  定義から，開集合 $U$ と $g\prec U$ について
  \[
    L(g)\leq\rho(U)
  \]
  が成り立つ．

  \textbf{2. $0\leq f\leq1$ のとき $L(f)\leq\int_Kf\,\d\rho$．}
  $n\in\N$ を取り，$k=1,\dots,n$ に対して
  \[
    f_k:=\min\left\{\max\left\{f-\frac{k-1}n,0\right\},\frac1n\right\},\qquad
    U_k:=\left\{f>\frac{k-1}n\right\},\qquad
    F_k:=\left\{f\geq\frac kn\right\}
  \]
  とおく．$f_k\in C(K)$，$U_k$ は開集合，$F_k$ は閉集合であり，
  $0\leq f\leq1$ より $f=f_1+\cdots+f_n$ である．
  $nf_k$ は $0\leq nf_k\leq1$ をみたし，$U_k$ の外で $0$，$F_k$ 上で $1$ なので，
  \[
    nf_k\prec U_k,\qquad nf_k\geq\mathbf{1}_{F_k}
  \]
  である．したがって，$L(nf_k)\leq\rho(U_k)$ と $\rho(F_k)\leq\int_Knf_k\,\d\rho$ を得る．
  また $U_1\subset K$ より $\rho(U_1)\leq1$，$k\geq2$ では $U_k\subset F_{k-1}$ である．よって
  \[
    \begin{aligned}
      L(f)=\sum_{k=1}^nL(f_k)
      &\leq\frac1n\sum_{k=1}^n\rho(U_k)
      \leq\frac1n+\frac1n\sum_{k=1}^{n-1}\rho(F_k)\\
      &\leq\frac1n+\sum_{k=1}^{n-1}\int_Kf_k\,\d\rho
      \leq\frac1n+\int_Kf\,\d\rho.
    \end{aligned}
  \]
  $n\to\infty$ として主張を得る．

  \textbf{3. 一般の $f\in C(K)$ で $L(f)=\int_Kf\,\d\rho$．}
  $m:=\lVert f\rVert_\infty$ とする．$m=0$ なら両辺とも $0$ である．
  $m>0$ なら $g:=(f+m)/(2m)$ は $0\leq g\leq1$ をみたすので，
  2 と $L(1)=1=\rho(K)$ より
  \[
    \frac{L(f)+m}{2m}=L(g)\leq\int_Kg\,\d\rho=\frac{\int_Kf\,\d\rho+m}{2m}
  \]
  となり，$L(f)\leq\int_Kf\,\d\rho$ を得る．$-f$ にも適用すれば等号が成り立つ．

  \textbf{4. 一意性．}
  $\rho'\in\Prob(K)$ も同じ表示をみたすとする．
  空でない閉集合 $F$ について補題~\ref{lem:meas-closed-approx}の $\phi_k$ を取ると，
  有界収束定理（定理~\ref{thm:meas-bounded-convergence}）より
  \[
    \rho(F)=\lim_k\int_K\phi_k\,\d\rho=\lim_kL(\phi_k)=\lim_k\int_K\phi_k\,\d\rho'=\rho'(F)
  \]
  である．閉集合全体は $\Borel(K)$ を生成する $\pi$-系なので，
  一致定理（定理~\ref{thm:meas-uniqueness}）より $\rho=\rho'$ である．
\end{proof}
