\begin{definition}{連続関数}{continuous}
  関数 $f \colon \R \to \R$ が点 $a$ で\textbf{連続}であるとは，
  任意の $\varepsilon > 0$ に対してある $\delta > 0$ が存在し，
  $\abs{x - a} < \delta$ ならば $\abs{f(x) - f(a)} < \varepsilon$
  が成り立つことをいう．
\end{definition}

\begin{theorem}{中間値の定理}{ivt}
  $f$ が閉区間 $[a,b]$ 上で連続であり，$f(a) \neq f(b)$ ならば，
  $f(a)$ と $f(b)$ の間の任意の値 $c$ に対して，
  $f(\xi) = c$ となる $\xi \in (a,b)$ が存在する．
\end{theorem}

\begin{proposition}{有界性}{bounded}
  閉区間上の連続関数は有界である．
\end{proposition}
